    53→const DURATION_MAX = 60_000

    55→/** Effective duration of a frame for badges/scrubber (mirrors core math). */
    56→function frameMs(clip: Clip, index: number): number {
    57→  const f = clip.frames[index]
    58→  if (!f) return 0
    59→  if (f.durationMs > 0) return f.durationMs
    60→  const fps = Math.min(60, Math.max(1, Math.round(clip.fps || 8)))
    61→  return Math.max(1, Math.round(1000 / fps))
    62→}
    63→
    64→function formatMs(ms: number): string {
    65→  return ms >= 1000 ? `${(ms / 1000).toFixed(ms % 1000 === 0 ? 0 : 2)}s` : `${ms}ms`
    66→}

    77→  h,

    79→  playing,
    80→  onSelect,
    81→  onDragStart,
    82→  onDragOver,
    83→  onDrop,
    84→  onDragEnd,
    85→}: {
    86→  clip: Clip
    87→  frame: Frame
    88→  index: number
    89→  active: boolean
    90→  dropBefore: boolean
    91→  w: number
    92→  h: number
    93→  tick: number
    94→  playing: boolean
    95→  onSelect: () => void

   128→        onClick={onSelect}

   136→      >
   137→        <canvas ref={ref} width={56} height={56} className="pixelforge-checker h-11 w-11 rounded-sm sm:h-14 sm:w-14" aria-hidden />
   138→        <span className="absolute bottom-0.5 right-1 font-mono text-[9px] text-zinc-400">{index + 1}</span>

   140→      {/* duration badge → popover editor (Task 6 §2) */}
   141→      <Popover open={durOpen} onOpenChange={setDurOpen}>
   142→        <PopoverTrigger asChild>
   143→          <button
   144→            type="button"
   145→            className={`absolute -top-1.5 -right-0.5 flex h-4 items-center gap-0.5 rounded-full border px-1 font-mono text-[8px] leading-none transition-colors ${
   146→              ms > 0
   147→                ? 'border-amber-500/60 bg-amber-500/15 text-amber-300 hover:bg-amber-500/25'
   148→                : 'border-zinc-700 bg-zinc-900 text-zinc-500 hover:text-zinc-300'
   149→            }`}
   150→            aria-label={`Edit duration of frame ${index + 1}`}
   151→            title="Frame duration (0 = inherit clip FPS)"
   152→          >
   153→            <Timer className="h-2 w-2" aria-hidden />
   154→            {ms > 0 ? `${ms}` : 'auto'}
   155→          </button>
   156→        </PopoverTrigger>
   157→        <PopoverContent side="top" align="center" className="w-56 p-3">
   158→          <p className="mb-2 font-mono text-[10px] uppercase tracking-wider text-zinc-400">
   159→            Frame {index + 1} duration
   160→          </p>
   161→          <StepperInput
   162→            value={ms}
   163→            onChange={(v) => setFrameDuration(index, Math.min(DURATION_MAX, Math.max(0, v)))}
   164→            min={0}
   165→            max={DURATION_MAX}
   166→            step={10}
   167→            suffix="ms"
   168→            ariaLabel={`Duration of frame ${index + 1} in milliseconds`}
   169→            size="sm"
   170→          />
   171→          <p className="mt-1.5 text-[10px] leading-snug text-zinc-500">
   172→            {ms > 0 ? (
   173→              <>Overrides the clip FPS for this frame.</>
   174→            ) : (
   175→              <>
   176→                <span className="text-amber-400">auto</span> — inherits the clip FPS
   177→                ({frameMs(clip, index)}ms at {clip.fps}fps).
   178→              </>
   179→            )}
   180→          </p>
   181→          {ms > 0 && (
   182→            <Button
   183→              variant="ghost"
   184→              size="sm"
   185→              className="mt-1 h-6 w-full text-[10px] text-zinc-400 hover:text-teal-300"
   186→              onClick={() => setFrameDuration(index, 0)}
   187→            >
   188→              Reset to auto ({clip.fps}fps)
   189→            </Button>
   190→          )}
   191→        </PopoverContent>
   192→      </Popover>
   193→    </div>
   194→  )
   195→}

   199→function Scrubber({ clip, activeIndex }: { clip: Clip; activeIndex: number }) {
   200→  const trackRef = useRef<HTMLDivElement>(null)
   201→  const markerRef = useRef<HTMLDivElement>(null)
   202→  const [hoverMs, setHoverMs] = useState<number | null>(null)
   203→
   204→  // Playhead marker follows the shared engine via direct DOM writes — no
   205→  // React re-render per animation tick (smooth at any fps).
   206→  useEffect(
   207→    () =>
   208→      editorPlayback.onPosition((pos, total) => {
   209→        const el = markerRef.current
   210→        if (el) el.style.left = `${total > 0 ? Math.min(100, (pos / total) * 100) : 0}%`
   211→      }),
   212→    []
   213→  )
   214→
   215→  const msFromEvent = (clientX: number): number => {
   216→    const track = trackRef.current
   217→    if (!track) return 0
   218→    const rect = track.getBoundingClientRect()
   219→    const total = clip.frames.reduce((s, _, i) => s + frameMs(clip, i), 0)
   220→    const ratio = Math.min(1, Math.max(0, (clientX - rect.left) / rect.width))
   221→    return Math.round(ratio * total)
   222→  }
   223→
   224→  const scrub = (clientX: number) => editorPlayback.scrubToMs(msFromEvent(clientX))
   225→
   226→  return (
   227→    <div className="px-2 pb-1">
   228→      <div
   229→        ref={trackRef}
   230→        role="slider"
   231→        aria-label="Timeline position"
   232→        aria-valuemin={1}
   233→        aria-valuemax={clip.frames.length}
   234→        aria-valuenow={activeIndex + 1}
   235→        aria-valuetext={`Frame ${activeIndex + 1} of ${clip.frames.length}`}
   236→        tabIndex={0}
   237→        className="relative flex h-5 cursor-pointer touch-none items-stretch overflow-hidden rounded-md border border-zinc-800 bg-zinc-900"
   238→        onPointerDown={(e) => {
   239→          e.currentTarget.setPointerCapture(e.pointerId)
   240→          scrub(e.clientX)
   241→        }}
   242→        onPointerMove={(e) => {
   243→          if (e.buttons & 1) scrub(e.clientX)
   244→          setHoverMs(msFromEvent(e.clientX))
   245→        }}
   246→        onPointerLeave={() => setHoverMs(null)}
   247→        onKeyDown={(e) => {
   248→          if (e.key === 'ArrowRight') editorPlayback.step(1)
   249→          else if (e.key === 'ArrowLeft') editorPlayback.step(-1)
   250→        }}
   251→      >
   252→        {/* proportional frame segments (width = effective duration) */}
   253→        {clip.frames.map((f, i) => (
   254→          <div
   255→            key={f.id}
   256→            title={`Frame ${i + 1} · ${frameMs(clip, i)}ms`}
   257→            style={{ flexGrow: frameMs(clip, i), flexBasis: 0 }}
   258→            className={`min-w-[3px] border-r border-zinc-950/80 last:border-r-0 ${
   259→              i === activeIndex ? 'bg-teal-700/50' : i % 2 === 0 ? 'bg-zinc-800/70' : 'bg-zinc-800/40'
   260→            }`}
   261→          />
   262→        ))}
   263→        {/* playhead marker */}
   264→        <div
   265→          ref={markerRef}
   266→          aria-hidden
   267→          className="pointer-events-none absolute inset-y-0 left-0 w-0.5 bg-teal-400 shadow-[0_0_6px_rgba(45,212,191,0.8)]"
   268→        />
   269→      </div>
   270→      <div className="mt-0.5 flex justify-between font-mono text-[9px] text-zinc-500">
   271→        <span>
   272→          {hoverMs != null ? `scrub ${formatMs(hoverMs)}` : `${clip.frames.length} frames`}
   273→        </span>
   274→        <span>{formatMs(clip.frames.reduce((s, _, i) => s + frameMs(clip, i), 0))} cycle</span>
   275→      </div>
   276→    </div>
   277→  )
   278→}

   390→  const activeFrameIndex = useEditor((s) => s.activeFrameIndex)

   393→  const timelineOpen = useEditor((s) => s.timelineOpen)
   394→  const selectClip = useEditor((s) => s.selectClip)
   395→  const selectFrame = useEditor((s) => s.selectFrame)
   396→  const setTimelineOpen = useEditor((s) => s.setTimelineOpen)
   397→  const addFrame = useEditor((s) => s.addFrame)
   398→  const deleteFrame = useEditor((s) => s.deleteFrame)
   399→  const moveFrame = useEditor((s) => s.moveFrame)
   400→  const addClip = useEditor((s) => s.addClip)
   401→  const deleteClip = useEditor((s) => s.deleteClip)
   402→  const setClipProp = useEditor((s) => s.setClipProp)
   403→  const [renamingClip, setRenamingClip] = useState<string | null>(null)
   404→  // drag-and-drop reorder state
   405→  const dragFromRef = useRef<number | null>(null)
   406→  const [dropTarget, setDropTarget] = useState<number | null>(null)
   407→  const stripRef = useRef<HTMLDivElement>(null)
   408→
   409→  const clip = getActiveClip(useEditor.getState())
   410→  const frameIndex = Math.min(activeFrameIndex, clip.frames.length - 1)
   411→
   412→  // keep the active thumb visible while playing / scrubbing
   413→  useEffect(() => {
   414→    const strip = stripRef.current
   415→    if (!strip) return
   416→    const el = strip.querySelector<HTMLElement>(`[data-frame-index="${frameIndex}"]`)
   417→    el?.scrollIntoView({ block: 'nearest', inline: 'nearest', behavior: playing ? 'auto' : 'smooth' })
   418→  }, [frameIndex, playing])
   419→
   420→  const commitDrop = (target: number) => {
   421→    const from = dragFromRef.current
   422→    dragFromRef.current = null
   423→    setDropTarget(null)
   424→    if (from == null || from === target) return
   425→    moveFrame(from, target)
   426→  }
   427→
   428→  const transport = (
   429→    <>
   430→      <Tooltip>
   431→        <TooltipTrigger asChild>
   432→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400" onClick={() => editorPlayback.restart()} aria-label="Restart from first frame">
   433→            <ChevronsLeft className="h-4 w-4" />
   434→          </Button>
   435→        </TooltipTrigger>
   436→        <TooltipContent>Restart</TooltipContent>
   437→      </Tooltip>
   438→      <Tooltip>
   439→        <TooltipTrigger asChild>
   440→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400" onClick={() => editorPlayback.step(-1)} aria-label="Previous frame">
   441→            <ChevronLeft className="h-4 w-4" />
   442→          </Button>
   443→        </TooltipTrigger>
   444→        <TooltipContent>Previous frame (,)</TooltipContent>
   445→      </Tooltip>
   446→      <Tooltip>
   447→        <TooltipTrigger asChild>
   448→          <Button
   449→            variant={playing ? 'secondary' : 'ghost'}
   450→            size="icon"
   451→            className={`h-7 w-7 ${playing ? 'bg-teal-500/15 text-teal-300 ring-1 ring-teal-500/40' : 'text-zinc-300'}`}
   452→            onClick={() => editorPlayback.toggle()}
   453→            aria-label={playing ? 'Pause animation' : 'Play animation'}
   454→          >
   455→            {playing ? <Pause className="h-4 w-4" /> : <Play className="h-4 w-4" />}
   456→          </Button>
   457→        </TooltipTrigger>
   458→        <TooltipContent>Play / pause (Enter)</TooltipContent>
   459→      </Tooltip>
   460→      <Tooltip>
   461→        <TooltipTrigger asChild>
   462→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400" onClick={() => editorPlayback.step(1)} aria-label="Next frame">
   463→            <ArrowRight className="h-4 w-4" />
   464→          </Button>
   465→        </TooltipTrigger>
   466→        <TooltipContent>Next frame (.)</TooltipContent>
   467→      </Tooltip>
   468→      <Tooltip>
   469→        <TooltipTrigger asChild>
   470→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400" onClick={() => editorPlayback.stop()} aria-label="Stop and rewind">
   471→            <Square className="h-3.5 w-3.5" />
   472→          </Button>
   473→        </TooltipTrigger>
   474→        <TooltipContent>Stop &amp; rewind</TooltipContent>
   475→      </Tooltip>
   476→    </>
   477→  )
   478→
   479→  return (
   480→    <TooltipProvider delayDuration={200}>
   481→      <section aria-label="Animation timeline" className="shrink-0 border-t border-zinc-800 bg-zinc-950/80">
   482→        {/* clips + transport row */}
   483→        <div className="flex h-9 items-center gap-1 overflow-x-auto border-b border-zinc-800 px-2 pixelforge-scroll">
   484→          {project.clips.map((c) => (
   485→            <div key={c.id} className="flex shrink-0 items-center">
   486→              {renamingClip === c.id ? (
   487→                <Input
   488→                  autoFocus
   489→                  defaultValue={c.name}
   490→                  className="h-6 w-24 border-zinc-700 bg-zinc-900 font-mono text-xs"
   491→                  onBlur={(e) => {
   492→                    setClipProp(c.id, { name: e.target.value || c.name })
   493→                    setRenamingClip(null)
   494→                  }}
   495→                  onKeyDown={(e) => {
   496→                    if (e.key === 'Enter') (e.target as HTMLInputElement).blur()
   497→                    if (e.key === 'Escape') setRenamingClip(null)
   498→                  }}
   499→                />
   500→              ) : (
   501→                <button
   502→                  type="button"
   503→                  onDoubleClick={() => setRenamingClip(c.id)}
   504→                  onClick={() => selectClip(c.id)}
   505→                  className={`rounded-t-md border-b-2 px-3 py-1 font-mono text-xs transition-colors ${
   506→                    c.id === clip.id
   507→                      ? 'border-teal-400 bg-zinc-900 text-teal-300'
   508→                      : 'border-transparent text-zinc-400 hover:text-zinc-300'
   509→                  }`}
   510→                  title={`${c.name} — ${c.frames.length} frames · double-click to rename`}
