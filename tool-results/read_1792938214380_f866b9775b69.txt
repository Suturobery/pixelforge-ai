    34→// --------------------------------------------------------- free-scale handles
    35→// The marquee selection gets 8 handles (corners + edge midpoints). The dots
    36→// are a visual overlay; the CANVAS hit-tests them in onPointerDown so they
    37→// never block painting, and dragging one resizes the selection square freely
    38→// — on release the content is NN-resampled into the new footprint (scaleTo).
    39→
    40→type ScaleHandleId = 'nw' | 'n' | 'ne' | 'e' | 'se' | 's' | 'sw' | 'w'
    41→
    42→const HANDLE_ANCHORS: Array<{ id: ScaleHandleId; fx: 0 | 0.5 | 1; fy: 0 | 0.5 | 1 }> = [
    43→  { id: 'nw', fx: 0, fy: 0 },
    44→  { id: 'n', fx: 0.5, fy: 0 },
    45→  { id: 'ne', fx: 1, fy: 0 },
    46→  { id: 'e', fx: 1, fy: 0.5 },
    47→  { id: 'se', fx: 1, fy: 1 },
    48→  { id: 's', fx: 0.5, fy: 1 },
    49→  { id: 'sw', fx: 0, fy: 1 },
    50→  { id: 'w', fx: 0, fy: 0.5 },
    51→]
    52→
    53→const HANDLE_CURSOR: Record<ScaleHandleId, string> = {
    54→  nw: 'nwse-resize',
    55→  n: 'ns-resize',
    56→  ne: 'nesw-resize',
    57→  e: 'ew-resize',
    58→  se: 'nwse-resize',
    59→  s: 'ns-resize',
    60→  sw: 'nesw-resize',
    61→  w: 'ew-resize',
    62→}
    63→
    64→/** The 8 scale-handle points of a rect, in canvas coordinates. */
    65→function handlePoints(r: SelectionRect): Array<{ id: ScaleHandleId; x: number; y: number }> {
    66→  return HANDLE_ANCHORS.map((h) => ({ id: h.id, x: r.x + r.w * h.fx, y: r.y + r.h * h.fy }))
    67→}
    68→
    69→/** Shrink a rect into the canvas, never smaller than 1×1. */
    70→function clampSelRect(r: SelectionRect, W: number, H: number): SelectionRect {
    71→  const x = Math.max(0, Math.min(r.x, W - 1))
    72→  const y = Math.max(0, Math.min(r.y, H - 1))
    73→  return {
    74→    x,
    75→    y,
    76→    w: Math.max(1, Math.min(r.x + r.w, W) - x),
    77→    h: Math.max(1, Math.min(r.y + r.h, H) - y),
    78→  }
    79→}
    80→
    81→interface Interaction {
    82→  mode: 'paint' | 'shape' | 'move' | 'pan' | 'marquee-create' | 'sel-move' | 'sel-scale'
    83→  tool?: ToolId
    84→  start: { x: number; y: number }
    85→  last: { x: number; y: number }
    86→  handle?: StrokeHandle
    87→  snapshot?: Uint8ClampedArray
    88→  changed: boolean
    89→  panStart?: { cx: number; cy: number; panX: number; panY: number }
    90→  /** marquee-create / sel-move / sel-scale */
    91→  selectionStart?: SelectionRect
    92→  /** sel-scale: which handle is being dragged */
    93→  scaleHandle?: ScaleHandleId
    94→  /** sel-move: currently applied pixel offset (committed on pointer-up). */
    95→  moveDx?: number
    96→  moveDy?: number
    97→}

    99→export function CanvasStage() {
   100→  const containerRef = useRef<HTMLDivElement>(null)
   101→  const canvasRef = useRef<HTMLCanvasElement>(null)
   102→  const boardRef = useRef<HTMLDivElement>(null)
   103→  const offRef = useRef<HTMLCanvasElement | null>(null)
   104→  const interactionRef = useRef<Interaction | null>(null)
   105→  const spaceRef = useRef(false)
   106→  const [spaceHeld, setSpaceHeld] = useState(false)
   107→
   108→  const project = useEditor((s) => s.project)
   109→  const activeClipId = useEditor((s) => s.activeClipId)
   110→  const activeFrameIndex = useEditor((s) => s.activeFrameIndex)
   111→  const zoom = useEditor((s) => s.zoom)
   112→  const panX = useEditor((s) => s.panX)
   113→  const panY = useEditor((s) => s.panY)
   114→  const showGrid = useEditor((s) => s.showGrid)
   115→  const onionSkin = useEditor((s) => s.onionSkin)
   116→  const onionOpacity = useAppSettings((s) => s.onionOpacity)
   117→  const onionPrevTint = useAppSettings((s) => s.onionPrevTint)
   118→  const onionNextTint = useAppSettings((s) => s.onionNextTint)
   119→  const tool = useEditor((s) => s.tool)
   120→  const primary = useEditor((s) => s.primary)
   121→  const brushSize = useEditor((s) => s.brushSize)
   122→  const mirrorX = useEditor((s) => s.mirrorX)
   123→  const ellipseFilled = useEditor((s) => s.ellipseFilled)
   124→  const selection = useEditor((s) => s.selection)
   125→  const playing = useEditor((s) => s.playing)
   126→  const renderTick = useEditor((s) => s.renderTick)
   127→  const hover = useEditor((s) => s.hover)
   128→  const projectId = useEditor((s) => s.projectId)
   129→  const filePath = useEditor((s) => s.filePath)
   130→  const loadEpoch = useEditor((s) => s.loadEpoch)
   131→  const [, forceRedraw] = useState(0)
   132→  // marching-ants animation tick (only runs while a selection exists)
   133→  const [antsTick, setAntsTick] = useState(0)
   134→  // free-scale draft: the in-progress selection rect while a scale handle is
   135→  // being dragged (drives the ants overlay; commits as ONE scaleTo op on
   136→  // release). The ref mirrors the state so pointerup reads the latest value.
   137→  const [scaleDraft, setScaleDraft] = useState<SelectionRect | null>(null)
   138→  const scaleDraftRef = useRef<SelectionRect | null>(null)
   139→  // true when the composited active frame is fully transparent — we show a
   140→  // "this frame is empty" hint so a blank canvas is never mistaken for lost
   141→  // pixel data (top user confusion after adding an empty frame)
   142→  const [frameEmpty, setFrameEmpty] = useState(false)
   143→
   144→  // zoom anchored at the viewport center (buttons used to expand the canvas
   145→  // from its top-left corner, drifting it off-screen)
   146→  const zoomBy = useCallback((factor: number) => {
   147→    const st = useEditor.getState()
   148→    const container = containerRef.current
   149→    const nz = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, st.zoom * factor))
   150→    if (!container) {
   151→      st.setZoom(nz)
   152→      return
   153→    }
   154→    const rect = container.getBoundingClientRect()
   155→    const cx = rect.width / 2
   156→    const cy = rect.height / 2
   157→    const wx = (cx - st.panX) / st.zoom
   158→    const wy = (cy - st.panY) / st.zoom
   159→    st.setZoom(nz)
   160→    st.setPan(cx - wx * nz, cy - wy * nz)
   161→  }, [])
   162→
   163→  const { width: W, height: H } = project
   164→
   165→  // ------------------------------------------------------------- composite
   166→  useEffect(() => {
   167→    if (!offRef.current) offRef.current = document.createElement('canvas')
   168→    const off = offRef.current
   169→    if (off.width !== W || off.height !== H) {
   170→      off.width = W
   171→      off.height = H
   172→    }
   173→    const ctx = off.getContext('2d')
   174→    if (!ctx) return
   175→    const st = useEditor.getState()
   176→    const clip = getActiveClip(st)
   177→    const len = clip.frames.length
   178→    const idx = Math.min(activeFrameIndex, len - 1)
   179→    const frame = clip.frames[idx]
   180→    if (!frame) {
   181→      ctx.clearRect(0, 0, W, H)
   182→      return
   183→    }
   184→    // Onion skin (Task 6 §4): prev/next counts are DOCUMENT settings
   185→    // (undoable via settings.setOnion); opacity + tints are view prefs.
   186→    // Ghosts wrap the loop seam only when the clip actually loops.
   187→    const ghosts: OnionGhost[] = []
   188→    if (onionSkin && len > 1) {
   189→      const { prev: prevCount, next: nextCount } = st.project.settings.onion
   190→      for (let d = 1; d <= Math.min(prevCount, ONION_MAX); d++) {
   191→        const i = idx - d
   192→        if (i < 0 && !clip.loop) break
   193→        const f = clip.frames[((i % len) + len) % len]
   194→        if (f && f !== frame) {
   195→          ghosts.push({ frame: f, depth: d - 1, dir: 'prev', tint: onionPrevTint, opacity: onionOpacity })
   196→        }
   197→      }
   198→      for (let d = 1; d <= Math.min(nextCount, ONION_MAX); d++) {
   199→        const i = idx + d
   200→        if (i > len - 1 && !clip.loop) break
   201→        const f = clip.frames[i % len]
   202→        if (f && f !== frame) {
   203→          ghosts.push({ frame: f, depth: d - 1, dir: 'next', tint: onionNextTint, opacity: onionOpacity })
   204→        }
   205→      }
   206→    }
   207→    drawFrame(ctx, clip, frame, W, H, { ghosts })
   208→    // detect a fully-empty ACTIVE frame — scan the cel buffers directly so
   209→    // onion-skin ghosts don't mask the "empty frame" hint
   210→    let hasPixel = false
   211→    for (const layer of clip.layers) {
   212→      if (!layer.visible) continue
   213→      const cel = frame.cels[layer.id]
   214→      if (!cel) continue // absent cel = transparent (C2)
   215→      const d = cel.buffer
   216→      for (let i = 3; i < d.length; i += 4) {
   217→        if (d[i] > 0) {
   218→          hasPixel = true
   219→          break
   220→        }
   221→      }
   222→      if (hasPixel) break
   223→    }
   224→    setFrameEmpty((cur) => (cur === hasPixel ? cur : !hasPixel))
   225→  }, [project, activeClipId, activeFrameIndex, renderTick, onionSkin, onionOpacity, onionPrevTint, onionNextTint, W, H])
   226→
   227→  // marching ants: step the dash offset while a selection is visible
   228→  useEffect(() => {
   229→    if (!selection || playing) return
   230→    const t = setInterval(() => setAntsTick((v) => v + 1), 120)
   231→    return () => clearInterval(t)
   232→  }, [selection, playing])
   233→
   234→  // ----------------------------------------------------------------- paint
   235→  useEffect(() => {
   236→    const canvas = canvasRef.current
   237→    const board = boardRef.current
   238→    if (!canvas || !board || !offRef.current) return
   239→    const dpr = window.devicePixelRatio || 1
   240→    const cw = W * zoom
   241→    const ch = H * zoom
   242→    const bw = Math.max(1, Math.round(cw * dpr))
   243→    const bh = Math.max(1, Math.round(ch * dpr))
   244→    if (canvas.width !== bw || canvas.height !== bh) {
   245→      canvas.width = bw
   246→      canvas.height = bh
   247→    }
   248→    canvas.style.width = `${cw}px`
   249→    canvas.style.height = `${ch}px`
   250→    board.style.width = `${cw}px`
   251→    board.style.height = `${ch}px`
   252→    board.style.transform = `translate(${panX}px, ${panY}px)`
   253→
   254→    const ctx = canvas.getContext('2d')
   255→    if (!ctx) return
   256→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   257→    ctx.imageSmoothingEnabled = false
   258→    ctx.clearRect(0, 0, cw, ch)
   259→    ctx.drawImage(offRef.current, 0, 0, cw, ch)
   260→
   261→    // pixel grid
   262→    if (showGrid && zoom >= 5) {
   263→      ctx.lineWidth = 1
   264→      for (let x = 0; x <= W; x++) {
   265→        const strong = x % 8 === 0
   266→        ctx.strokeStyle = strong ? 'rgba(255,255,255,0.16)' : 'rgba(255,255,255,0.07)'
   267→        const px = Math.round(x * zoom * dpr) / dpr + 0.5 / dpr
   268→        ctx.beginPath()
   269→        ctx.moveTo(px, 0)
   270→        ctx.lineTo(px, ch)
   271→        ctx.stroke()
   272→      }
   273→      for (let y = 0; y <= H; y++) {
   274→        const strong = y % 8 === 0
   275→        ctx.strokeStyle = strong ? 'rgba(255,255,255,0.16)' : 'rgba(255,255,255,0.07)'
   276→        const py = Math.round(y * zoom * dpr) / dpr + 0.5 / dpr
   277→        ctx.beginPath()
   278→        ctx.moveTo(0, py)
   279→        ctx.lineTo(cw, py)
   280→        ctx.stroke()
   281→      }
   282→    }
   283→
   284→    // hover cursor (paint tools)
   285→    if (hover && !playing && tool !== 'eyedropper' && tool !== 'move' && tool !== 'marquee') {
   286→      const size = tool === 'fill' ? 1 : brushSize
   287→      const off2 = Math.floor((Math.max(1, size) - 1) / 2)
   288→      const hx = (hover.x - off2) * zoom
   289→      const hy = (hover.y - off2) * zoom
   290→      const hs = Math.max(1, size) * zoom
   291→      ctx.strokeStyle = 'rgba(45, 212, 191, 0.9)'
   292→      ctx.lineWidth = 1
   293→      ctx.strokeRect(hx + 0.5, hy + 0.5, hs - 1, hs - 1)
   294→    }
   295→
   296→    // marching-ants selection outline (two alternating dashed passes);
   297→    // while a scale handle is dragged the DRAFT rect is outlined instead
   298→    if (selection && !playing) {
   299→      const ants = scaleDraft ?? selection
   300→      const sx = ants.x * zoom
   301→      const sy = ants.y * zoom
   302→      const sw = ants.w * zoom
   303→      const sh = ants.h * zoom
   304→      const dash = Math.max(3, Math.round(zoom / 2))
   305→      ctx.save()
   306→      ctx.lineWidth = 1
   307→      ctx.setLineDash([dash, dash])
   308→      ctx.lineDashOffset = -(antsTick % 2) * dash // crawl
   309→      ctx.strokeStyle = 'rgba(0, 0, 0, 0.95)'
   310→      ctx.strokeRect(sx + 0.5, sy + 0.5, sw - 1, sh - 1)
   311→      ctx.lineDashOffset = -(antsTick % 2) * dash + dash / 2
   312→      ctx.strokeStyle = 'rgba(255, 255, 255, 0.95)'
   313→      ctx.strokeRect(sx + 0.5, sy + 0.5, sw - 1, sh - 1)
   314→      ctx.restore()
   315→    }
   316→    // NOTE: activeFrameIndex must be a dep — selectFrame() only composites the
   317→    // offscreen buffer; without this dep the visible canvas keeps showing the
   318→    // previous frame until the next hover/tick change ("frame click delay").
   319→    // NOTE: the onion values must be deps too — toggling the switch or changing
   320→    // opacity/tints re-composites the offscreen buffer only; without these deps
   321→    // the visible canvas keeps showing the pre-onion blit (stale ghosts).
   322→  }, [project, W, H, zoom, panX, panY, showGrid, renderTick, hover, tool, brushSize, playing, activeFrameIndex, selection, scaleDraft, ellipseFilled, antsTick, onionSkin, onionOpacity, onionPrevTint, onionNextTint])
   323→
   324→  // ------------------------------------------------------------ fit / zoom
   325→  const fitView = useCallback(() => {
   326→    const container = containerRef.current
   327→    if (!container) return
   328→    const rect = container.getBoundingClientRect()
   329→    const z = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, Math.floor(Math.min((rect.width - 48) / W, (rect.height - 48) / H))))
   330→    const st = useEditor.getState()
   331→    st.setZoom(z)
   332→    st.setPan((rect.width - W * z) / 2, (rect.height - H * z) / 2)
   333→  }, [W, H])
   334→
   335→  useEffect(() => {
   336→    fitView()
   337→    // re-fit when a different project is loaded/newed (loadProject resets view state)
   338→  }, [W, H, projectId, filePath, loadEpoch])
   339→
   340→  // refit when the stage container is resized (device rotation, breakpoint
   341→  // flip desktop ↔ mobile, panel toggles) so the canvas stays centered
   342→  useEffect(() => {
   343→    const container = containerRef.current
   344→    if (!container) return
   345→    let lastW = container.clientWidth
   346→    let lastH = container.clientHeight
   347→    let raf = 0
   348→    const check = () => {
   349→      cancelAnimationFrame(raf)
   350→      raf = requestAnimationFrame(() => {
   351→        const w = container.clientWidth
   352→        const h = container.clientHeight
   353→        if (Math.abs(w - lastW) > 2 || Math.abs(h - lastH) > 2) {
   354→          lastW = w
   355→          lastH = h
   356→          fitView()
   357→        }
   358→      })
   359→    }
   360→    const ro = new ResizeObserver(check)
   361→    ro.observe(container)
   362→    return () => {
   363→      ro.disconnect()
   364→      cancelAnimationFrame(raf)
   365→    }
   366→  }, [fitView])
   367→
   368→  // wheel zoom (anchored at cursor)
   369→  useEffect(() => {
   370→    const container = containerRef.current
   371→    if (!container) return
   372→    const onWheel = (e: WheelEvent) => {
   373→      e.preventDefault()
   374→      const st = useEditor.getState()
   375→      const rect = container.getBoundingClientRect()
   376→      const cx = e.clientX - rect.left
   377→      const cy = e.clientY - rect.top
   378→      const factor = Math.exp(-e.deltaY * 0.0015)
   379→      const nz = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, st.zoom * factor))
   380→      const wx = (cx - st.panX) / st.zoom
   381→      const wy = (cy - st.panY) / st.zoom
   382→      st.setZoom(nz)
   383→      st.setPan(cx - wx * nz, cy - wy * nz)
   384→    }
   385→    container.addEventListener('wheel', onWheel, { passive: false })
   386→    return () => container.removeEventListener('wheel', onWheel)
   387→  }, [])
   388→
   389→  // space-to-pan tracking
   390→  useEffect(() => {
   391→    const down = (e: KeyboardEvent) => {
   392→      if (e.code === 'Space' && !e.repeat) {
   393→        const t = e.target as HTMLElement
   394→        if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable)) return
   395→        spaceRef.current = true
   396→        setSpaceHeld(true)
   397→      }
   398→    }
   399→    const up = (e: KeyboardEvent) => {
   400→      if (e.code === 'Space') {
   401→        spaceRef.current = false
   402→        setSpaceHeld(false)
   403→      }
   404→    }
   405→    window.addEventListener('keydown', down)
   406→    window.addEventListener('keyup', up)
   407→    return () => {
   408→      window.removeEventListener('keydown', down)
   409→      window.removeEventListener('keyup', up)
   410→    }
   411→  }, [])
   412→
   413→  // ------------------------------------------------------------- pointers
   414→  const toPixel = useCallback(
   415→    (clientX: number, clientY: number) => {
   416→      const container = containerRef.current
   417→      if (!container) return { x: 0, y: 0 }
   418→      const rect = container.getBoundingClientRect()
   419→      const x = Math.floor((clientX - rect.left - panX) / zoom)
   420→      const y = Math.floor((clientY - rect.top - panY) / zoom)
   421→      return { x, y }
   422→    },
   423→    [panX, panY, zoom]
   424→  )
   425→
   426→  const onPointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
   427→    if (playing) return
   428→    const canvas = canvasRef.current
   429→    if (!canvas) return
   430→    const st = useEditor.getState()
   431→    const px = toPixel(e.clientX, e.clientY)
   432→    canvas.setPointerCapture(e.pointerId)
   433→
   434→    // pan: middle button, right button, or space held
   435→    if (e.button === 1 || e.button === 2 || spaceRef.current) {
   436→      interactionRef.current = {
   437→        mode: 'pan',
   438→        start: px,
   439→        last: px,
   440→        changed: false,
   441→        panStart: { cx: e.clientX, cy: e.clientY, panX: st.panX, panY: st.panY },
   442→      }
   443→      return
   444→    }
   445→    if (e.button !== 0) return
   446→
   447→    const { width: w, height: h } = st.project
   448→    const color = hexToRgba(st.primary)
   449→
   450→    switch (st.tool) {
   451→      case 'pencil':
   452→      case 'eraser': {
   453→        const handle = st.beginStroke()
   454→        if (!handle) return
   455→        const c: [number, number, number, number] = st.tool === 'eraser' ? [0, 0, 0, 0] : color
   456→        stampBrush(handle.buffer, w, h, px.x, px.y, st.brushSize, c, st.mirrorX)
   457→        interactionRef.current = {
   458→          mode: 'paint',
   459→          tool: st.tool,
   460→          start: px,
   461→          last: px,
   462→          handle,
   463→          changed: true,
   464→        }
   465→        st.bump()
   466→        break
   467→      }
   468→      case 'fill': {
   469→        const handle = st.beginStroke()
   470→        if (!handle) return
   471→        const changed = floodFill(handle.buffer, w, h, px.x, px.y, color, st.mirrorX)
   472→        st.endStroke('Fill', changed)
   473→        break
   474→      }
   475→      case 'eyedropper': {
   476→        const off = offRef.current
   477→        if (off && px.x >= 0 && px.y >= 0 && px.x < w && px.y < h) {
   478→          const octx = off.getContext('2d', { willReadFrequently: true })
   479→          if (octx) {
   480→            const d = octx.getImageData(px.x, px.y, 1, 1).data
   481→            if (d[3] > 0) {
   482→              const hex = `#${[d[0], d[1], d[2]].map((v) => v.toString(16).padStart(2, '0')).join('')}`
   483→              st.setPrimary(hex)
   484→            }
   485→          }
   486→        }
   487→        break
   488→      }
   489→      case 'line':
   490→      case 'rect':
   491→      case 'ellipse': {
   492→        const handle = st.beginStroke()
   493→        if (!handle) return
   494→        interactionRef.current = {
   495→          mode: 'shape',
   496→          tool: st.tool,
   497→          start: px,
   498→          last: px,
   499→          handle,
   500→          snapshot: new Uint8ClampedArray(handle.buffer),
   501→          changed: false,
   502→        }
   503→        break
   504→      }
   505→      case 'marquee': {
   506→        const sel = st.selection
   507→        // free-scale handles first — screen-space proximity around the 8
   508→        // handle points wins over move/create so the dots feel sticky
   509→        if (sel) {
   510→          const container = containerRef.current
   511→          if (container) {
   512→            const rect = container.getBoundingClientRect()
   513→            const fx = (e.clientX - rect.left - panX) / zoom
   514→            const fy = (e.clientY - rect.top - panY) / zoom
   515→            const radius = (e.pointerType === 'touch' ? 14 : 6) / zoom
   516→            const hit = handlePoints(sel).find(
   517→              (h) => Math.abs(h.x - fx) <= radius && Math.abs(h.y - fy) <= radius
   518→            )
   519→            if (hit) {
   520→              interactionRef.current = {
   521→                mode: 'sel-scale',
   522→                tool: 'marquee',
   523→                start: px,
   524→                last: px,
   525→                changed: false,
   526→                selectionStart: sel,
   527→                scaleHandle: hit.id,
   528→              }
   529→              scaleDraftRef.current = { ...sel }
   530→              setScaleDraft({ ...sel })
   531→              return
   532→            }
   533→          }
   534→        }
   535→        if (sel && pointInSelection(sel, px.x, px.y)) {
   536→          // drag INSIDE an existing selection → move those pixels (stroke
   537→          // flow: preview mutates only the handed-out working buffer)
   538→          const handle = st.beginStroke()
   539→          if (!handle) return
   540→          interactionRef.current = {
   541→            mode: 'sel-move',
   542→            tool: 'marquee',
   543→            start: px,
   544→            last: px,
   545→            handle,
   546→            snapshot: new Uint8ClampedArray(handle.buffer),
   547→            changed: false,
   548→            selectionStart: sel,
   549→          }
   550→        } else {
   551→          // drag OUTSIDE → draw a new selection rect (view state only)
   552→          interactionRef.current = {
   553→            mode: 'marquee-create',
   554→            tool: 'marquee',
   555→            start: px,
   556→            last: px,
   557→            changed: false,
   558→          }
   559→          st.setSelection(rectFromPoints(px.x, px.y, px.x, px.y))
   560→        }
   561→        break
   562→      }
   563→      case 'move': {
   564→        const handle = st.beginStroke()
   565→        if (!handle) return
   566→        interactionRef.current = {
   567→          mode: 'move',
   568→          tool: 'move',
   569→          start: px,
   570→          last: px,
   571→          handle,
   572→          snapshot: new Uint8ClampedArray(handle.buffer),
   573→          changed: false,
   574→        }
   575→        break
   576→      }
   577→    }
   578→  }
   579→
   580→  const onPointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
   581→    const st = useEditor.getState()
   582→    const px = toPixel(e.clientX, e.clientY)
   583→    st.setHover({ x: px.x, y: px.y })
   584→    const it = interactionRef.current
   585→    if (!it || playing) return
   586→
   587→    if (it.mode === 'pan' && it.panStart) {
   588→      st.setPan(it.panStart.panX + (e.clientX - it.panStart.cx), it.panStart.panY + (e.clientY - it.panStart.cy))
   589→      return
   590→    }
   591→
   592→    // marquee-create runs without a stroke handle (view state only)
   593→    if (it.mode === 'marquee-create') {
   594→      const rect = rectFromPoints(it.start.x, it.start.y, px.x, px.y)
   595→      st.setSelection(rect)
   596→      it.changed = true
   597→      return
   598→    }
   599→
   600→    // free-scale: resize the draft square (content commits once on release)
   601→    if (it.mode === 'sel-scale' && it.selectionStart && it.scaleHandle) {
   602→      const s = it.selectionStart
   603→      let x0 = s.x
   604→      let y0 = s.y
   605→      let x1 = s.x + s.w
   606→      let y1 = s.y + s.h
   607→      if (it.scaleHandle.includes('w')) x0 = px.x
   608→      if (it.scaleHandle.includes('e')) x1 = px.x
   609→      if (it.scaleHandle.startsWith('n')) y0 = px.y
   610→      if (it.scaleHandle.startsWith('s')) y1 = px.y
   611→      const draft = clampSelRect(
   612→        {
   613→          x: Math.min(x0, x1),
   614→          y: Math.min(y0, y1),
   615→          w: Math.max(1, Math.abs(x1 - x0)),
   616→          h: Math.max(1, Math.abs(y1 - y0)),
   617→        },
   618→        st.project.width,
   619→        st.project.height
   620→      )
   621→      const cur = scaleDraftRef.current
   622→      if (
   623→        !cur ||
   624→        draft.x !== cur.x ||
   625→        draft.y !== cur.y ||
   626→        draft.w !== cur.w ||
   627→        draft.h !== cur.h
   628→      ) {
   629→        scaleDraftRef.current = draft
   630→        setScaleDraft(draft)
   631→      }
   632→      it.changed =
   633→        draft.x !== s.x || draft.y !== s.y || draft.w !== s.w || draft.h !== s.h
   634→      return
   635→    }
   636→
   637→    const handle = it.handle
   638→    if (!handle) return
   639→    const { width: w, height: h } = st.project
   640→
   641→    if (it.mode === 'paint' && it.tool) {
   642→      const c: [number, number, number, number] = it.tool === 'eraser' ? [0, 0, 0, 0] : hexToRgba(st.primary)
   643→      drawLinePixels(handle.buffer, w, h, it.last.x, it.last.y, px.x, px.y, st.brushSize, c, st.mirrorX)
   644→      it.last = px
   645→      it.changed = true
   646→      st.bump()
   647→      return
   648→    }
   649→
   650→    if (it.mode === 'shape' && it.snapshot) {
   651→      handle.buffer.set(it.snapshot)
   652→      if (it.tool === 'line') {
   653→        const c = hexToRgba(st.primary)
   654→        drawLinePixels(handle.buffer, w, h, it.start.x, it.start.y, px.x, px.y, st.brushSize, c, st.mirrorX)
   655→      } else if (it.tool === 'rect') {
   656→        const c = hexToRgba(st.primary)
   657→        drawRectPixels(handle.buffer, w, h, it.start.x, it.start.y, px.x, px.y, st.brushSize, c, e.shiftKey, st.mirrorX)
   658→      } else {
   659→        // ellipse: outline uses the primary color; filled uses the secondary
   660→        // (v3.1 §2.1 shape convention). Shift inverts the fill toggle.
   661→        const filled = e.shiftKey ? !st.ellipseFilled : st.ellipseFilled
   662→        const c = hexToRgba(filled ? st.secondary : st.primary)
   663→        drawEllipsePixels(handle.buffer, w, h, it.start.x, it.start.y, px.x, px.y, st.brushSize, c, filled, st.mirrorX)
   664→      }
   665→      it.last = px
   666→      it.changed = true
   667→      st.bump()
   668→      return
   669→    }
   670→
   671→    if (it.mode === 'sel-move' && it.snapshot && it.selectionStart) {
   672→      const dx = px.x - it.start.x
   673→      const dy = px.y - it.start.y
   674→      if (dx === (it.moveDx ?? 0) && dy === (it.moveDy ?? 0)) {
   675→        return
   676→      }
   677→      composeSelectionMove(
   678→        handle.buffer,
   679→        it.snapshot,
   680→        w,
   681→        h,
   682→        it.selectionStart,
   683→        dx,
   684→        dy
   685→      )
   686→      it.moveDx = dx
   687→      it.moveDy = dy
   688→      it.changed = dx !== 0 || dy !== 0
   689→      it.last = px
   690→      st.bump()
   691→      return
   692→    }
   693→
   694→    if (it.mode === 'move' && it.snapshot) {
   695→      const dx = px.x - it.start.x
   696→      const dy = px.y - it.start.y
   697→      const moved = moveBuffer(it.snapshot, w, h, dx, dy)
   698→      handle.buffer.set(moved)
   699→      it.changed = dx !== 0 || dy !== 0
   700→      st.bump()
   701→    }
   702→  }
   703→
   704→  const finishInteraction = () => {
   705→    const st = useEditor.getState()
   706→    const it = interactionRef.current
   707→    interactionRef.current = null
   708→    if (!it || it.mode === 'pan') return
   709→    if (it.mode === 'paint' && it.tool) {
   710→      st.endStroke(it.tool === 'eraser' ? 'Eraser' : 'Pencil', it.changed)
   711→    } else if (it.mode === 'shape' && it.tool) {
   712→      st.endStroke(it.tool === 'line' ? 'Line' : it.tool === 'ellipse' ? 'Ellipse' : 'Rect', it.changed)
   713→    } else if (it.mode === 'move') {
   714→      st.endStroke('Move layer', it.changed)
   715→    } else if (it.mode === 'marquee-create') {
   716→      // a click without a drag collapses to a deselect (predictable UX)
   717→      const st2 = useEditor.getState()
   718→      if (st2.selection && st2.selection.w <= 1 && st2.selection.h <= 1 && !it.changed) {
   719→        st2.setSelection(null)
   720→      }
   721→    } else if (it.mode === 'sel-scale' && it.selectionStart) {
   722→      // commit: NN-resample the content into the dragged footprint (one
   723→      // undoable scaleTo op; the store clamps + no-ops identical footprints)
   724→      const draft = scaleDraftRef.current
   725→      scaleDraftRef.current = null
   726→      setScaleDraft(null)
   727→      if (draft) {
   728→        const changed =
   729→          draft.x !== it.selectionStart.x ||
   730→          draft.y !== it.selectionStart.y ||
   731→          draft.w !== it.selectionStart.w ||
   732→          draft.h !== it.selectionStart.h
   733→        if (changed) useEditor.getState().selectionScaleTo(draft)
   734→      }
   735→    } else if (it.mode === 'sel-move' && it.selectionStart) {
   736→      // commit the final offset into history (the working buffer already
   737→      // holds the preview; endStroke diffs it against the stroke start)
   738→      useEditor
   739→        .getState()
   740→        .endStroke('Move selection', it.changed)
   741→      if (it.changed) {
   742→        const project = useEditor.getState().project
   743→        useEditor.setState({
   744→          selection: shiftRectClamped(
   745→            it.selectionStart,
   746→            it.moveDx ?? 0,
   747→            it.moveDy ?? 0,
   748→            project.width,
   749→            project.height
   750→          ),
   751→        })
   752→      }
   753→    }
   754→  }
   755→
   756→  const onPointerUp = (e: React.PointerEvent<HTMLCanvasElement>) => {
   757→    const canvas = canvasRef.current
   758→    if (canvas?.hasPointerCapture(e.pointerId)) canvas.releasePointerCapture(e.pointerId)
   759→    finishInteraction()
   760→  }
   761→
   762→  const scaleRect = tool === 'marquee' && !playing ? scaleDraft ?? selection : null
   763→  const handleHover =
   764→    scaleRect && hover && !spaceHeld
   765→      ? handlePoints(scaleRect).find(
   766→          (h) =>
   767→            Math.abs(h.x - (hover.x + 0.5)) <= 6 / zoom &&
   768→            Math.abs(h.y - (hover.y + 0.5)) <= 6 / zoom
   769→        )
   770→      : undefined
   771→
   772→  const cursor = playing
   773→    ? 'default'
   774→    : spaceHeld
   775→      ? 'grab'
   776→      : handleHover
   777→        ? HANDLE_CURSOR[handleHover.id]
   778→        : tool === 'move'
   779→          ? 'move'
   780→          : tool === 'marquee' && selection && hover && pointInSelection(selection, hover.x, hover.y)
   781→            ? 'move'
   782→            : 'crosshair'
   783→
   784→  return (
   785→    <div className="@container relative min-w-0 flex-1 overflow-hidden bg-zinc-900/60">
   786→      <div
   787→        ref={containerRef}
   788→        className="absolute inset-0 overflow-hidden"
   789→        onContextMenu={(e) => e.preventDefault()}
   790→      >
   791→        <div
   792→          ref={boardRef}
   793→          className="pixelforge-checker absolute left-0 top-0 shadow-[0_0_0_1px_rgba(255,255,255,0.18),0_8px_32px_rgba(0,0,0,0.5)]"
   794→        >
   795→          <canvas
   796→            ref={canvasRef}
   797→            className="absolute left-0 top-0 touch-none select-none"
   798→            style={{ cursor, imageRendering: 'pixelated' }}
   799→            onPointerDown={onPointerDown}
   800→            onPointerMove={onPointerMove}
   801→            onPointerUp={onPointerUp}
   802→            onPointerLeave={(e) => {
   803→              useEditor.getState().setHover(null)
   804→              onPointerUp(e)
   805→            }}
   806→          />
   807→
   808→          {/* free-scale handles — visual only (pointer-events-none; the canvas
   809→              hit-tests them) — they track the live draft while dragging */}
   810→          {scaleRect && (
   811→            <div className="pointer-events-none absolute inset-0">
   812→              {handlePoints(scaleRect).map((h) => (
   813→                <div
   814→                  key={h.id}
   815→                  className="absolute h-2 w-2 -translate-x-1/2 -translate-y-1/2 rounded-[2px] border border-zinc-900 bg-teal-300 shadow"
   816→                  style={{ left: h.x * zoom, top: h.y * zoom }}
   817→                />
   818→              ))}
   819→            </div>
   820→          )}
   821→        </div>
   822→      </div>
   823→
   824→      {/* floating animation preview — lives on the edit area itself */}
   825→      <PreviewOverlay />
   826→
   827→      {/* floating palette — draggable + hideable (same pattern as preview) */}
   828→      <PalettePanel />
   829→
   830→      {/* floating view controls — draggable + hideable (persisted) */}
   831→      <ViewToolbar onZoomBy={zoomBy} onFit={fitView} />
   832→
   833→      {/* contextual tool options (ellipse fill toggle, selection actions) */}
   834→      <ToolOptionsBar />
   835→
   836→      {/* empty-frame hint — makes it explicit that pixels were NOT lost when
   837→          the playhead sits on a blank frame (e.g. right after Add frame) */}
   838→      {frameEmpty && !playing && (
   839→        <div
   840→          role="status"
   841→          className="pointer-events-none absolute left-1/2 top-3 z-10 -translate-x-1/2 rounded-full border border-zinc-700/80 bg-zinc-950/90 px-3 py-1.5 text-center shadow-lg backdrop-blur"
   842→        >
   843→          <p className="font-mono text-[11px] text-zinc-200">
   844→            Frame {Math.min(activeFrameIndex + 1, getActiveClip(useEditor.getState()).frames.length)} is empty
   845→          </p>
   846→          <p className="font-mono text-[10px] text-zinc-400">
   847→            draw here — your other frames are safe on the timeline below
   848→          </p>
   849→        </div>
   850→      )}
   851→    </div>
   852→  )
   853→}
