    55→    55→/** Effective duration of a frame for badges/scrubber (mirrors core math). */
    56→    56→function frameMs(clip: Clip, index: number): number {
    57→    57→  const f = clip.frames[index]
    58→    58→  if (!f) return 0
    59→    59→  if (f.durationMs > 0) return f.durationMs
    60→    60→  const fps = Math.min(60, Math.max(1, Math.round(clip.fps || 8)))
    61→    61→  return Math.max(1, Math.round(1000 / fps))
    62→    62→}
    63→    63→
    64→    64→function formatMs(ms: number): string {
    65→    65→  return ms >= 1000 ? `${(ms / 1000).toFixed(ms % 1000 === 0 ? 0 : 2)}s` : `${ms}ms`
    66→    66→}

   199→   199→function Scrubber({ clip, activeIndex }: { clip: Clip; activeIndex: number }) {
   200→   200→  const trackRef = useRef<HTMLDivElement>(null)
   201→   201→  const markerRef = useRef<HTMLDivElement>(null)
   202→   202→  const [hoverMs, setHoverMs] = useState<number | null>(null)
   203→   203→
   204→   204→  // Playhead marker follows the shared engine via direct DOM writes — no
   205→   205→  // React re-render per animation tick (smooth at any fps).
   206→   206→  useEffect(
   207→   207→    () =>
   208→   208→      editorPlayback.onPosition((pos, total) => {
   209→   209→        const el = markerRef.current
   210→   210→        if (el) el.style.left = `${total > 0 ? Math.min(100, (pos / total) * 100) : 0}%`
   211→   211→      }),
   212→   212→    []
   213→   213→  )
   214→   214→
   215→   215→  const msFromEvent = (clientX: number): number => {
   216→   216→    const track = trackRef.current
   217→   217→    if (!track) return 0
   218→   218→    const rect = track.getBoundingClientRect()
   219→   219→    const total = clip.frames.reduce((s, _, i) => s + frameMs(clip, i), 0)
   220→   220→    const ratio = Math.min(1, Math.max(0, (clientX - rect.left) / rect.width))
   221→   221→    return Math.round(ratio * total)
   222→   222→  }
   223→   223→
   224→   224→  const scrub = (clientX: number) => editorPlayback.scrubToMs(msFromEvent(clientX))
   225→   225→
   226→   226→  return (
   227→   227→    <div className="px-2 pb-1">
   228→   228→      <div
   229→   229→        ref={trackRef}
   230→   230→        role="slider"
   231→   231→        aria-label="Timeline position"
   232→   232→        aria-valuemin={1}
   233→   233→        aria-valuemax={clip.frames.length}
   234→   234→        aria-valuenow={activeIndex + 1}
   235→   235→        aria-valuetext={`Frame ${activeIndex + 1} of ${clip.frames.length}`}
   236→   236→        tabIndex={0}
   237→   237→        className="relative flex h-5 cursor-pointer touch-none items-stretch overflow-hidden rounded-md border border-zinc-800 bg-zinc-900"
   238→   238→        onPointerDown={(e) => {
   239→   239→          e.currentTarget.setPointerCapture(e.pointerId)
   240→   240→          scrub(e.clientX)
   241→   241→        }}
   242→   242→        onPointerMove={(e) => {
   243→   243→          if (e.buttons & 1) scrub(e.clientX)
   244→   244→          setHoverMs(msFromEvent(e.clientX))
   245→   245→        }}
   246→   246→        onPointerLeave={() => setHoverMs(null)}
   247→   247→        onKeyDown={(e) => {
   248→   248→          if (e.key === 'ArrowRight') editorPlayback.step(1)
   249→   249→          else if (e.key === 'ArrowLeft') editorPlayback.step(-1)
   250→   250→        }}
   251→   251→      >
   252→   252→        {/* proportional frame segments (width = effective duration) */}
   253→   253→        {clip.frames.map((f, i) => (
   254→   254→          <div
   255→   255→            key={f.id}
   256→   256→            title={`Frame ${i + 1} · ${frameMs(clip, i)}ms`}
   257→   257→            style={{ flexGrow: frameMs(clip, i), flexBasis: 0 }}
   258→   258→            className={`min-w-[3px] border-r border-zinc-950/80 last:border-r-0 ${
   259→   259→              i === activeIndex ? 'bg-teal-700/50' : i % 2 === 0 ? 'bg-zinc-800/70' : 'bg-zinc-800/40'
   260→   260→            }`}
   261→   261→          />
   262→   262→        ))}
   263→   263→        {/* playhead marker */}
   264→   264→        <div
   265→   265→          ref={markerRef}
   266→   266→          aria-hidden
   267→   267→          className="pointer-events-none absolute inset-y-0 left-0 w-0.5 bg-teal-400 shadow-[0_0_6px_rgba(45,212,191,0.8)]"
   268→   268→        />
   269→   269→      </div>
   270→   270→      <div className="mt-0.5 flex justify-between font-mono text-[9px] text-zinc-500">
   271→   271→        <span>
   272→   272→          {hoverMs != null ? `scrub ${formatMs(hoverMs)}` : `${clip.frames.length} frames`}
   273→   273→        </span>
   274→   274→        <span>{formatMs(clip.frames.reduce((s, _, i) => s + frameMs(clip, i), 0))} cycle</span>
   275→   275→      </div>
   276→   276→    </div>
   277→   277→  )
   278→   278→}
