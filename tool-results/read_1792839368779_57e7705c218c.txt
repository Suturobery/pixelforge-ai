    30→    30→interface ActiveLoc {
    31→    31→  clip: Clip
    32→    32→  frame: Frame
    33→    33→  frameIndex: number
    34→    34→  layer: Layer

    37→    37→function locate(
    38→    38→  project: Project,
    39→    39→  activeClipId: string,
    40→    40→  activeFrameIndex: number,
    41→    41→  activeLayerId: string
    42→    42→): ActiveLoc | null {
    43→    43→  const clip =
    44→    44→    project.clips.find((c) => c.id === activeClipId) ?? project.clips[0]
    45→    45→  if (!clip || clip.frames.length === 0) return null
    46→    46→  const frameIndex = Math.min(Math.max(0, activeFrameIndex), clip.frames.length - 1)
    47→    47→  const frame = clip.frames[frameIndex]
    48→    48→  if (!frame || frame.layers.length === 0) return null
    49→    49→  const layer =
    50→    50→    frame.layers.find((l) => l.id === activeLayerId) ?? frame.layers[frame.layers.length - 1]
    51→    51→  return { clip, frame, frameIndex, layer }
    52→    52→}

    54→    54→/** Immutably replace a layer's buffer deep inside the project. */
    55→    55→function withLayerData(
    56→    56→  project: Project,
    57→    57→  clipId: string,
    58→    58→  frameId: string,
    59→    59→  layerId: string,
    60→    60→  data: Uint8ClampedArray
    61→    61→): Project {
    62→    62→  return {
    63→    63→    ...project,
    64→    64→    clips: project.clips.map((c) =>
    65→    65→      c.id !== clipId
    66→    66→        ? c
    67→    67→        : {
    68→    68→            ...c,
    69→    69→            frames: c.frames.map((f) =>
    70→    70→              f.id !== frameId
    71→    71→                ? f
    72→    72→                : { ...f, layers: f.layers.map((l) => (l.id !== layerId ? l : { ...l, data })) }
    73→    73→            ),
    74→    74→          }
    75→    75→    ),
    76→    76→  }
    77→    77→}

    79→    79→/**
    80→    80→ * Layers are CLIP-WIDE: every frame in a clip shares the same layer stack
    81→    81→ * (same names, same order — Aseprite-style). Each frame keeps its own cell
    82→    82→ * buffers/ids, so structural layer ops must apply to ALL frames of the clip,
    83→    83→ * not just the visible one (previously adding a layer on frame 1 made it
    84→    84→ * disappear when switching to frame 2 — the "why is there only 1 layer?" bug).
    85→    85→ */
    86→    86→function mapClipFrames(
    87→    87→  project: Project,
    88→    88→  clipId: string,
    89→    89→  fn: (frame: Frame, index: number) => Frame
    90→    90→): Project {
    91→    91→  return {
    92→    92→    ...project,
    93→    93→    clips: project.clips.map((c) =>
    94→    94→      c.id !== clipId ? c : { ...c, frames: c.frames.map((f, i) => fn(f, i)) }
    95→    95→    ),
    96→    96→  }
    97→    97→}

   307→   307→    if (!loc) return null
   308→   308→    const { clip, frame, frameIndex, layer } = loc
   309→   309→    const before = layer.data
   310→   310→    const working = new Uint8ClampedArray(before)
   311→   311→    const newFrame: Frame = {
   312→   312→      ...frame,
   313→   313→      layers: frame.layers.map((l) => (l.id === layer.id ? { ...l, data: working } : l)),
   314→   314→    }

   317→   317→      clips: s.project.clips.map((c) =>

   328→   328→  endStroke: (label, changed) => {

   330→   330→    const st = stroke
   331→   331→    stroke = null
   332→   332→    if (!changed) return
   333→   333→    const s = get()
   334→   334→    const entry: HistoryEntry = { kind: 'pixels', label, ...st, after: st.before }
   335→   335→    // Locate the working buffer in the *current* project (it may have moved if
   336→   336→    // a structural change happened mid-stroke — rare, but stay safe)
   337→   337→    const loc = locate(s.project, st.clipId, s.activeFrameIndex, st.layerId)
   338→   338→    let after = st.before
   339→   339→    if (loc && loc.clip.id === st.clipId && loc.frame.id === st.frameId && loc.layer.id === st.layerId) {
   340→   340→      after = loc.layer.data
   341→   341→    } else {
   342→   342→      const clip = s.project.clips.find((c) => c.id === st.clipId)
   343→   343→      const frame = clip?.frames.find((f) => f.id === st.frameId)
   344→   344→      const layer = frame?.layers.find((l) => l.id === st.layerId)
   345→   345→      if (layer) after = layer.data
   346→   346→      else return
   347→   347→    }
   348→   348→    entry.after = after
   349→   349→    set({
   350→   350→      past: [...s.past, entry].slice(-useAppSettings.getState().historyLimit),
   351→   351→      future: [],
   352→   352→      dirty: true,
   353→   353→    })
   354→   354→  },
   355→   355→
   356→   356→  cancelStroke: () => {
   357→   357→    if (!stroke) return
   358→   358→    const st = stroke
   359→   359→    stroke = null
   360→   360→    const s = get()
   361→   361→    set({
   362→   362→      project: withLayerData(s.project, st.clipId, st.frameId, st.layerId, st.before),
   363→   363→      renderTick: s.renderTick + 1,
   364→   364→    })
   365→   365→  },
   366→   366→
   367→   367→  bump: () => set((s) => ({ renderTick: s.renderTick + 1 })),
   368→   368→
   369→   369→  undo: () => {
   370→   370→    const s = get()
   371→   371→    if (stroke) return
   372→   372→    const past = [...s.past]
   373→   373→    while (past.length > 0) {
   374→   374→      const entry = past.pop() as HistoryEntry
   375→   375→      if (entry.kind === 'pixels') {
   376→   376→        const cur = locate(s.project, entry.clipId, s.activeFrameIndex, entry.layerId)
   377→   377→        const layer = cur && cur.clip.id === entry.clipId && cur.frame.id === entry.frameId && cur.layer.id === entry.layerId
   378→   378→          ? cur.layer
   379→   379→          : s.project.clips
   380→   380→              .find((c) => c.id === entry.clipId)
   381→   381→              ?.frames.find((f) => f.id === entry.frameId)
   382→   382→              ?.layers.find((l) => l.id === entry.layerId)
   383→   383→        if (!layer) continue // structure changed underneath; keep unwinding
   384→   384→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.before)
   385→   385→        const redoEntry: HistoryEntry = { ...entry, before: layer.data, after: entry.after }
   386→   386→        set({ project, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   387→   387→        return
   388→   388→      }
   389→   389→      const redoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: s.project, after: entry.after }
   390→   390→      set({ project: entry.before, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   391→   391→      return
   392→   392→    }
   393→   393→  },
   394→   394→
   395→   395→  redo: () => {
   396→   396→    const s = get()
   397→   397→    if (stroke) return
   398→   398→    const future = [...s.future]
   399→   399→    while (future.length > 0) {
   400→   400→      const entry = future.pop() as HistoryEntry
   401→   401→      if (entry.kind === 'pixels') {
   402→   402→        const layer = s.project.clips
   403→   403→          .find((c) => c.id === entry.clipId)
   404→   404→          ?.frames.find((f) => f.id === entry.frameId)
   405→   405→          ?.layers.find((l) => l.id === entry.layerId)
   406→   406→        if (!layer) continue
   407→   407→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.after)
   408→   408→        const undoEntry: HistoryEntry = { ...entry, before: entry.before, after: layer.data }
   409→   409→        set({ project, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   410→   410→        return
   411→   411→      }
   412→   412→      const undoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: entry.before, after: s.project }
   413→   413→      set({ project: entry.after, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   414→   414→      return
   415→   415→    }
   416→   416→  },
   417→   417→
   418→   418→  // --- structural commands ---
   419→   419→
   420→   420→  addFrame: (copyContent) => {
   421→   421→    const s = get()
   422→   422→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   423→   423→    if (!loc) return
   424→   424→    const insertAt = s.activeFrameIndex + 1
   425→   425→    const newFrame: Frame = {
   426→   426→      id: uid('fr'),
   427→   427→      layers: loc.frame.layers.map((l) => ({
   428→   428→        id: uid('ly'),
   429→   429→        name: l.name,
   430→   430→        visible: l.visible,
   431→   431→        opacity: l.opacity,
   432→   432→        data: copyContent
   433→   433→          ? new Uint8ClampedArray(l.data)
   434→   434→          : new Uint8ClampedArray(s.project.width * s.project.height * 4),
   435→   435→      })),
   436→   436→    }
   437→   437→    get().update((p) => {
   438→   438→      p.clips = p.clips.map((c) =>
   439→   439→        c.id !== loc.clip.id
   440→   440→          ? c
   441→   441→          : { ...c, frames: [...c.frames.slice(0, insertAt), newFrame, ...c.frames.slice(insertAt)] }
   442→   442→      )
   443→   443→    }, copyContent ? 'Duplicate frame' : 'Add frame')
   444→   444→    set({ activeFrameIndex: insertAt, activeLayerId: newFrame.layers.at(-1)?.id ?? '' })
   445→   445→    toast(
   446→   446→      copyContent
   447→   447→        ? { title: `Frame ${insertAt + 1} duplicated` }
   448→   448→        : {
   449→   449→            title: `Frame ${insertAt + 1} added (empty)`,
   450→   450→            description: `The canvas now shows the new frame — frame ${s.activeFrameIndex + 1} and its pixels are safe on the timeline below.`,
   451→   451→          }
   452→   452→    )
   453→   453→  },
   454→   454→
   455→   455→  deleteFrame: (index) => {
   456→   456→    const s = get()
   457→   457→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   458→   458→    if (!loc || loc.clip.frames.length <= 1) return
   459→   459→    get().update((p) => {
   460→   460→      p.clips = p.clips.map((c) =>
   461→   461→        c.id !== loc.clip.id ? c : { ...c, frames: c.frames.filter((_, i) => i !== index) }
   462→   462→      )
   463→   463→    }, 'Delete frame')
   464→   464→    const len = loc.clip.frames.length - 1
   465→   465→    set({ activeFrameIndex: Math.min(index, len - 1), activeLayerId: loc.layer.id })
   466→   466→  },
   467→   467→
   468→   468→  moveFrame: (from, to) => {
   469→   469→    const s = get()
   470→   470→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   471→   471→    if (!loc) return
   472→   472→    const len = loc.clip.frames.length
   473→   473→    if (to < 0 || to >= len || from === to) return
   474→   474→    get().update((p) => {
   475→   475→      p.clips = p.clips.map((c) => {
   476→   476→        if (c.id !== loc.clip.id) return c
   477→   477→        const frames = [...c.frames]
   478→   478→        const [f] = frames.splice(from, 1)
   479→   479→        frames.splice(to, 0, f)
   480→   480→        return { ...c, frames }
   481→   481→      })
   482→   482→    }, 'Reorder frame')
   483→   483→    set({ activeFrameIndex: to })
   484→   484→  },
   485→   485→
   486→   486→  addLayer: () => {
   487→   487→    const s = get()
   488→   488→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   489→   489→    if (!loc) return
   490→   490→    const name = `layer ${loc.frame.layers.length + 1}`
   491→   491→    // CLIP-WIDE: every frame gets its own cell of the new layer (same name,
   492→   492→    // independent buffer) so the stack looks identical on every frame
   493→   493→    get().update((p) => {
   494→   494→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   495→   495→        ...f,
   496→   496→        layers: [...f.layers, createLayer(s.project.width, s.project.height, name)],
   497→   497→      }))
   498→   498→    }, 'Add layer')
   499→   499→    // grab the id on the *active* frame (ids are per-frame)
   500→   500→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   501→   501→    const newLayerIdOnActiveFrame = cur?.frame.layers.at(-1)?.id ?? ''
   502→   502→    set({ activeLayerId: newLayerIdOnActiveFrame })
   503→   503→    toast({
   504→   504→      title: `Layer "${name}" added`,
   505→   505→      description: `Added to all ${loc.clip.frames.length} frame${loc.clip.frames.length === 1 ? '' : 's'} in this clip.`,
   506→   506→    })
   507→   507→  },

   509→   509→  deleteLayer: (layerId) => {
   510→   510→    const s = get()
   511→   511→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   512→   512→    if (!loc || loc.frame.layers.length <= 1) return
   513→   513→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   514→   514→    if (idx < 0) return
   515→   515→    // CLIP-WIDE: drop the same slot from every frame
   516→   516→    get().update((p) => {
   517→   517→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   518→   518→        ...f,
   519→   519→        layers: f.layers.filter((_, i) => i !== idx),
   520→   520→      }))
   521→   521→    }, 'Delete layer')
   522→   522→    const next = loc.frame.layers[Math.min(idx, loc.frame.layers.length - 2)]
   523→   523→    set({ activeLayerId: next?.id ?? '' })
   524→   524→  },

   526→   526→  duplicateLayer: (layerId) => {
   527→   527→    const s = get()
   528→   528→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   529→   529→    if (!loc) return
   530→   530→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   531→   531→    if (idx < 0) return
   532→   532→    const srcName = loc.frame.layers[idx].name
   533→   533→    // CLIP-WIDE: duplicate the slot in every frame (each frame copies its own
   534→   534→    // cell content), inserted directly above the original
   535→   535→    get().update((p) => {
   536→   536→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   537→   537→        const src = f.layers[idx]
   538→   538→        if (!src) return f
   539→   539→        const copy: Layer = {
   540→   540→          ...src,
   541→   541→          id: uid('ly'),
   542→   542→          name: `${srcName} copy`,
   543→   543→          data: new Uint8ClampedArray(src.data),
   544→   544→        }
   545→   545→        return {
   546→   546→          ...f,
   547→   547→          layers: [...f.layers.slice(0, idx + 1), copy, ...f.layers.slice(idx + 1)],
   548→   548→        }
   549→   549→      })
   550→   550→    }, 'Duplicate layer')
   551→   551→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   552→   552→    const copyIdOnActiveFrame = cur?.frame.layers[idx + 1]?.id ?? ''
   553→   553→    set({ activeLayerId: copyIdOnActiveFrame })
   554→   554→  },
   555→   555→
   556→   556→  mergeDown: (layerId) => {
   557→   557→    const s = get()
   558→   558→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   559→   559→    if (!loc) return
   560→   560→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   561→   561→    if (idx <= 0) return // nothing below
   562→   562→    const { width: w, height: h } = s.project
   563→   563→    const bottomId = loc.frame.layers[idx - 1].id
   564→   564→    // CLIP-WIDE: composite the same slot pair on every frame
   565→   565→    get().update((p) => {
   566→   566→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   567→   567→        if (idx >= f.layers.length) return f
   568→   568→        const fTop = f.layers[idx]
   569→   569→        const fBottom = f.layers[idx - 1]
   570→   570→        if (!fTop || !fBottom) return f
   571→   571→        const merged = new Uint8ClampedArray(fBottom.data)
   572→   572→        for (let i = 0; i < w * h; i++) {
   573→   573→          const ta = (fTop.data[i * 4 + 3] / 255) * fTop.opacity
   574→   574→          if (ta >= 1) {
   575→   575→            merged[i * 4] = fTop.data[i * 4]
   576→   576→            merged[i * 4 + 1] = fTop.data[i * 4 + 1]
   577→   577→            merged[i * 4 + 2] = fTop.data[i * 4 + 2]
   578→   578→            merged[i * 4 + 3] = fTop.data[i * 4 + 3]
   579→   579→          } else if (ta > 0) {
   580→   580→            const ba = merged[i * 4 + 3] / 255
   581→   581→            const outA = ta + ba * (1 - ta)
   582→   582→            if (outA === 0) continue
   583→   583→            for (let ch = 0; ch < 3; ch++) {
   584→   584→              merged[i * 4 + ch] = Math.round(
   585→   585→                (fTop.data[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   586→   586→              )
   587→   587→            }
   588→   588→            merged[i * 4 + 3] = Math.round(outA * 255)
   589→   589→          }
   590→   590→        }
   591→   591→        return {
   592→   592→          ...f,
   593→   593→          layers: f.layers.filter((_, i) => i !== idx).map((l, i) => (i === idx - 1 ? { ...l, data: merged } : l)),
   594→   594→        }
   595→   595→      })
   596→   596→    }, 'Merge down')
   597→   597→    set({ activeLayerId: bottomId })
   598→   598→  },
   599→   599→
   600→   600→  moveLayer: (layerId, dir) => {
   601→   601→    const s = get()
   602→   602→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   603→   603→    if (!loc) return
   604→   604→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   605→   605→    const to = idx + dir
   606→   606→    if (idx < 0 || to < 0 || to >= loc.frame.layers.length) return
   607→   607→    // CLIP-WIDE: reorder the same slot on every frame
   608→   608→    get().update((p) => {
   609→   609→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   610→   610→        if (to >= f.layers.length) return f
   611→   611→        const layers = [...f.layers]
   612→   612→        const [l] = layers.splice(idx, 1)
   613→   613→        if (!l) return f
   614→   614→        layers.splice(to, 0, l)
   615→   615→        return { ...f, layers }
   616→   616→      })
   617→   617→    }, 'Reorder layer')
   618→   618→    // the moved layer keeps its id on the active frame — selection unchanged
   619→   619→  },

   621→   621→  setLayerProp: (layerId, patch) => {

   634→   634→    }, 'Layer property')
   635→   635→  },

   750→   750→export function getActiveFrame(s: Pick<EditorState, 'project' | 'activeClipId' | 'activeFrameIndex'>): Frame {
