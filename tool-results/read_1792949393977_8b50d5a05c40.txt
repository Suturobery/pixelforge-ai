    34→function connectMockEditor(): Promise<void> {
    35→  return new Promise((resolve, reject) => {
    36→    const ws = new WebSocket(`ws://127.0.0.1:${PORT}/`);
    37→    ws.onopen = () => {
    38→      ws.send(JSON.stringify({ type: 'hello', bridgeVersion: 1, appVersion: 'test', mcpEnabled: true }));
    39→    };
    40→    ws.onerror = (e) => reject(new Error(`mock editor ws error: ${e}`));
    41→    ws.onmessage = (ev) => {
    42→      const msg = JSON.parse(String(ev.data)) as Record<string, unknown>;
    43→      if (msg.type === 'hello-ok') {
    44→        resolve();
    45→        return;
    46→      }
    47→      if (msg.type === 'tool-request') {
    48→        const responder = toolResponders.get(String(msg.tool));
    49→        const id = String(msg.id);
    50→        if (!responder) {
    51→          ws.send(JSON.stringify({ type: 'tool-result', id, ok: false, error: { code: 'no_mock_responder', message: `mock has no responder for ${msg.tool}` } }));
    52→          return;
    53→        }
    54→        try {
    55→          const out = responder((msg.args ?? {}) as Record<string, unknown>, { tier: String(msg.tier), clientName: String(msg.clientName) });
    56→          ws.send(JSON.stringify({ type: 'tool-result', id, ok: true, data: out, undoable: true, stateHash: 'mockhash-0001', summary: `mock:${msg.tool}` }));
    57→        } catch (err) {
    58→          ws.send(JSON.stringify({ type: 'tool-result', id, ok: false, error: { code: 'mock_error', message: err instanceof Error ? err.message : 'mock failed' } }));
    59→        }
    60→        return;
    61→      }
    62→      if (msg.type === 'ctrl-result') {
    63→        const pendingCtrl = ctrlWaiters.shift();
    64→        pendingCtrl?.(msg);
    65→        return;
    66→      }
    67→    };
    68→    wsSend = (m: unknown) => ws.send(JSON.stringify(m));
    69→    mockWs = ws;
    70→  });
    71→}
    72→
    73→let wsReadyResolve: (() => void) | null = null;
    74→let ctrlWaiters: Array<(msg: Record<string, unknown>) => void> = [];
    75→function ctrl<T = Record<string, unknown>>(payload: unknown): Promise<T> {
    76→  return new Promise((resolve) => {
    77→    ctrlWaiters.push((msg) => resolve(msg as T));
    78→    wsSend({ type: 'ctrl-request', id: `c${ctrlWaiters.length}`, ctrl: payload });
    79→  });
    80→}

    88→/** Full SDK client bound to a bearer token. */
    89→async function sdkClient(bearer: string): Promise<Client> {
    90→  const transport = new StreamableHTTPClientTransport(new URL(MCP_URL), {
    91→    requestInit: { headers: { authorization: `Bearer ${bearer}` } },
    92→  });
    93→  const client = new Client({ name: 'contract-test', version: '0.0.0' });
    94→  await client.connect(transport);
    95→  return client;
    96→}

    98→async function rawPost(body: string, headers: Record<string, string>): Promise<Response> {
    99→  return fetch(MCP_URL, { method: 'POST', headers: { 'content-type': 'application/json', accept: 'application/json', ...headers }, body });

   102→interface Envelope {
   103→  ok: boolean;

   111→function textEnvelope(result: unknown): Envelope {
   112→  const r = result as { content?: Array<{ type: string; text?: string }>; isError?: boolean; structuredContent?: unknown };
   113→  const text = r?.content?.find((c) => c.type === 'text')?.text ?? '{}';
   114→  return JSON.parse(text) as Envelope;
   115→}

   174→  test('oversized body is rejected (413)', async () => {
   175→    // NOTE: bun's fetch client (v1.3.14) poisons its connection pool when a
   176→    // server responds 413 while the client is still sending a 2 MB body —
   177→    // every later request on that pool hangs. Real MCP SDK clients never
   178→    // send oversized bodies, so the probe runs through curl (its own
   179→    // process/connection) to keep the suite hermetic.
   180→    const big = 'x'.repeat(2 * 1024 * 1024 + 10);
   181→    // a 2 MB argv element exceeds MAX_ARG_STRLEN — ship the body via a file
   182→    const bodyFile = `/tmp/mcp-oversized-${Date.now()}.bin`;
   183→    await Bun.write(bodyFile, big);
   184→    try {
   185→      // Bun.spawn (NOT spawnSync): the server runs in THIS process — a sync
   186→      // spawn would block the event loop and deadlock the request.
   187→      const proc = Bun.spawn([
   188→        'curl', '-s', '-o', '/dev/null', '-w', '%{http_code}',
   189→        '-X', 'POST', MCP_URL,
   190→        '-H', 'content-type: application/json',
   191→        '--data-binary', `@${bodyFile}`,
   192→        '--max-time', '15',
   193→      ]);
   194→      const code = await new Response(proc.stdout).text();
   195→      await proc.exited;
   196→      expect(code.trim()).toBe('413');
   197→    } finally {
   198→      await Bun.write(bodyFile, new Uint8Array(0));
   199→      const { rmSync } = await import('node:fs');
   200→      rmSync(bodyFile, { force: true });
   201→    }
   202→  });
