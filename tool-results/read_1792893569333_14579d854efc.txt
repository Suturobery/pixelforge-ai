    44→function buffer(paint: (x: number, y: number) => [number, number, number, number]): Uint8ClampedArray {
    45→  const buf = new Uint8ClampedArray(W * H * 4);
    46→  for (let y = 0; y < H; y++) {
    47→    for (let x = 0; x < W; x++) {
    48→      const i = (y * W + x) * 4;
    49→      const [r, g, b, a] = paint(x, y);
    50→      buf[i] = r;
    51→      buf[i + 1] = g;
    52→      buf[i + 2] = b;
    53→      buf[i + 3] = a;
    54→    }
    55→  }
    56→  return buf;
    57→}
    58→
    59→function cel(paint: (x: number, y: number) => [number, number, number, number], linked = false): Cel {
    60→  return { buffer: buffer(paint), linked };
    61→}

    84→    loopMode: 'forward',
    85→    layers: [layer('lay_bg', 'bg', { linked: true }), layer('lay_fg', 'fg')],
    86→    parts: [],
    87→    tags: [{ name: 'breathe', from: 1, to: 2 }],
    88→    frames: [
    89→      frame('f0', { lay_bg: bgCel, lay_fg: cel((x, y) => (x < 4 ? FG_RED : [0, 0, 0, 0])) }),

    94→      frame('f3', { lay_bg: bgCel }, 250),
    95→    ],
    96→  };
    97→  return { clip, bgCel };

   162→async function decodeArtifact(bytes: Uint8Array) {
   163→  return decodePng(bytes);

   166→function expectBuffersEqual(a: Uint8Array | Uint8ClampedArray, b: Uint8Array | Uint8ClampedArray, ctx?: string) {
   167→  expect(a.length).toBe(b.length);
   168→  for (let i = 0; i < a.length; i++) {
   169→    if (a[i] !== b[i]) {
   170→      throw new Error(`byte mismatch at ${i} (${ctx ?? ''}): ${a[i]} !== ${b[i]}`);
   171→    }
   172→  }
   173→}

   175→function pixel(buf: Uint8Array | Uint8ClampedArray, w: number, x: number, y: number): [number, number, number, number] {
   176→  const i = (y * w + x) * 4;
   177→  return [buf[i], buf[i + 1], buf[i + 2], buf[i + 3]];
   178→}

   180→/** Error-code helper: run fn, assert ExportError code. */
   181→async function expectExportError(fn: () => unknown, code: ExportError['code']) {
   182→  try {
   183→    await fn();
   184→    throw new Error(`expected ExportError ${code}, but nothing was thrown`);
   185→  } catch (err) {
   186→    expect(err).toBeInstanceOf(ExportError);
   187→    expect((err as ExportError).code).toBe(code);
   188→  }
   189→}
