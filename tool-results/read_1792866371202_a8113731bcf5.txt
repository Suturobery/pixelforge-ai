     1→// PixelForge AI — editor state store (Zustand) — Task 4 core-backed shim.
     2→//
     3→// Architecture (v3.1 §12 Task 4 "Port renderer onto core"):
     4→//   • @pixelforge/core owns the DOCUMENT: model v2 (per-clip layer stacks +
     5→//     per-frame cels, linked cels, trash), the CommandBus (the single
     6→//     mutation path — UI/AI all execute commands), tile-diff history with a
     7→//     memory budget, and the command journal (crash recovery source).
     8→//   • Zustand keeps UI + VIEW state (tool, zoom, pan, panels, playback,
     9→//     active pointers) and mirrors bus-derived values (project reference,
    10→//     dirty, undo/redo availability) so components stay plain subscribers.
    11→//   • Action names from the prototype store are PRESERVED — components call
    12→//     the same API; each action now maps onto a bus command instead of
    13→//     mutating a v1 document by hand.
    14→//
    15→// The bus lives OUTSIDE React. Every bus change event re-syncs the mirrored
    16→// slice; mid-stroke painting bumps `renderTick` only (no bus traffic).

    79→/** Settings `historyLimit` presets map onto v3.1 byte budgets (§2.5). */
    80→function historyBudgetBytes(limit: number): number {
    81→  switch (limit) {
    82→    case 20: return 16 * 1024 * 1024
    83→    case 40: return 32 * 1024 * 1024
    84→    case 100: return 64 * 1024 * 1024
    85→    case 200: return 128 * 1024 * 1024
    86→    default: return 64 * 1024 * 1024
    87→  }
    88→}
    89→bus.setHistoryBudget(historyBudgetBytes(useAppSettings.getState().historyLimit))
    90→useAppSettings.subscribe((s, prev) => {
    91→  if (s.historyLimit !== prev.historyLimit) {
    92→    bus.setHistoryBudget(historyBudgetBytes(s.historyLimit))
    93→  }
    94→})

    96→// Shim-registered command: palette presets replace the whole palette in ONE
    97→// undoable step (built-ins only add/remove single colors).
    98→bus.register({
    99→  type: 'palette.replaceAll',
   100→  defaultLabel: 'Apply palette',
   101→  validate(payload) {
   102→    const p = payload as { colors?: unknown }
   103→    const colors = p?.colors
   104→    if (!Array.isArray(colors) || colors.length === 0 || colors.length > 256) {
   105→      throw new CommandError('VALIDATION_FAILED', 'palette.replaceAll: colors must be a 1..256 array')
   106→    }
   107→    for (const c of colors) {
   108→      if (typeof c !== 'string' || !/^#[0-9a-f]{6}([0-9a-f]{2})?$/i.test(c)) {
   109→        throw new CommandError('VALIDATION_FAILED', `palette.replaceAll: bad hex "${String(c)}"`)
   110→      }
   111→    }
   112→  },
   113→  apply(payload, ctx) {
   114→    return {
   115→      project: { ...ctx.project, palette: [...(payload as { colors: string[] }).colors] },
   116→      label: 'Apply palette',
   117→    }
   118→  },
   119→})

   132→/** Resolve the active pointers against the LIVE document (safe fallbacks). */
   133→function activeRefs(s: Pick<EditorState, 'activeClipId' | 'activeFrameIndex' | 'activeLayerId'>): ActiveRefs | null {
   134→  const project = bus.getProject()
   135→  const clip = project.clips.find((c) => c.id === s.activeClipId) ?? project.clips[0]
   136→  if (!clip) return null
   137→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   138→  const frame = clip.frames[frameIndex]
   139→  if (!frame) return null
   140→  const layer = clip.layers.find((l) => l.id === s.activeLayerId) ?? clip.layers[clip.layers.length - 1]
   141→  if (!layer) return null
   142→  return { clipId: clip.id, frameId: frame.id, frameIndex, layerId: layer.id }
   143→}
   144→
   145→function findNewLayerId(before: Set<string>, project: Project, clipId: string): string | null {
   146→  const clip = project.clips.find((c) => c.id === clipId)
   147→  if (!clip) return null
   148→  for (let i = clip.layers.length - 1; i >= 0; i--) {
   149→    const l = clip.layers[i]
   150→    if (!before.has(l.id)) return l.id
   151→  }
   152→  return null
   153→}
   154→
   155→function layerIds(project: Project, clipId: string): Set<string> {
   156→  const clip = project.clips.find((c) => c.id === clipId)
   157→  return new Set(clip?.layers.map((l) => l.id) ?? [])
   158→}

   160→/**
   161→ * Structural sanity for documents entering the editor (Task 4 §5 data-loss
   162→ * protection): at least one clip, and every clip needs a frame to draw on
   163→ * and a layer to draw with. The core v1 shape guard is deliberately loose
   164→ * (serialization format check); THIS is the editor-entry gate. Degenerate
   165→ * rows/files are refused with a toast instead of limping or crashing.
   166→ */
   167→export function isUsableProject(p: unknown): p is Project {
   168→  if (!p || typeof p !== 'object') return false
   169→  const doc = p as Project
   170→  if (!Array.isArray(doc.clips) || doc.clips.length === 0) return false
   171→  return doc.clips.every(
   172→    (c) =>
   173→      c &&
   174→      Array.isArray(c.layers) &&
   175→      c.layers.length > 0 &&
   176→      Array.isArray(c.frames) &&
   177→      c.frames.length > 0
   178→  )
   179→}

   301→let docVersionCounter = 0
   302→let loadEpochCounter = 0
   303→
   304→function syncFromBus(extra: Partial<EditorState> = {}): void {
   305→  docVersionCounter += 1
   306→  const stats = bus.historyStats()
   307→  useEditor.setState({
   308→    project: bus.getProject(),
   309→    docVersion: docVersionCounter,
   310→    renderTick: docVersionCounter,
   311→    dirty: bus.isDirty(),
   312→    canUndo: bus.canUndo(),
   313→    canRedo: bus.canRedo(),
   314→    historyUndoDepth: stats.undoDepth,
   315→    historyRedoDepth: stats.redoDepth,
   316→    historyBytes: stats.bytes,
   317→    ...extra,
   318→  })
   319→  revalidatePointers()
   320→}
   321→
   322→/** Keep the active pointers resolvable after undo/load/structural edits. */
   323→function revalidatePointers(): void {
   324→  const s = useEditor.getState()
   325→  const project = bus.getProject()
   326→  let clip = project.clips.find((c) => c.id === s.activeClipId)
   327→  let patch: Partial<EditorState> | null = null
   328→  if (!clip) {
   329→    clip = project.clips[0]
   330→    if (!clip) return
   331→    patch = { activeClipId: clip.id }
   332→  }
   333→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   334→  const frame = clip.frames[frameIndex]
   335→  const layerExists = clip.layers.some((l) => l.id === s.activeLayerId)
   336→  const nextLayerId = layerExists ? s.activeLayerId : (clip.layers[clip.layers.length - 1]?.id ?? '')
   337→  if (frameIndex !== s.activeFrameIndex || !layerExists || patch) {
   338→    patch = { ...(patch ?? {}), activeFrameIndex: frameIndex, activeLayerId: nextLayerId }
   339→  }
   340→  if (patch && (patch.activeFrameIndex !== s.activeFrameIndex || patch.activeLayerId !== s.activeLayerId || patch.activeClipId !== s.activeClipId)) {
   341→    useEditor.setState(patch)
   342→  }
   343→}

   367→  hover: null,

   489→    try {
   490→      bus.redo()

   502→      bus.execute('frame.add', { clipId: refs.clipId, afterIndex: refs.frameIndex, copy: copyContent }, ORIGIN)
   503→    } catch {
   504→      return
   505→    }

   507→    set({ activeFrameIndex: insertAt, playing: false })
   508→    toast(
   509→      copyContent
   510→        ? { title: `Frame ${insertAt + 1} duplicated` }
   511→        : {
   512→            title: `Frame ${insertAt + 1} added (empty)`,

   515→    )
   516→  },
   517→
   518→  deleteFrame: (index) => {
   519→    const refs = activeRefs(get())
   520→    if (!refs) return
   521→    const clip = bus.getProject().clips.find((c) => c.id === refs.clipId)
   522→    if (!clip || clip.frames.length <= 1) return
   523→    try {
   524→      bus.execute('frame.delete', { clipId: refs.clipId, frameIndex: index }, ORIGIN)
   525→    } catch {
   526→      return
   527→    }
   528→    const newLen = clip.frames.length - 1
   529→    set({ activeFrameIndex: Math.min(Math.max(0, index), newLen - 1) })
   530→  },
   531→
   532→  moveFrame: (from, to) => {
   533→    const refs = activeRefs(get())
   534→    if (!refs) return
   535→    const clip = bus.getProject().clips.find((c) => c.id === refs.clipId)
   536→    if (!clip) return
   537→    if (to < 0 || to >= clip.frames.length || from === to) return
   538→    try {
   539→      bus.execute('frame.move', { clipId: refs.clipId, from, to }, ORIGIN)
   540→    } catch {
   541→      return
   542→    }
   543→    set({ activeFrameIndex: to })
   544→  },
   545→
   546→  addLayer: () => {
   547→    const refs = activeRefs(get())
   548→    if (!refs) return
   549→    const clip = bus.getProject().clips.find((c) => c.id === refs.clipId)
   550→    if (!clip) return
   551→    const name = `layer ${clip.layers.length + 1}`
   552→    try {
   553→      bus.execute('layer.add', { clipId: refs.clipId, name }, ORIGIN)
   554→    } catch {
   555→      return
   556→    }
   557→    // layer.add appends to the top — the new top layer IS the new layer.
   558→    const topId = bus.getProject().clips.find((c) => c.id === refs.clipId)?.layers.at(-1)?.id
   559→    if (topId) set({ activeLayerId: topId })
   560→    toast({
   561→      title: `Layer "${name}" added`,
   562→      description: `Added to all ${clip.frames.length} frame${clip.frames.length === 1 ? '' : 's'} in this clip.`,
   563→    })
   564→  },
   565→
   566→  deleteLayer: (layerId) => {
   567→    const refs = activeRefs(get())
   568→    if (!refs) return
   569→    const clip = bus.getProject().clips.find((c) => c.id === refs.clipId)
   570→    if (!clip || clip.layers.length <= 1) return
   571→    const idx = clip.layers.findIndex((l) => l.id === layerId)
   572→    if (idx < 0) return
   573→    try {
   574→      bus.execute('layer.delete', { clipId: refs.clipId, layerId }, ORIGIN)
   575→    } catch {
   576→      return
   577→    }
   578→    const next = clip.layers[Math.min(idx, clip.layers.length - 2)]
   579→    set({ activeLayerId: next?.id ?? clip.layers[0]?.id ?? '' })
   580→  },
   581→
   582→  duplicateLayer: (layerId) => {
   583→    const refs = activeRefs(get())
   584→    if (!refs) return
   585→    const before = layerIds(bus.getProject(), refs.clipId)
   586→    try {
   587→      bus.execute('layer.duplicate', { clipId: refs.clipId, layerId }, ORIGIN)
   588→    } catch {
   589→      return
   590→    }
   591→    const newId = findNewLayerId(before, bus.getProject(), refs.clipId)
   592→    if (newId) set({ activeLayerId: newId })
   593→  },
   594→
   595→  mergeDown: (layerId) => {
   596→    const refs = activeRefs(get())
   597→    if (!refs) return
   598→    const clip = bus.getProject().clips.find((c) => c.id === refs.clipId)
   599→    if (!clip) return
   600→    const idx = clip.layers.findIndex((l) => l.id === layerId)
   601→    if (idx <= 0) return // nothing below
   602→    const bottomId = clip.layers[idx - 1].id
   603→    try {
   604→      bus.execute('layer.mergeDown', { clipId: refs.clipId, layerId }, ORIGIN)
   605→    } catch {
   606→      return
   607→    }
   608→    set({ activeLayerId: bottomId })
   609→  },
   610→
   611→  moveLayer: (layerId, dir) => {
   612→    const refs = activeRefs(get())
   613→    if (!refs) return
   614→    try {
   615→      bus.execute('layer.move', { clipId: refs.clipId, layerId, dir }, ORIGIN)
   616→    } catch {
   617→      /* out of range no-ops in the command itself */
   618→    }
   619→  },
   620→
   621→  setLayerProp: (layerId, patch) => {
   622→    const refs = activeRefs(get())
   623→    if (!refs) return
   624→    try {
   625→      bus.execute('layer.setProps', { clipId: refs.clipId, layerId, patch }, ORIGIN)
   626→    } catch {
   627→      /* invalid patch — ignore */
   628→    }
   629→  },
   630→
   631→  addClip: (name) => {
   632→    const fps = Math.min(60, Math.max(1, useAppSettings.getState().defaultClipFps))
   633→    let clipId: string | null = null
   634→    try {
   635→      bus.execute('clip.add', { name, fps }, ORIGIN)
   636→      clipId = bus.getProject().clips.at(-1)?.id ?? null
   637→    } catch {
   638→      return
   639→    }
   640→    if (clipId) {
   641→      const clip = bus.getProject().clips.find((c) => c.id === clipId)
   642→      set({
   643→        activeClipId: clipId,
   644→        activeFrameIndex: 0,
   645→        activeLayerId: clip?.layers[0]?.id ?? '',
   646→        playing: false,
   647→      })
   648→    }
   649→  },
   650→
   651→  deleteClip: (clipId) => {
   652→    const s = get()
   653→    const project = bus.getProject()
   654→    if (project.clips.length <= 1) return
   655→    const idx = project.clips.findIndex((c) => c.id === clipId)
   656→    if (idx < 0) return
   657→    try {
   658→      bus.execute('clip.delete', { clipId }, ORIGIN)
   659→    } catch {
   660→      return
   661→    }
   662→    const next = bus.getProject().clips[Math.min(idx, project.clips.length - 2)]
   663→    set({
   664→      activeClipId: next.id,
   665→      activeFrameIndex: 0,
   666→      activeLayerId: next.layers.at(-1)?.id ?? '',
   667→      playing: false,
   668→    })
   669→  },
   670→
   671→  setClipProp: (clipId, patch) => {
   672→    try {
   673→      bus.execute('clip.setProps', { clipId, patch }, ORIGIN)
   674→    } catch {
   675→      /* invalid patch — ignore */
   676→    }
   677→  },
   678→
   679→  addPaletteColor: (hex) => {
   680→    if (bus.getProject().palette.includes(hex.toLowerCase())) return
   681→    try {
   682→      bus.execute('palette.addColor', { hex }, ORIGIN)
   683→    } catch {
   684→      /* bad hex — ignore */
   685→    }
   686→  },
   687→
   688→  removePaletteColor: (hex) => {
   689→    if (!bus.getProject().palette.includes(hex.toLowerCase())) return
   690→    try {
   691→      bus.execute('palette.removeColor', { hex }, ORIGIN)
   692→    } catch {
   693→      /* ignore */
   694→    }
   695→  },
   696→
   697→  setPalette: (colors, label) => {
   698→    try {
   699→      bus.execute('palette.replaceAll', { colors }, ORIGIN, { label })
   700→    } catch {
   701→      /* bad colors — ignore */
   702→    }
   703→  },
   704→
   705→  // --- AI candidate import (one undoable transaction each) ---
   706→
   707→  importCandidateAsLayer: (buffer, name) => {
   708→    const refs = activeRefs(get())
   709→    if (!refs) return
   710→    let newLayerId: string | null = null
   711→    try {
   712→      bus.transaction('Import AI candidate as layer', 'ai', () => {
   713→        bus.execute('layer.add', { clipId: refs.clipId, name }, 'ai')
   714→        newLayerId = bus.getProject().clips.find((c) => c.id === refs.clipId)?.layers.at(-1)?.id ?? null
   715→        if (!newLayerId) throw new CommandError('INVALID_STATE', 'layer add produced no id')
   716→        bus.execute(
   717→          'pixels.applyOp',
   718→          { clipId: refs.clipId, frameId: refs.frameId, layerId: newLayerId, ops: [{ op: 'setBuffer', buffer: new Uint8ClampedArray(buffer) }] },
   719→          'ai'
   720→        )
   721→      })
   722→    } catch {
   723→      toast({ title: 'Import failed', description: 'The candidate could not be imported.', variant: 'destructive' })
   724→      return
   725→    }
   726→    if (newLayerId) set({ activeLayerId: newLayerId })
   727→  },
   728→
   729→  importCandidateAsFrame: (buffer, name) => {
   730→    const refs = activeRefs(get())
   731→    if (!refs) return
   732→    const insertAt = refs.frameIndex + 1
   733→    let ok = false
   734→    try {
   735→      bus.transaction('Import AI candidate as frame', 'ai', () => {
   736→        bus.execute('frame.add', { clipId: refs.clipId, afterIndex: refs.frameIndex, copy: false }, 'ai')
   737→        const clip = bus.getProject().clips.find((c) => c.id === refs.clipId)
   738→        const newFrame = clip?.frames[insertAt]
   739→        if (!newFrame) throw new CommandError('INVALID_STATE', 'frame add produced no frame')
   740→        bus.execute(
   741→          'pixels.applyOp',
   742→          { clipId: refs.clipId, frameId: newFrame.id, layerId: refs.layerId, ops: [{ op: 'setBuffer', buffer: new Uint8ClampedArray(buffer) }] },
   743→          'ai'
   744→        )
   745→        ok = true
   746→      })
   747→    } catch {
   748→      toast({ title: 'Import failed', description: 'The candidate could not be imported.', variant: 'destructive' })
   749→      return
   750→    }
   751→    if (ok) set({ activeFrameIndex: insertAt, playing: false })
   752→  },
   753→
   754→  // --- selection & ui ---

   773→  },

   818→export function getActiveClip(s: Pick<EditorState, 'project' | 'activeClipId'>): Project['clips'][number] {
   819→  return s.project.clips.find((c) => c.id === s.activeClipId) ?? s.project.clips[0]

   822→export function getActiveFrame(
   823→  s: Pick<EditorState, 'project' | 'activeClipId' | 'activeFrameIndex'>
   824→): Project['clips'][number]['frames'][number] {
