    34→    34→// --------------------------------------------------------- free-scale handles
    35→    35→// The marquee selection gets 8 handles (corners + edge midpoints). The dots
    36→    36→// are a visual overlay; the CANVAS hit-tests them in onPointerDown so they
    37→    37→// never block painting, and dragging one resizes the selection square freely
    38→    38→// — on release the content is NN-resampled into the new footprint (scaleTo).
    39→    39→
    40→    40→type ScaleHandleId = 'nw' | 'n' | 'ne' | 'e' | 'se' | 's' | 'sw' | 'w'
    41→    41→
    42→    42→const HANDLE_ANCHORS: Array<{ id: ScaleHandleId; fx: 0 | 0.5 | 1; fy: 0 | 0.5 | 1 }> = [
    43→    43→  { id: 'nw', fx: 0, fy: 0 },
    44→    44→  { id: 'n', fx: 0.5, fy: 0 },
    45→    45→  { id: 'ne', fx: 1, fy: 0 },
    46→    46→  { id: 'e', fx: 1, fy: 0.5 },
    47→    47→  { id: 'se', fx: 1, fy: 1 },
    48→    48→  { id: 's', fx: 0.5, fy: 1 },
    49→    49→  { id: 'sw', fx: 0, fy: 1 },
    50→    50→  { id: 'w', fx: 0, fy: 0.5 },
    51→    51→]
    52→    52→
    53→    53→const HANDLE_CURSOR: Record<ScaleHandleId, string> = {
    54→    54→  nw: 'nwse-resize',
    55→    55→  n: 'ns-resize',
    56→    56→  ne: 'nesw-resize',
    57→    57→  e: 'ew-resize',
    58→    58→  se: 'nwse-resize',
    59→    59→  s: 'ns-resize',
    60→    60→  sw: 'nesw-resize',
    61→    61→  w: 'ew-resize',
    62→    62→}
    63→    63→
    64→    64→/** The 8 scale-handle points of a rect, in canvas coordinates. */
    65→    65→function handlePoints(r: SelectionRect): Array<{ id: ScaleHandleId; x: number; y: number }> {
    66→    66→  return HANDLE_ANCHORS.map((h) => ({ id: h.id, x: r.x + r.w * h.fx, y: r.y + r.h * h.fy }))
    67→    67→}
    68→    68→
    69→    69→/** Shrink a rect into the canvas, never smaller than 1×1. */
    70→    70→function clampSelRect(r: SelectionRect, W: number, H: number): SelectionRect {
    71→    71→  const x = Math.max(0, Math.min(r.x, W - 1))
    72→    72→  const y = Math.max(0, Math.min(r.y, H - 1))
    73→    73→  return {
    74→    74→    x,
    75→    75→    y,
    76→    76→    w: Math.max(1, Math.min(r.x + r.w, W) - x),
    77→    77→    h: Math.max(1, Math.min(r.y + r.h, H) - y),
    78→    78→  }
    79→    79→}

    84→    84→  start: { x: number; y: number }
    85→    85→  last: { x: number; y: number }

    88→    88→  changed: boolean

    90→    90→  /** marquee-create / sel-move / sel-scale */
    91→    91→  selectionStart?: SelectionRect

    94→    94→  /** sel-move: currently applied pixel offset (committed on pointer-up). */
    95→    95→  moveDx?: number
    96→    96→  moveDy?: number
    97→    97→}

    99→    99→export function CanvasStage() {
   100→   100→  const containerRef = useRef<HTMLDivElement>(null)
   101→   101→  const canvasRef = useRef<HTMLCanvasElement>(null)
   102→   102→  const boardRef = useRef<HTMLDivElement>(null)
   103→   103→  const offRef = useRef<HTMLCanvasElement | null>(null)
   104→   104→  const interactionRef = useRef<Interaction | null>(null)
   105→   105→  const spaceRef = useRef(false)
   106→   106→  const [spaceHeld, setSpaceHeld] = useState(false)
   107→   107→
   108→   108→  const project = useEditor((s) => s.project)
   109→   109→  const activeClipId = useEditor((s) => s.activeClipId)
   110→   110→  const activeFrameIndex = useEditor((s) => s.activeFrameIndex)
   111→   111→  const zoom = useEditor((s) => s.zoom)
   112→   112→  const panX = useEditor((s) => s.panX)
   113→   113→  const panY = useEditor((s) => s.panY)
   114→   114→  const showGrid = useEditor((s) => s.showGrid)
   115→   115→  const onionSkin = useEditor((s) => s.onionSkin)
   116→   116→  const onionOpacity = useAppSettings((s) => s.onionOpacity)
   117→   117→  const onionPrevTint = useAppSettings((s) => s.onionPrevTint)
   118→   118→  const onionNextTint = useAppSettings((s) => s.onionNextTint)
   119→   119→  const tool = useEditor((s) => s.tool)
   120→   120→  const primary = useEditor((s) => s.primary)
   121→   121→  const brushSize = useEditor((s) => s.brushSize)
   122→   122→  const mirrorX = useEditor((s) => s.mirrorX)
   123→   123→  const ellipseFilled = useEditor((s) => s.ellipseFilled)
   124→   124→  const selection = useEditor((s) => s.selection)
   125→   125→  const playing = useEditor((s) => s.playing)
   126→   126→  const renderTick = useEditor((s) => s.renderTick)
   127→   127→  const hover = useEditor((s) => s.hover)
   128→   128→  const projectId = useEditor((s) => s.projectId)
   129→   129→  const filePath = useEditor((s) => s.filePath)
   130→   130→  const loadEpoch = useEditor((s) => s.loadEpoch)
   131→   131→  const [, forceRedraw] = useState(0)
   132→   132→  // marching-ants animation tick (only runs while a selection exists)
   133→   133→  const [antsTick, setAntsTick] = useState(0)
   134→   134→  // free-scale draft: the in-progress selection rect while a scale handle is
   135→   135→  // being dragged (drives the ants overlay; commits as ONE scaleTo op on
   136→   136→  // release). The ref mirrors the state so pointerup reads the latest value.
   137→   137→  const [scaleDraft, setScaleDraft] = useState<SelectionRect | null>(null)
   138→   138→  const scaleDraftRef = useRef<SelectionRect | null>(null)
   139→   139→  // true when the composited active frame is fully transparent — we show a
   140→   140→  // "this frame is empty" hint so a blank canvas is never mistaken for lost
   141→   141→  // pixel data (top user confusion after adding an empty frame)
   142→   142→  const [frameEmpty, setFrameEmpty] = useState(false)
   143→   143→
   144→   144→  // zoom anchored at the viewport center (buttons used to expand the canvas
   145→   145→  // from its top-left corner, drifting it off-screen)
   146→   146→  const zoomBy = useCallback((factor: number) => {
   147→   147→    const st = useEditor.getState()
   148→   148→    const container = containerRef.current
   149→   149→    const nz = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, st.zoom * factor))
   150→   150→    if (!container) {
   151→   151→      st.setZoom(nz)
   152→   152→      return
   153→   153→    }
   154→   154→    const rect = container.getBoundingClientRect()
   155→   155→    const cx = rect.width / 2
   156→   156→    const cy = rect.height / 2
   157→   157→    const wx = (cx - st.panX) / st.zoom
   158→   158→    const wy = (cy - st.panY) / st.zoom
   159→   159→    st.setZoom(nz)
   160→   160→    st.setPan(cx - wx * nz, cy - wy * nz)
   161→   161→  }, [])
   162→   162→
   163→   163→  const { width: W, height: H } = project
   164→   164→
   165→   165→  // ------------------------------------------------------------- composite
   166→   166→  useEffect(() => {
   167→   167→    if (!offRef.current) offRef.current = document.createElement('canvas')
   168→   168→    const off = offRef.current
   169→   169→    if (off.width !== W || off.height !== H) {
   170→   170→      off.width = W
   171→   171→      off.height = H
   172→   172→    }
   173→   173→    const ctx = off.getContext('2d')
   174→   174→    if (!ctx) return
   175→   175→    const st = useEditor.getState()
   176→   176→    const clip = getActiveClip(st)
   177→   177→    const len = clip.frames.length
   178→   178→    const idx = Math.min(activeFrameIndex, len - 1)
   179→   179→    const frame = clip.frames[idx]
   180→   180→    if (!frame) {
   181→   181→      ctx.clearRect(0, 0, W, H)
   182→   182→      return
   183→   183→    }
   184→   184→    // Onion skin (Task 6 §4): prev/next counts are DOCUMENT settings
   185→   185→    // (undoable via settings.setOnion); opacity + tints are view prefs.
   186→   186→    // Ghosts wrap the loop seam only when the clip actually loops.
   187→   187→    const ghosts: OnionGhost[] = []
   188→   188→    if (onionSkin && len > 1) {
   189→   189→      const { prev: prevCount, next: nextCount } = st.project.settings.onion
   190→   190→      for (let d = 1; d <= Math.min(prevCount, ONION_MAX); d++) {
   191→   191→        const i = idx - d
   192→   192→        if (i < 0 && !clip.loop) break
   193→   193→        const f = clip.frames[((i % len) + len) % len]
   194→   194→        if (f && f !== frame) {
   195→   195→          ghosts.push({ frame: f, depth: d - 1, dir: 'prev', tint: onionPrevTint, opacity: onionOpacity })
   196→   196→        }
   197→   197→      }
   198→   198→      for (let d = 1; d <= Math.min(nextCount, ONION_MAX); d++) {
   199→   199→        const i = idx + d
   200→   200→        if (i > len - 1 && !clip.loop) break
   201→   201→        const f = clip.frames[i % len]
   202→   202→        if (f && f !== frame) {
   203→   203→          ghosts.push({ frame: f, depth: d - 1, dir: 'next', tint: onionNextTint, opacity: onionOpacity })
   204→   204→        }
   205→   205→      }
   206→   206→    }
   207→   207→    drawFrame(ctx, clip, frame, W, H, { ghosts })
   208→   208→    // detect a fully-empty ACTIVE frame — scan the cel buffers directly so
   209→   209→    // onion-skin ghosts don't mask the "empty frame" hint
   210→   210→    let hasPixel = false
   211→   211→    for (const layer of clip.layers) {
   212→   212→      if (!layer.visible) continue
   213→   213→      const cel = frame.cels[layer.id]
   214→   214→      if (!cel) continue // absent cel = transparent (C2)
   215→   215→      const d = cel.buffer
   216→   216→      for (let i = 3; i < d.length; i += 4) {
   217→   217→        if (d[i] > 0) {
   218→   218→          hasPixel = true
   219→   219→          break
   220→   220→        }
   221→   221→      }
   222→   222→      if (hasPixel) break
   223→   223→    }
   224→   224→    setFrameEmpty((cur) => (cur === hasPixel ? cur : !hasPixel))
   225→   225→  }, [project, activeClipId, activeFrameIndex, renderTick, onionSkin, onionOpacity, onionPrevTint, onionNextTint, W, H])
   226→   226→
   227→   227→  // marching ants: step the dash offset while a selection is visible
   228→   228→  useEffect(() => {
   229→   229→    if (!selection || playing) return
   230→   230→    const t = setInterval(() => setAntsTick((v) => v + 1), 120)
   231→   231→    return () => clearInterval(t)
   232→   232→  }, [selection, playing])
   233→   233→
   234→   234→  // ----------------------------------------------------------------- paint
   235→   235→  useEffect(() => {
   236→   236→    const canvas = canvasRef.current
   237→   237→    const board = boardRef.current
   238→   238→    if (!canvas || !board || !offRef.current) return
   239→   239→    const dpr = window.devicePixelRatio || 1
   240→   240→    const cw = W * zoom
   241→   241→    const ch = H * zoom
   242→   242→    const bw = Math.max(1, Math.round(cw * dpr))
   243→   243→    const bh = Math.max(1, Math.round(ch * dpr))
   244→   244→    if (canvas.width !== bw || canvas.height !== bh) {
   245→   245→      canvas.width = bw
   246→   246→      canvas.height = bh
   247→   247→    }
   248→   248→    canvas.style.width = `${cw}px`
   249→   249→    canvas.style.height = `${ch}px`
   250→   250→    board.style.width = `${cw}px`
   251→   251→    board.style.height = `${ch}px`
   252→   252→    board.style.transform = `translate(${panX}px, ${panY}px)`
   253→   253→
   254→   254→    const ctx = canvas.getContext('2d')
   255→   255→    if (!ctx) return
   256→   256→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   257→   257→    ctx.imageSmoothingEnabled = false
   258→   258→    ctx.clearRect(0, 0, cw, ch)
   259→   259→    ctx.drawImage(offRef.current, 0, 0, cw, ch)
   260→   260→
   261→   261→    // pixel grid
   262→   262→    if (showGrid && zoom >= 5) {
   263→   263→      ctx.lineWidth = 1
   264→   264→      for (let x = 0; x <= W; x++) {
   265→   265→        const strong = x % 8 === 0
   266→   266→        ctx.strokeStyle = strong ? 'rgba(255,255,255,0.16)' : 'rgba(255,255,255,0.07)'
   267→   267→        const px = Math.round(x * zoom * dpr) / dpr + 0.5 / dpr
   268→   268→        ctx.beginPath()
   269→   269→        ctx.moveTo(px, 0)
   270→   270→        ctx.lineTo(px, ch)
   271→   271→        ctx.stroke()
   272→   272→      }
   273→   273→      for (let y = 0; y <= H; y++) {
   274→   274→        const strong = y % 8 === 0
   275→   275→        ctx.strokeStyle = strong ? 'rgba(255,255,255,0.16)' : 'rgba(255,255,255,0.07)'
   276→   276→        const py = Math.round(y * zoom * dpr) / dpr + 0.5 / dpr
   277→   277→        ctx.beginPath()
   278→   278→        ctx.moveTo(0, py)
   279→   279→        ctx.lineTo(cw, py)
   280→   280→        ctx.stroke()
   281→   281→      }
   282→   282→    }
   283→   283→
   284→   284→    // hover cursor (paint tools)
   285→   285→    if (hover && !playing && tool !== 'eyedropper' && tool !== 'move' && tool !== 'marquee') {
   286→   286→      const size = tool === 'fill' ? 1 : brushSize
   287→   287→      const off2 = Math.floor((Math.max(1, size) - 1) / 2)
   288→   288→      const hx = (hover.x - off2) * zoom
   289→   289→      const hy = (hover.y - off2) * zoom
   290→   290→      const hs = Math.max(1, size) * zoom
   291→   291→      ctx.strokeStyle = 'rgba(45, 212, 191, 0.9)'
   292→   292→      ctx.lineWidth = 1
   293→   293→      ctx.strokeRect(hx + 0.5, hy + 0.5, hs - 1, hs - 1)
   294→   294→    }
   295→   295→
   296→   296→    // marching-ants selection outline (two alternating dashed passes);
   297→   297→    // while a scale handle is dragged the DRAFT rect is outlined instead
   298→   298→    if (selection && !playing) {
   299→   299→      const ants = scaleDraft ?? selection
   300→   300→      const sx = ants.x * zoom
   301→   301→      const sy = ants.y * zoom
   302→   302→      const sw = ants.w * zoom
   303→   303→      const sh = ants.h * zoom
   304→   304→      const dash = Math.max(3, Math.round(zoom / 2))
   305→   305→      ctx.save()
   306→   306→      ctx.lineWidth = 1
   307→   307→      ctx.setLineDash([dash, dash])
   308→   308→      ctx.lineDashOffset = -(antsTick % 2) * dash // crawl
   309→   309→      ctx.strokeStyle = 'rgba(0, 0, 0, 0.95)'
   310→   310→      ctx.strokeRect(sx + 0.5, sy + 0.5, sw - 1, sh - 1)
   311→   311→      ctx.lineDashOffset = -(antsTick % 2) * dash + dash / 2
   312→   312→      ctx.strokeStyle = 'rgba(255, 255, 255, 0.95)'
   313→   313→      ctx.strokeRect(sx + 0.5, sy + 0.5, sw - 1, sh - 1)
   314→   314→      ctx.restore()
   315→   315→    }
   316→   316→    // NOTE: activeFrameIndex must be a dep — selectFrame() only composites the
   317→   317→    // offscreen buffer; without this dep the visible canvas keeps showing the
   318→   318→    // previous frame until the next hover/tick change ("frame click delay").
   319→   319→    // NOTE: the onion values must be deps too — toggling the switch or changing
   320→   320→    // opacity/tints re-composites the offscreen buffer only; without these deps
   321→   321→    // the visible canvas keeps showing the pre-onion blit (stale ghosts).
   322→   322→  }, [project, W, H, zoom, panX, panY, showGrid, renderTick, hover, tool, brushSize, playing, activeFrameIndex, selection, scaleDraft, ellipseFilled, antsTick, onionSkin, onionOpacity, onionPrevTint, onionNextTint])
   323→   323→
   324→   324→  // ------------------------------------------------------------ fit / zoom
   325→   325→  const fitView = useCallback(() => {
   326→   326→    const container = containerRef.current
   327→   327→    if (!container) return
   328→   328→    const rect = container.getBoundingClientRect()
   329→   329→    const z = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, Math.floor(Math.min((rect.width - 48) / W, (rect.height - 48) / H))))
   330→   330→    const st = useEditor.getState()
   331→   331→    st.setZoom(z)
   332→   332→    st.setPan((rect.width - W * z) / 2, (rect.height - H * z) / 2)
   333→   333→  }, [W, H])
   334→   334→
   335→   335→  useEffect(() => {
   336→   336→    fitView()
   337→   337→    // re-fit when a different project is loaded/newed (loadProject resets view state)
   338→   338→  }, [W, H, projectId, filePath, loadEpoch])
   339→   339→
   340→   340→  // refit when the stage container is resized (device rotation, breakpoint
   341→   341→  // flip desktop ↔ mobile, panel toggles) so the canvas stays centered
   342→   342→  useEffect(() => {
   343→   343→    const container = containerRef.current
   344→   344→    if (!container) return
   345→   345→    let lastW = container.clientWidth
   346→   346→    let lastH = container.clientHeight
   347→   347→    let raf = 0
   348→   348→    const check = () => {
   349→   349→      cancelAnimationFrame(raf)
   350→   350→      raf = requestAnimationFrame(() => {
   351→   351→        const w = container.clientWidth
   352→   352→        const h = container.clientHeight
   353→   353→        if (Math.abs(w - lastW) > 2 || Math.abs(h - lastH) > 2) {
   354→   354→          lastW = w
   355→   355→          lastH = h
   356→   356→          fitView()
   357→   357→        }
   358→   358→      })
   359→   359→    }
   360→   360→    const ro = new ResizeObserver(check)
   361→   361→    ro.observe(container)
   362→   362→    return () => {
   363→   363→      ro.disconnect()
   364→   364→      cancelAnimationFrame(raf)
   365→   365→    }
   366→   366→  }, [fitView])
   367→   367→
   368→   368→  // wheel zoom (anchored at cursor)
   369→   369→  useEffect(() => {
   370→   370→    const container = containerRef.current
   371→   371→    if (!container) return
   372→   372→    const onWheel = (e: WheelEvent) => {
   373→   373→      e.preventDefault()
   374→   374→      const st = useEditor.getState()
   375→   375→      const rect = container.getBoundingClientRect()
   376→   376→      const cx = e.clientX - rect.left
   377→   377→      const cy = e.clientY - rect.top
   378→   378→      const factor = Math.exp(-e.deltaY * 0.0015)
   379→   379→      const nz = Math.max(MIN_ZOOM, Math.min(MAX_ZOOM, st.zoom * factor))
   380→   380→      const wx = (cx - st.panX) / st.zoom
   381→   381→      const wy = (cy - st.panY) / st.zoom
   382→   382→      st.setZoom(nz)
   383→   383→      st.setPan(cx - wx * nz, cy - wy * nz)
   384→   384→    }
   385→   385→    container.addEventListener('wheel', onWheel, { passive: false })
   386→   386→    return () => container.removeEventListener('wheel', onWheel)
   387→   387→  }, [])
   388→   388→
   389→   389→  // space-to-pan tracking
   390→   390→  useEffect(() => {
   391→   391→    const down = (e: KeyboardEvent) => {
   392→   392→      if (e.code === 'Space' && !e.repeat) {
   393→   393→        const t = e.target as HTMLElement
   394→   394→        if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable)) return
   395→   395→        spaceRef.current = true
   396→   396→        setSpaceHeld(true)
   397→   397→      }
   398→   398→    }
   399→   399→    const up = (e: KeyboardEvent) => {
   400→   400→      if (e.code === 'Space') {
   401→   401→        spaceRef.current = false
   402→   402→        setSpaceHeld(false)
   403→   403→      }
   404→   404→    }
   405→   405→    window.addEventListener('keydown', down)
   406→   406→    window.addEventListener('keyup', up)
   407→   407→    return () => {
   408→   408→      window.removeEventListener('keydown', down)
   409→   409→      window.removeEventListener('keyup', up)
   410→   410→    }

   530→   530→              setScaleDraft({ ...sel })
   531→   531→              return
   532→   532→            }

   541→   541→            mode: 'sel-move',
   542→   542→            tool: 'marquee',

   546→   546→            snapshot: new Uint8ClampedArray(handle.buffer),

   548→   548→            selectionStart: sel,

   551→   551→          // drag OUTSIDE → draw a new selection rect (view state only)
   552→   552→          interactionRef.current = {
   553→   553→            mode: 'marquee-create',
   554→   554→            tool: 'marquee',
   555→   555→            start: px,
   556→   556→            last: px,
   557→   557→            changed: false,
   558→   558→          }

   567→   567→          mode: 'move',
   568→   568→          tool: 'move',

   575→   575→        break
   576→   576→      }

   580→   580→  const onPointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
   581→   581→    const st = useEditor.getState()

   692→   692→    }
   693→   693→
   694→   694→    if (it.mode === 'move' && it.snapshot) {

   704→   704→  const finishInteraction = () => {
   705→   705→    const st = useEditor.getState()
   706→   706→    const it = interactionRef.current
   707→   707→    interactionRef.current = null
   708→   708→    if (!it || it.mode === 'pan') return
   709→   709→    if (it.mode === 'paint' && it.tool) {
   710→   710→      st.endStroke(it.tool === 'eraser' ? 'Eraser' : 'Pencil', it.changed)
   711→   711→    } else if (it.mode === 'shape' && it.tool) {
   712→   712→      st.endStroke(it.tool === 'line' ? 'Line' : it.tool === 'ellipse' ? 'Ellipse' : 'Rect', it.changed)
   713→   713→    } else if (it.mode === 'move') {
   714→   714→      st.endStroke('Move layer', it.changed)
   715→   715→    } else if (it.mode === 'marquee-create') {
   716→   716→      // a click without a drag collapses to a deselect (predictable UX)
   717→   717→      const st2 = useEditor.getState()
   718→   718→      if (st2.selection && st2.selection.w <= 1 && st2.selection.h <= 1 && !it.changed) {
   719→   719→        st2.setSelection(null)
   720→   720→      }
   721→   721→    } else if (it.mode === 'sel-scale' && it.selectionStart) {
   722→   722→      // commit: NN-resample the content into the dragged footprint (one
   723→   723→      // undoable scaleTo op; the store clamps + no-ops identical footprints)
   724→   724→      const draft = scaleDraftRef.current
   725→   725→      scaleDraftRef.current = null
   726→   726→      setScaleDraft(null)
   727→   727→      if (draft) {
   728→   728→        const changed =
   729→   729→          draft.x !== it.selectionStart.x ||
   730→   730→          draft.y !== it.selectionStart.y ||
   731→   731→          draft.w !== it.selectionStart.w ||
   732→   732→          draft.h !== it.selectionStart.h
   733→   733→        if (changed) useEditor.getState().selectionScaleTo(draft)
   734→   734→      }
   735→   735→    } else if (it.mode === 'sel-move' && it.selectionStart) {
   736→   736→      // commit the final offset into history (the working buffer already
   737→   737→      // holds the preview; endStroke diffs it against the stroke start)
   738→   738→      useEditor
   739→   739→        .getState()
   740→   740→        .endStroke('Move selection', it.changed)

   743→   743→        useEditor.setState({
   744→   744→          selection: shiftRectClamped(
   745→   745→            it.selectionStart,
   746→   746→            it.moveDx ?? 0,

   748→   748→            project.width,
   749→   749→            project.height
   750→   750→          ),
   751→   751→        })
   752→   752→      }
   753→   753→    }

   781→   781→            ? 'move'
   782→   782→            : 'crosshair'
   783→   783→
   784→   784→  return (

   810→   810→          {scaleRect && (
