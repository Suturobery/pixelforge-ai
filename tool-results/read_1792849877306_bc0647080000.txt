    30→    30→interface ActiveLoc {
    31→    31→  clip: Clip
    32→    32→  frame: Frame
    33→    33→  frameIndex: number
    34→    34→  layer: Layer

    37→    37→function locate(
    38→    38→  project: Project,
    39→    39→  activeClipId: string,
    40→    40→  activeFrameIndex: number,
    41→    41→  activeLayerId: string
    42→    42→): ActiveLoc | null {
    43→    43→  const clip =
    44→    44→    project.clips.find((c) => c.id === activeClipId) ?? project.clips[0]
    45→    45→  if (!clip || clip.frames.length === 0) return null
    46→    46→  const frameIndex = Math.min(Math.max(0, activeFrameIndex), clip.frames.length - 1)
    47→    47→  const frame = clip.frames[frameIndex]
    48→    48→  if (!frame || frame.layers.length === 0) return null
    49→    49→  const layer =
    50→    50→    frame.layers.find((l) => l.id === activeLayerId) ?? frame.layers[frame.layers.length - 1]
    51→    51→  return { clip, frame, frameIndex, layer }
    52→    52→}

    54→    54→/** Immutably replace a layer's buffer deep inside the project. */
    55→    55→function withLayerData(
    56→    56→  project: Project,
    57→    57→  clipId: string,
    58→    58→  frameId: string,
    59→    59→  layerId: string,
    60→    60→  data: Uint8ClampedArray
    61→    61→): Project {
    62→    62→  return {
    63→    63→    ...project,
    64→    64→    clips: project.clips.map((c) =>
    65→    65→      c.id !== clipId
    66→    66→        ? c
    67→    67→        : {
    68→    68→            ...c,
    69→    69→            frames: c.frames.map((f) =>
    70→    70→              f.id !== frameId
    71→    71→                ? f
    72→    72→                : { ...f, layers: f.layers.map((l) => (l.id !== layerId ? l : { ...l, data })) }
    73→    73→            ),
    74→    74→          }
    75→    75→    ),
    76→    76→  }
    77→    77→}

    79→    79→/**
    80→    80→ * Layers are CLIP-WIDE: every frame in a clip shares the same layer stack
    81→    81→ * (same names, same order — Aseprite-style). Each frame keeps its own cell
    82→    82→ * buffers/ids, so structural layer ops must apply to ALL frames of the clip,
    83→    83→ * not just the visible one. Returns the NEW clips array (assign it back to
    84→    84→ * `p.clips` — returning a whole Project here once made `p.clips` a Project
    85→    85→ * object instead of an array, crashing every layer op with
    86→    86→ * "project.clips.find is not a function").
    87→    87→ */
    88→    88→function mapClipFrames(
    89→    89→  project: Project,
    90→    90→  clipId: string,
    91→    91→  fn: (frame: Frame, index: number) => Frame
    92→    92→): Clip[] {
    93→    93→  return project.clips.map((c) =>
    94→    94→    c.id !== clipId ? c : { ...c, frames: c.frames.map((f, i) => fn(f, i)) }
    95→    95→  )
    96→    96→}

   309→   309→    const working = new Uint8ClampedArray(before)
   310→   310→    const newFrame: Frame = {
   311→   311→      ...frame,
   312→   312→      layers: frame.layers.map((l) => (l.id === layer.id ? { ...l, data: working } : l)),

   334→   334→    // Locate the working buffer in the *current* project (it may have moved if
   335→   335→    // a structural change happened mid-stroke — rare, but stay safe)
   336→   336→    const loc = locate(s.project, st.clipId, s.activeFrameIndex, st.layerId)
   337→   337→    let after = st.before
   338→   338→    if (loc && loc.clip.id === st.clipId && loc.frame.id === st.frameId && loc.layer.id === st.layerId) {
   339→   339→      after = loc.layer.data
   340→   340→    } else {
   341→   341→      const clip = s.project.clips.find((c) => c.id === st.clipId)
   342→   342→      const frame = clip?.frames.find((f) => f.id === st.frameId)
   343→   343→      const layer = frame?.layers.find((l) => l.id === st.layerId)
   344→   344→      if (layer) after = layer.data
   345→   345→      else return
   346→   346→    }

   348→   348→    set({
   349→   349→      past: [...s.past, entry].slice(-useAppSettings.getState().historyLimit),

   352→   352→    })

   359→   359→    const s = get()
   360→   360→    set({
   361→   361→      project: withLayerData(s.project, st.clipId, st.frameId, st.layerId, st.before),
   362→   362→      renderTick: s.renderTick + 1,
   363→   363→    })
   364→   364→  },
   365→   365→
   366→   366→  bump: () => set((s) => ({ renderTick: s.renderTick + 1 })),
   367→   367→
   368→   368→  undo: () => {
   369→   369→    const s = get()
   370→   370→    if (stroke) return
   371→   371→    const past = [...s.past]
   372→   372→    while (past.length > 0) {
   373→   373→      const entry = past.pop() as HistoryEntry
   374→   374→      if (entry.kind === 'pixels') {
   375→   375→        const cur = locate(s.project, entry.clipId, s.activeFrameIndex, entry.layerId)
   376→   376→        const layer = cur && cur.clip.id === entry.clipId && cur.frame.id === entry.frameId && cur.layer.id === entry.layerId
   377→   377→          ? cur.layer
   378→   378→          : s.project.clips
   379→   379→              .find((c) => c.id === entry.clipId)
   380→   380→              ?.frames.find((f) => f.id === entry.frameId)
   381→   381→              ?.layers.find((l) => l.id === entry.layerId)
   382→   382→        if (!layer) continue // structure changed underneath; keep unwinding
   383→   383→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.before)
   384→   384→        const redoEntry: HistoryEntry = { ...entry, before: layer.data, after: entry.after }
   385→   385→        set({ project, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   386→   386→        return
   387→   387→      }
   388→   388→      const redoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: s.project, after: entry.after }
   389→   389→      set({ project: entry.before, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   390→   390→      return
   391→   391→    }
   392→   392→  },
   393→   393→
   394→   394→  redo: () => {
   395→   395→    const s = get()
   396→   396→    if (stroke) return
   397→   397→    const future = [...s.future]
   398→   398→    while (future.length > 0) {
   399→   399→      const entry = future.pop() as HistoryEntry
   400→   400→      if (entry.kind === 'pixels') {
   401→   401→        const layer = s.project.clips
   402→   402→          .find((c) => c.id === entry.clipId)
   403→   403→          ?.frames.find((f) => f.id === entry.frameId)
   404→   404→          ?.layers.find((l) => l.id === entry.layerId)
   405→   405→        if (!layer) continue
   406→   406→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.after)
   407→   407→        const undoEntry: HistoryEntry = { ...entry, before: entry.before, after: layer.data }
   408→   408→        set({ project, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   409→   409→        return
   410→   410→      }
   411→   411→      const undoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: entry.before, after: s.project }
   412→   412→      set({ project: entry.after, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   413→   413→      return
   414→   414→    }
   415→   415→  },

   417→   417→  // --- structural commands ---
   418→   418→
   419→   419→  addFrame: (copyContent) => {
   420→   420→    const s = get()
   421→   421→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   422→   422→    if (!loc) return
   423→   423→    const insertAt = s.activeFrameIndex + 1
   424→   424→    const newFrame: Frame = {
   425→   425→      id: uid('fr'),
   426→   426→      layers: loc.frame.layers.map((l) => ({
   427→   427→        id: uid('ly'),
   428→   428→        name: l.name,
   429→   429→        visible: l.visible,
   430→   430→        opacity: l.opacity,
   431→   431→        data: copyContent
   432→   432→          ? new Uint8ClampedArray(l.data)
   433→   433→          : new Uint8ClampedArray(s.project.width * s.project.height * 4),
   434→   434→      })),
   435→   435→    }
   436→   436→    get().update((p) => {
   437→   437→      p.clips = p.clips.map((c) =>
   438→   438→        c.id !== loc.clip.id
   439→   439→          ? c
   440→   440→          : { ...c, frames: [...c.frames.slice(0, insertAt), newFrame, ...c.frames.slice(insertAt)] }
   441→   441→      )
   442→   442→    }, copyContent ? 'Duplicate frame' : 'Add frame')
   443→   443→    set({ activeFrameIndex: insertAt, activeLayerId: newFrame.layers.at(-1)?.id ?? '' })
   444→   444→    toast(
   445→   445→      copyContent
   446→   446→        ? { title: `Frame ${insertAt + 1} duplicated` }
   447→   447→        : {
   448→   448→            title: `Frame ${insertAt + 1} added (empty)`,
   449→   449→            description: `The canvas now shows the new frame — frame ${s.activeFrameIndex + 1} and its pixels are safe on the timeline below.`,
   450→   450→          }
   451→   451→    )
   452→   452→  },
   453→   453→
   454→   454→  deleteFrame: (index) => {
   455→   455→    const s = get()
   456→   456→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   457→   457→    if (!loc || loc.clip.frames.length <= 1) return
   458→   458→    const keptName = loc.layer.name // layer ids are per-frame — match by name
   459→   459→    get().update((p) => {
   460→   460→      p.clips = p.clips.map((c) =>
   461→   461→        c.id !== loc.clip.id ? c : { ...c, frames: c.frames.filter((_, i) => i !== index) }
   462→   462→      )
   463→   463→    }, 'Delete frame')
   464→   464→    const len = loc.clip.frames.length - 1
   465→   465→    const nextIdx = Math.min(index, len - 1)
   466→   466→    const nextFrame = loc.clip.frames.filter((_, i) => i !== index)[nextIdx]
   467→   467→    const nextLayer =
   468→   468→      nextFrame?.layers.find((l) => l.name === keptName) ?? nextFrame?.layers.at(-1)
   469→   469→    set({ activeFrameIndex: nextIdx, activeLayerId: nextLayer?.id ?? '' })
   470→   470→  },

   472→   472→  moveFrame: (from, to) => {
   473→   473→    const s = get()
   474→   474→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   475→   475→    if (!loc) return
   476→   476→    const len = loc.clip.frames.length

   479→   479→      p.clips = p.clips.map((c) => {
   480→   480→        if (c.id !== loc.clip.id) return c
   481→   481→        const frames = [...c.frames]
   482→   482→        const [f] = frames.splice(from, 1)
   483→   483→        frames.splice(to, 0, f)
   484→   484→        return { ...c, frames }
   485→   485→      })
   486→   486→    }, 'Reorder frame')
   487→   487→    set({ activeFrameIndex: to })
   488→   488→  },
   489→   489→
   490→   490→  addLayer: () => {
   491→   491→    const s = get()
   492→   492→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   493→   493→    if (!loc) return
   494→   494→    const name = `layer ${loc.frame.layers.length + 1}`
   495→   495→    // CLIP-WIDE: every frame gets its own cell of the new layer (same name,
   496→   496→    // independent buffer) so the stack looks identical on every frame
   497→   497→    get().update((p) => {
   498→   498→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   499→   499→        ...f,
   500→   500→        layers: [...f.layers, createLayer(s.project.width, s.project.height, name)],
   501→   501→      }))
   502→   502→    }, 'Add layer')

   506→   506→    set({ activeLayerId: newLayerIdOnActiveFrame })

   508→   508→      title: `Layer "${name}" added`,
   509→   509→      description: `Added to all ${loc.clip.frames.length} frame${loc.clip.frames.length === 1 ? '' : 's'} in this clip.`,
   510→   510→    })
   511→   511→  },

   513→   513→  deleteLayer: (layerId) => {
   514→   514→    const s = get()
   515→   515→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   516→   516→    if (!loc || loc.frame.layers.length <= 1) return
   517→   517→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   518→   518→    if (idx < 0) return
   519→   519→    // CLIP-WIDE: drop the same slot from every frame
   520→   520→    get().update((p) => {
   521→   521→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   522→   522→        ...f,
   523→   523→        layers: f.layers.filter((_, i) => i !== idx),
   524→   524→      }))
   525→   525→    }, 'Delete layer')
   526→   526→    const next = loc.frame.layers[Math.min(idx, loc.frame.layers.length - 2)]
   527→   527→    set({ activeLayerId: next?.id ?? '' })
   528→   528→  },
   529→   529→
   530→   530→  duplicateLayer: (layerId) => {
   531→   531→    const s = get()
   532→   532→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   533→   533→    if (!loc) return
   534→   534→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   535→   535→    if (idx < 0) return
   536→   536→    const srcName = loc.frame.layers[idx].name
   537→   537→    // CLIP-WIDE: duplicate the slot in every frame (each frame copies its own
   538→   538→    // cell content), inserted directly above the original
   539→   539→    get().update((p) => {
   540→   540→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   541→   541→        const src = f.layers[idx]
   542→   542→        if (!src) return f
   543→   543→        const copy: Layer = {
   544→   544→          ...src,
   545→   545→          id: uid('ly'),
   546→   546→          name: `${srcName} copy`,
   547→   547→          data: new Uint8ClampedArray(src.data),
   548→   548→        }
   549→   549→        return {
   550→   550→          ...f,
   551→   551→          layers: [...f.layers.slice(0, idx + 1), copy, ...f.layers.slice(idx + 1)],
   552→   552→        }
   553→   553→      })
   554→   554→    }, 'Duplicate layer')
   555→   555→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   556→   556→    const copyIdOnActiveFrame = cur?.frame.layers[idx + 1]?.id ?? ''
   557→   557→    set({ activeLayerId: copyIdOnActiveFrame })
   558→   558→  },
   559→   559→
   560→   560→  mergeDown: (layerId) => {
   561→   561→    const s = get()
   562→   562→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   563→   563→    if (!loc) return
   564→   564→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   565→   565→    if (idx <= 0) return // nothing below
   566→   566→    const { width: w, height: h } = s.project
   567→   567→    const bottomId = loc.frame.layers[idx - 1].id
   568→   568→    // CLIP-WIDE: composite the same slot pair on every frame
   569→   569→    get().update((p) => {
   570→   570→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   571→   571→        if (idx >= f.layers.length) return f
   572→   572→        const fTop = f.layers[idx]
   573→   573→        const fBottom = f.layers[idx - 1]
   574→   574→        if (!fTop || !fBottom) return f
   575→   575→        const merged = new Uint8ClampedArray(fBottom.data)
   576→   576→        for (let i = 0; i < w * h; i++) {
   577→   577→          const ta = (fTop.data[i * 4 + 3] / 255) * fTop.opacity
   578→   578→          if (ta >= 1) {
   579→   579→            merged[i * 4] = fTop.data[i * 4]
   580→   580→            merged[i * 4 + 1] = fTop.data[i * 4 + 1]
   581→   581→            merged[i * 4 + 2] = fTop.data[i * 4 + 2]
   582→   582→            merged[i * 4 + 3] = fTop.data[i * 4 + 3]
   583→   583→          } else if (ta > 0) {
   584→   584→            const ba = merged[i * 4 + 3] / 255
   585→   585→            const outA = ta + ba * (1 - ta)
   586→   586→            if (outA === 0) continue
   587→   587→            for (let ch = 0; ch < 3; ch++) {
   588→   588→              merged[i * 4 + ch] = Math.round(
   589→   589→                (fTop.data[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   590→   590→              )
   591→   591→            }
   592→   592→            merged[i * 4 + 3] = Math.round(outA * 255)
   593→   593→          }
   594→   594→        }
   595→   595→        return {
   596→   596→          ...f,
   597→   597→          layers: f.layers.filter((_, i) => i !== idx).map((l, i) => (i === idx - 1 ? { ...l, data: merged } : l)),
   598→   598→        }
   599→   599→      })
   600→   600→    }, 'Merge down')
   601→   601→    set({ activeLayerId: bottomId })
   602→   602→  },
   603→   603→
   604→   604→  moveLayer: (layerId, dir) => {
   605→   605→    const s = get()
   606→   606→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   607→   607→    if (!loc) return
   608→   608→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)

   623→   623→  },

   754→   754→export function getActiveFrame(s: Pick<EditorState, 'project' | 'activeClipId' | 'activeFrameIndex'>): Frame {
