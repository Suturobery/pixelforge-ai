    98→    98→/** Settings `historyLimit` presets map onto v3.1 byte budgets (§2.5). */
    99→    99→function historyBudgetBytes(limit: number): number {
   100→   100→  switch (limit) {
   101→   101→    case 20: return 16 * 1024 * 1024
   102→   102→    case 40: return 32 * 1024 * 1024
   103→   103→    case 100: return 64 * 1024 * 1024
   104→   104→    case 200: return 128 * 1024 * 1024
   105→   105→    default: return 64 * 1024 * 1024
   106→   106→  }
   107→   107→}

   109→   109→useAppSettings.subscribe((s, prev) => {

   111→   111→    bus.setHistoryBudget(historyBudgetBytes(s.historyLimit))
   112→   112→  }

   151→   151→/** Resolve the active pointers against the LIVE document (safe fallbacks). */
   152→   152→function activeRefs(s: Pick<EditorState, 'activeClipId' | 'activeFrameIndex' | 'activeLayerId'>): ActiveRefs | null {
   153→   153→  const project = bus.getProject()
   154→   154→  const clip = project.clips.find((c) => c.id === s.activeClipId) ?? project.clips[0]
   155→   155→  if (!clip) return null
   156→   156→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   157→   157→  const frame = clip.frames[frameIndex]
   158→   158→  if (!frame) return null
   159→   159→  const layer = clip.layers.find((l) => l.id === s.activeLayerId) ?? clip.layers[clip.layers.length - 1]
   160→   160→  if (!layer) return null
   161→   161→  return { clipId: clip.id, frameId: frame.id, frameIndex, layerId: layer.id }
   162→   162→}

   164→   164→function findNewLayerId(before: Set<string>, project: Project, clipId: string): string | null {
   165→   165→  const clip = project.clips.find((c) => c.id === clipId)
   166→   166→  if (!clip) return null
   167→   167→  for (let i = clip.layers.length - 1; i >= 0; i--) {
   168→   168→    const l = clip.layers[i]
   169→   169→    if (!before.has(l.id)) return l.id
   170→   170→  }
   171→   171→  return null
   172→   172→}

   174→   174→function layerIds(project: Project, clipId: string): Set<string> {
   175→   175→  const clip = project.clips.find((c) => c.id === clipId)
   176→   176→  return new Set(clip?.layers.map((l) => l.id) ?? [])
   177→   177→}

   179→   179→/**
   180→   180→ * Stroke-flow helper for one-shot selection edits (cut/delete/paste):
   181→   181→ * begin → mutate the working buffer → end as ONE tile-diff history entry.
   182→   182→ * Any failure cancels the stroke — the document is never half-mutated.
   183→   183→ */
   184→   184→function withStroke(
   185→   185→  label: string,
   186→   186→  refs: ActiveRefs,
   187→   187→  fn: (handle: CoreStrokeHandle) => boolean
   188→   188→): boolean {
   189→   189→  let changed = false
   190→   190→  try {
   191→   191→    const handle = bus.beginStroke({ clipId: refs.clipId, frameId: refs.frameId, layerId: refs.layerId })
   192→   192→    changed = fn(handle)
   193→   193→    bus.endStroke(label, changed, ORIGIN)
   194→   194→    return changed
   195→   195→  } catch {
   196→   196→    if (bus.activeStroke) {
   197→   197→      try {
   198→   198→        bus.cancelStroke(ORIGIN)
   199→   199→      } catch {
   200→   200→        /* ignore */
   201→   201→      }
   202→   202→    }
   203→   203→    return false
   204→   204→  }
   205→   205→}

   207→   207→/**
   208→   208→ * Structural sanity for documents entering the editor (Task 4 §5 data-loss
   209→   209→ * protection): at least one clip, and every clip needs a frame to draw on
   210→   210→ * and a layer to draw with. The core v1 shape guard is deliberately loose
   211→   211→ * (serialization format check); THIS is the editor-entry gate. Degenerate
   212→   212→ * rows/files are refused with a toast instead of limping or crashing.
   213→   213→ */
   214→   214→export function isUsableProject(p: unknown): p is Project {
   215→   215→  if (!p || typeof p !== 'object') return false
   216→   216→  const doc = p as Project
   217→   217→  if (!Array.isArray(doc.clips) || doc.clips.length === 0) return false
   218→   218→  return doc.clips.every(
   219→   219→    (c) =>
   220→   220→      c &&
   221→   221→      Array.isArray(c.layers) &&
   222→   222→      c.layers.length > 0 &&
   223→   223→      Array.isArray(c.frames) &&
   224→   224→      c.frames.length > 0
   225→   225→  )
   226→   226→}

   380→   380→function syncFromBus(extra: Partial<EditorState> = {}): void {
   381→   381→  docVersionCounter += 1
   382→   382→  const stats = bus.historyStats()
   383→   383→  useEditor.setState({
   384→   384→    project: bus.getProject(),
   385→   385→    docVersion: docVersionCounter,
   386→   386→    renderTick: docVersionCounter,
   387→   387→    dirty: bus.isDirty(),
   388→   388→    canUndo: bus.canUndo(),
   389→   389→    canRedo: bus.canRedo(),
   390→   390→    historyUndoDepth: stats.undoDepth,
   391→   391→    historyRedoDepth: stats.redoDepth,
   392→   392→    historyBytes: stats.bytes,
   393→   393→    ...extra,
   394→   394→  })
   395→   395→  revalidatePointers()
   396→   396→}

   398→   398→/** Keep the active pointers resolvable after undo/load/structural edits. */
   399→   399→function revalidatePointers(): void {
   400→   400→  const s = useEditor.getState()
   401→   401→  const project = bus.getProject()
   402→   402→  let clip = project.clips.find((c) => c.id === s.activeClipId)
   403→   403→  let patch: Partial<EditorState> | null = null
   404→   404→  if (!clip) {
   405→   405→    clip = project.clips[0]
   406→   406→    if (!clip) return
   407→   407→    patch = { activeClipId: clip.id }
   408→   408→  }
   409→   409→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   410→   410→  const frame = clip.frames[frameIndex]
   411→   411→  const layerExists = clip.layers.some((l) => l.id === s.activeLayerId)
   412→   412→  const nextLayerId = layerExists ? s.activeLayerId : (clip.layers[clip.layers.length - 1]?.id ?? '')
   413→   413→  if (frameIndex !== s.activeFrameIndex || !layerExists || patch) {
   414→   414→    patch = { ...(patch ?? {}), activeFrameIndex: frameIndex, activeLayerId: nextLayerId }
   415→   415→  }
   416→   416→  if (patch && (patch.activeFrameIndex !== s.activeFrameIndex || patch.activeLayerId !== s.activeLayerId || patch.activeClipId !== s.activeClipId)) {
   417→   417→    useEditor.setState(patch)
   418→   418→  }
   419→   419→}
