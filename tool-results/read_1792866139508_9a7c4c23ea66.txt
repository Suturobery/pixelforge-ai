   414→   414→/**
   415→   415→ * Alpha compositing of the top cel onto the one below, per frame (ported
   416→   416→ * from the prototype store's mergeDown). The top layer is removed; the
   417→   417→ * operation is undoable via the structural snapshot, so it is not
   418→   418→ * trash-backed (documented deviation from delete semantics).
   419→   419→ */
   420→   420→export const layerMergeDown: CommandDefinition<{ clipId: string; layerId: string }> = {
   421→   421→  type: 'layer.mergeDown',
   422→   422→  defaultLabel: 'Merge down',
   423→   423→  validate(payload, ctx) {
   424→   424→    const p = assertObject(payload, 'layer.mergeDown');
   425→   425→    const { clipId, layerId } = parseLayerRef(p);
   426→   426→    const clip = findClip(ctx.project, clipId);
   427→   427→    if (!clip) failNotFound(`layer.mergeDown: clip "${clipId}" not found`);
   428→   428→    const found = findLayer(clip, layerId);
   429→   429→    if (!found) failNotFound(`layer.mergeDown: layer "${layerId}" not found`);
   430→   430→    if (found.index === 0) fail('layer.mergeDown: nothing below the bottom layer');
   431→   431→    if (found.layer.linked) {
   432→   432→      fail('layer.mergeDown: merging a linked layer is not supported (unlink first — C2)');
   433→   433→    }
   434→   434→  },
   435→   435→  apply(payload, ctx) {
   436→   436→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   437→   437→    const { index } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   438→   438→    const { width, height } = ctx.project;
   439→   439→    const project = mapClip(ctx.project, clip.id, (c) => ({
   440→   440→      ...c,
   441→   441→      layers: c.layers.filter((_, i) => i !== index),
   442→   442→      frames: c.frames.map((f) => {
   443→   443→        const top = f.cels[payload.layerId];
   444→   444→        const bottomLayer = c.layers[index - 1];
   445→   445→        if (!bottomLayer) return f;
   446→   446→        const bottom = f.cels[bottomLayer.id];
   447→   447→        if (!top && !bottom) return f;
   448→   448→        // Missing cel = transparent (C2) — merge treats it as empty.
   449→   449→        const merged = bottom
   450→   450→          ? new Uint8ClampedArray(bottom.buffer)
   451→   451→          : new Uint8ClampedArray(width * height * 4);
   452→   452→        if (top) {
   453→   453→          for (let i = 0; i < width * height; i++) {
   454→   454→            const ta = (top.buffer[i * 4 + 3] / 255) * (top.linked ? 1 : (c.layers[index]?.opacity ?? 1));
   455→   455→            if (ta >= 1) {
   456→   456→              merged[i * 4] = top.buffer[i * 4];
   457→   457→              merged[i * 4 + 1] = top.buffer[i * 4 + 1];
   458→   458→              merged[i * 4 + 2] = top.buffer[i * 4 + 2];
   459→   459→              merged[i * 4 + 3] = top.buffer[i * 4 + 3];
   460→   460→            } else if (ta > 0) {
   461→   461→              const ba = merged[i * 4 + 3] / 255;
   462→   462→              const outA = ta + ba * (1 - ta);
   463→   463→              if (outA === 0) continue;
   464→   464→              for (let ch = 0; ch < 3; ch++) {
   465→   465→                merged[i * 4 + ch] = Math.round(
   466→   466→                  (top.buffer[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   467→   467→                );
   468→   468→              }
   469→   469→              merged[i * 4 + 3] = Math.round(outA * 255);
   470→   470→            }
   471→   471→          }
   472→   472→        }
   473→   473→        return { ...f, cels: { ...f.cels, [bottomLayer.id]: { buffer: merged, linked: false } } };
   474→   474→      }),
   475→   475→    }));
   476→   476→    // Remove the top layer's cels everywhere.
   477→   477→    const cleaned = mapClip(project, clip.id, (c) => ({
   478→   478→      ...c,
   479→   479→      frames: c.frames.map((f) => {
   480→   480→        if (!f.cels[payload.layerId]) return f;
   481→   481→        const cels = { ...f.cels };
   482→   482→        delete cels[payload.layerId];
   483→   483→        return { ...f, cels };
   484→   484→      }),
   485→   485→    }));
   486→   486→    return { project: cleaned, label: 'Merge down' };
   487→   487→  },
   488→   488→};

   500→   500→    if (dir !== -1 && dir !== 1) fail('layer.move: dir must be -1 or 1');
   501→   501→  },
   502→   502→  apply(payload, ctx) {
   503→   503→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   504→   504→    const { index } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   505→   505→    const to = index + payload.dir;
   506→   506→    if (to < 0 || to >= clip.layers.length) {
   507→   507→      return { project: ctx.project, label: 'Reorder layer' };
   508→   508→    }
   509→   509→    const layers = [...clip.layers];
   510→   510→    const [moved] = layers.splice(index, 1);
   511→   511→    layers.splice(to, 0, moved);
   512→   512→    // Cel keys are layer ids — reordering the stack needs no cel changes.
   513→   513→    return { project: mapClip(ctx.project, clip.id, (c) => ({ ...c, layers })), label: 'Reorder layer' };
   514→   514→  },

   561→   561→/**
   562→   562→ * Linked-cel engine switch (C2/C19 — P0 engine support; UI toggle is P2).
   563→   563→ * ON  → pick the anchor frame's cel (first frame holding one) and share it
   564→   564→ *        across ALL frames (content of other frames collapses to the anchor).
   565→   565→ * OFF → break sharing: every frame gets an independent deep copy.
   566→   566→ */
   567→   567→export const layerSetLinked: CommandDefinition<{ clipId: string; layerId: string; linked: boolean }> = {
   568→   568→  type: 'layer.setLinked',
   569→   569→  defaultLabel: 'Linked cel',
   570→   570→  validate(payload, ctx) {
   571→   571→    const p = assertObject(payload, 'layer.setLinked');
   572→   572→    const { clipId, layerId } = parseLayerRef(p);
   573→   573→    const clip = findClip(ctx.project, clipId);
   574→   574→    if (!clip) failNotFound(`layer.setLinked: clip "${clipId}" not found`);
   575→   575→    if (!findLayer(clip, layerId)) failNotFound(`layer.setLinked: layer "${layerId}" not found`);
   576→   576→    if (typeof p.linked !== 'boolean') fail('layer.setLinked: "linked" must be a boolean');
   577→   577→  },
   578→   578→  apply(payload, ctx) {
   579→   579→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   580→   580→    const { layer } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   581→   581→    if (payload.linked === layer.linked) {
   582→   582→      return { project: ctx.project, label: 'Linked cel' };
   583→   583→    }
   584→   584→    if (payload.linked) {
   585→   585→      // Anchor = first frame that has a cel (else first frame → create).
   586→   586→      const anchorFrame = clip.frames.find((f) => f.cels[layer.id]) ?? clip.frames[0];
   587→   587→      if (!anchorFrame) fail('layer.setLinked: clip has no frames');
   588→   588→      const existing = anchorFrame.cels[layer.id];
   589→   589→      const shared: Cel = existing
   590→   590→        ? { buffer: existing.buffer, linked: true }
   591→   591→        : { buffer: new Uint8ClampedArray(ctx.project.width * ctx.project.height * 4), linked: true };
   592→   592→      const withFlag = mapClip(ctx.project, clip.id, (c) => ({
   593→   593→        ...c,
   594→   594→        layers: c.layers.map((l) => (l.id === layer.id ? { ...l, linked: true } : l)),
   595→   595→      }));
   596→   596→      return {
   597→   597→        project: setCelEverywhere(withFlag, clip.id, layer.id, shared),
   598→   598→        label: 'Link cel (shared)',
   599→   599→      };
   600→   600→    }
   601→   601→    // Unlink: give every frame its own copy of the current shared buffer.
   602→   602→    const project = mapClip(ctx.project, clip.id, (c) => ({
   603→   603→      ...c,
   604→   604→      layers: c.layers.map((l) => (l.id === layer.id ? { ...l, linked: false } : l)),
   605→   605→    }));
   606→   606→    const withCopies = mapClip(project, clip.id, (c) => ({
   607→   607→      ...c,
   608→   608→      frames: c.frames.map((f) => {
   609→   609→        const cel = f.cels[layer.id];
   610→   610→        if (!cel) return f;
   611→   611→        return { ...f, cels: { ...f.cels, [layer.id]: { buffer: cloneBuffer(cel.buffer), linked: false } } };
   612→   612→      }),
   613→   613→    }));
   614→   614→    return { project: withCopies, label: 'Unlink cel' };
   615→   615→  },
   616→   616→};

   664→   664→export const trashRestore: CommandDefinition<{ entryId: string }> = {
   665→   665→  type: 'trash.restore',
   666→   666→  defaultLabel: 'Restore from trash',
   667→   667→  validate(payload, ctx) {
   668→   668→    const p = assertObject(payload, 'trash.restore');
   669→   669→    const entryId = getString(p, 'entryId', 'trash.restore');
   670→   670→    if (!ctx.project.trash.some((t) => t.id === entryId)) {
   671→   671→      failNotFound(`trash.restore: entry "${entryId}" not found`);
   672→   672→    }
   673→   673→  },
   674→   674→  apply(payload, ctx) {
   675→   675→    const taken = takeTrash(ctx.project, payload.entryId);
   676→   676→    if (!taken) failNotFound('trash.restore: entry vanished');
   677→   677→    const { entry, project: without } = taken;
   678→   678→    const item = entry.item;
   679→   679→    if (item.kind === 'frame') {
   680→   680→      const clip = findClip(without, item.clipId);
   681→   681→      if (!clip) failNotFound('trash.restore: owning clip no longer exists');
   682→   682→      const index = Math.min(Math.max(0, item.frameIndex), clip.frames.length);
   683→   683→      return {
   684→   684→        project: mapClip(without, clip.id, (c) => ({
   685→   685→          ...c,
   686→   686→          frames: [...c.frames.slice(0, index), item.frame, ...c.frames.slice(index)],
   687→   687→        })),
   688→   688→        label: 'Restore frame',
   689→   689→      };
   690→   690→    }
   691→   691→    if (item.kind === 'layer') {
   692→   692→      const clip = findClip(without, item.clipId);
   693→   693→      if (!clip) failNotFound('trash.restore: owning clip no longer exists');
   694→   694→      const index = Math.min(Math.max(0, item.layerIndex), clip.layers.length);
   695→   695→      const layers = [...clip.layers.slice(0, index), item.layer, ...clip.layers.slice(index)];
   696→   696→      const restored = mapClip(without, clip.id, (c) => ({ ...c, layers }));
   697→   697→      // Restore cels to the frames they came from (missing frame ids skipped).
   698→   698→      const withCels = mapClip(restored, clip.id, (c) => ({
   699→   699→        ...c,
   700→   700→        frames: c.frames.map((f) => {
   701→   701→          const cel = item.celsByFrameId[f.id];
   702→   702→          if (!cel) return f;
   703→   703→          return { ...f, cels: { ...f.cels, [item.layer.id]: cel } };
   704→   704→        }),
   705→   705→      }));
   706→   706→      return { project: withCels, label: 'Restore layer' };
   707→   707→    }
   708→   708→    const index = Math.min(Math.max(0, item.clipIndex), without.clips.length);
   709→   709→    const clips = [...without.clips.slice(0, index), item.clip, ...without.clips.slice(index)];
   710→   710→    return { project: { ...without, clips }, label: 'Restore clip' };
   711→   711→  },
   712→   712→};
