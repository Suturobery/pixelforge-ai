     1→     1→// PixelForge AI — editor state store (Zustand) — Task 4 core-backed shim.
     2→     2→//
     3→     3→// Architecture (v3.1 §12 Task 4 "Port renderer onto core"):
     4→     4→//   • @pixelforge/core owns the DOCUMENT: model v2 (per-clip layer stacks +
     5→     5→//     per-frame cels, linked cels, trash), the CommandBus (the single
     6→     6→//     mutation path — UI/AI all execute commands), tile-diff history with a
     7→     7→//     memory budget, and the command journal (crash recovery source).
     8→     8→//   • Zustand keeps UI + VIEW state (tool, zoom, pan, panels, playback,
     9→     9→//     active pointers) and mirrors bus-derived values (project reference,
    10→    10→//     dirty, undo/redo availability) so components stay plain subscribers.
    11→    11→//   • Action names from the prototype store are PRESERVED — components call
    12→    12→//     the same API; each action now maps onto a bus command instead of
    13→    13→//     mutating a v1 document by hand.
    14→    14→//
    15→    15→// The bus lives OUTSIDE React. Every bus change event re-syncs the mirrored
    16→    16→// slice; mid-stroke painting bumps `renderTick` only (no bus traffic).

    79→    79→/** Settings `historyLimit` presets map onto v3.1 byte budgets (§2.5). */
    80→    80→function historyBudgetBytes(limit: number): number {
    81→    81→  switch (limit) {
    82→    82→    case 20: return 16 * 1024 * 1024
    83→    83→    case 40: return 32 * 1024 * 1024
    84→    84→    case 100: return 64 * 1024 * 1024
    85→    85→    case 200: return 128 * 1024 * 1024
    86→    86→    default: return 64 * 1024 * 1024
    87→    87→  }
    88→    88→}
    89→    89→bus.setHistoryBudget(historyBudgetBytes(useAppSettings.getState().historyLimit))
    90→    90→useAppSettings.subscribe((s, prev) => {
    91→    91→  if (s.historyLimit !== prev.historyLimit) {
    92→    92→    bus.setHistoryBudget(historyBudgetBytes(s.historyLimit))
    93→    93→  }

    96→    96→// Shim-registered command: palette presets replace the whole palette in ONE
    97→    97→// undoable step (built-ins only add/remove single colors).
    98→    98→bus.register({
    99→    99→  type: 'palette.replaceAll',
   100→   100→  defaultLabel: 'Apply palette',
   101→   101→  validate(payload) {
   102→   102→    const p = payload as { colors?: unknown }
   103→   103→    const colors = p?.colors
   104→   104→    if (!Array.isArray(colors) || colors.length === 0 || colors.length > 256) {
   105→   105→      throw new CommandError('VALIDATION_FAILED', 'palette.replaceAll: colors must be a 1..256 array')
   106→   106→    }
   107→   107→    for (const c of colors) {
   108→   108→      if (typeof c !== 'string' || !/^#[0-9a-f]{6}([0-9a-f]{2})?$/i.test(c)) {
   109→   109→        throw new CommandError('VALIDATION_FAILED', `palette.replaceAll: bad hex "${String(c)}"`)
   110→   110→      }
   111→   111→    }
   112→   112→  },
   113→   113→  apply(payload, ctx) {
   114→   114→    return {
   115→   115→      project: { ...ctx.project, palette: [...(payload as { colors: string[] }).colors] },

   125→   125→interface ActiveRefs {

   132→   132→/** Resolve the active pointers against the LIVE document (safe fallbacks). */
   133→   133→function activeRefs(s: Pick<EditorState, 'activeClipId' | 'activeFrameIndex' | 'activeLayerId'>): ActiveRefs | null {
   134→   134→  const project = bus.getProject()
   135→   135→  const clip = project.clips.find((c) => c.id === s.activeClipId) ?? project.clips[0]
   136→   136→  if (!clip) return null
   137→   137→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   138→   138→  const frame = clip.frames[frameIndex]
   139→   139→  if (!frame) return null
   140→   140→  const layer = clip.layers.find((l) => l.id === s.activeLayerId) ?? clip.layers[clip.layers.length - 1]
   141→   141→  if (!layer) return null
   142→   142→  return { clipId: clip.id, frameId: frame.id, frameIndex, layerId: layer.id }
   143→   143→}
   144→   144→
   145→   145→function findNewLayerId(before: Set<string>, project: Project, clipId: string): string | null {
   146→   146→  const clip = project.clips.find((c) => c.id === clipId)
   147→   147→  if (!clip) return null
   148→   148→  for (let i = clip.layers.length - 1; i >= 0; i--) {
   149→   149→    const l = clip.layers[i]
   150→   150→    if (!before.has(l.id)) return l.id
   151→   151→  }
   152→   152→  return null
   153→   153→}
   154→   154→
   155→   155→function layerIds(project: Project, clipId: string): Set<string> {
   156→   156→  const clip = project.clips.find((c) => c.id === clipId)
   157→   157→  return new Set(clip?.layers.map((l) => l.id) ?? [])
   158→   158→}

   160→   160→/**
   161→   161→ * Structural sanity for documents entering the editor (Task 4 §5 data-loss
   162→   162→ * protection): at least one clip, and every clip needs a frame to draw on
   163→   163→ * and a layer to draw with. The core v1 shape guard is deliberately loose
   164→   164→ * (serialization format check); THIS is the editor-entry gate. Degenerate
   165→   165→ * rows/files are refused with a toast instead of limping or crashing.
   166→   166→ */
   167→   167→export function isUsableProject(p: unknown): p is Project {
   168→   168→  if (!p || typeof p !== 'object') return false
   169→   169→  const doc = p as Project
   170→   170→  if (!Array.isArray(doc.clips) || doc.clips.length === 0) return false
   171→   171→  return doc.clips.every(
   172→   172→    (c) =>
   173→   173→      c &&
   174→   174→      Array.isArray(c.layers) &&
   175→   175→      c.layers.length > 0 &&
   176→   176→      Array.isArray(c.frames) &&
   177→   177→      c.frames.length > 0
   178→   178→  )
   179→   179→}

   301→   301→let docVersionCounter = 0
   302→   302→let loadEpochCounter = 0
   303→   303→
   304→   304→function syncFromBus(extra: Partial<EditorState> = {}): void {
   305→   305→  docVersionCounter += 1
   306→   306→  const stats = bus.historyStats()
   307→   307→  useEditor.setState({
   308→   308→    project: bus.getProject(),
   309→   309→    docVersion: docVersionCounter,
   310→   310→    renderTick: docVersionCounter,
   311→   311→    dirty: bus.isDirty(),
   312→   312→    canUndo: bus.canUndo(),
   313→   313→    canRedo: bus.canRedo(),
   314→   314→    historyUndoDepth: stats.undoDepth,
   315→   315→    historyRedoDepth: stats.redoDepth,
   316→   316→    historyBytes: stats.bytes,
   317→   317→    ...extra,
   318→   318→  })
   319→   319→  revalidatePointers()
   320→   320→}
   321→   321→
   322→   322→/** Keep the active pointers resolvable after undo/load/structural edits. */
   323→   323→function revalidatePointers(): void {
   324→   324→  const s = useEditor.getState()
   325→   325→  const project = bus.getProject()
   326→   326→  let clip = project.clips.find((c) => c.id === s.activeClipId)
   327→   327→  let patch: Partial<EditorState> | null = null
   328→   328→  if (!clip) {
   329→   329→    clip = project.clips[0]
   330→   330→    if (!clip) return
   331→   331→    patch = { activeClipId: clip.id }
   332→   332→  }
   333→   333→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   334→   334→  const frame = clip.frames[frameIndex]
   335→   335→  const layerExists = clip.layers.some((l) => l.id === s.activeLayerId)
   336→   336→  const nextLayerId = layerExists ? s.activeLayerId : (clip.layers[clip.layers.length - 1]?.id ?? '')
   337→   337→  if (frameIndex !== s.activeFrameIndex || !layerExists || patch) {
   338→   338→    patch = { ...(patch ?? {}), activeFrameIndex: frameIndex, activeLayerId: nextLayerId }
   339→   339→  }
   340→   340→  if (patch && (patch.activeFrameIndex !== s.activeFrameIndex || patch.activeLayerId !== s.activeLayerId || patch.activeClipId !== s.activeClipId)) {
   341→   341→    useEditor.setState(patch)
   342→   342→  }
   343→   343→}
