     1→     1→// PixelForge AI — Motion Composer panel (Task 10, v3.1 §4.1).
     2→     2→//
     3→     3→// Deterministic part-based animation workflow:
     4→     4→//   Select source clip + anchor frame → Tag parts (marquee or manual rect,
     5→     5→//   regions never cuts) → Pick preset (Idle/Walk/Run/Attack/Jump) →
     6→     6→//   Adjust parameters → Preview (pure plan, NEVER committed until Accept) →
     7→     7→//   QA warnings → Accept (one undoable bus command) → edit frames as usual.
     8→     8→//
     9→     9→// The preview is the same pure composer the bus command runs — what you
    10→    10→// accept is byte-identical to what you previewed. Nothing here mutates the
    11→    11→// document except the two CommandBus actions (part.set / motion.apply).

    64→    64→/** Composited display buffer for one preview frame (plan + static layers). */
    65→    65→function displayBuffer(
    66→    66→  project: Parameters<typeof composeFrameRgba>[0],
    67→    67→  clip: Parameters<typeof composeFrameRgba>[1],
    68→    68→  anchorFrame: Parameters<typeof composeFrameRgba>[2],
    69→    69→  plan: MotionPlan,
    70→    70→  frameIndex: number
    71→    71→): Uint8ClampedArray {
    72→    72→  const cels = { ...anchorFrame.cels }
    73→    73→  for (const [layerId, buf] of Object.entries(plan.frames[frameIndex].cels)) {
    74→    74→    cels[layerId] = { buffer: buf, linked: false }
    75→    75→  }
    76→    76→  return composeFrameRgba(project, clip, { id: anchorFrame.id, durationMs: 0, cels })
    77→    77→}

    79→    79→export function MotionPanel() {
    80→    80→  const project = useEditor((s) => s.project)
    81→    81→  const activeClipId = useEditor((s) => s.activeClipId)
    82→    82→  const selection = useEditor((s) => s.selection)
    83→    83→  const setRightTab = useEditor((s) => s.setRightTab)
    84→    84→  const setClipParts = useEditor((s) => s.setClipParts)
    85→    85→  const applyMotionPreset = useEditor((s) => s.applyMotionPreset)
    86→    86→  const docVersion = useEditor((s) => s.docVersion)
    87→    87→  const { toast } = useToast()
    88→    88→
    89→    89→  const clip = getActiveClip({ project, activeClipId })
    90→    90→  const layerId = clip.layers.at(-1)?.id ?? ''
    91→    91→
    92→    92→  // --- source / parts state ------------------------------------------------
    93→    93→  const [anchorIndex, setAnchorIndex] = useState(0)
    94→    94→  const [draftName, setDraftName] = useState('body')
    95→    95→  const [draftRect, setDraftRect] = useState<PartRect>({ x: 8, y: 8, w: 16, h: 16 })
    96→    96→  const [draftWholeLayer, setDraftWholeLayer] = useState(false)
    97→    97→  const [dirtyDraft, setDirtyDraft] = useState(false)
    98→    98→
    99→    99→  // --- preview state (declared before the sync effect reads setPlan) --------
   100→   100→  const [plan, setPlan] = useState<MotionPlan | null>(null)
   101→   101→  const [issues, setIssues] = useState<QAIssue[]>([])
   102→   102→  const [playing, setPlaying] = useState(false)
   103→   103→  const [previewIdx, setPreviewIdx] = useState(0)
   104→   104→  const [showOverlay, setShowOverlay] = useState(true)
   105→   105→  const [targetMode, setTargetMode] = useState<'new' | 'overwrite'>('new')
   106→   106→  const [newClipName, setNewClipName] = useState('idle-1')
   107→   107→
   108→   108→  // draft parts start as a copy of the committed clip parts; draft state is
   109→   109→  // reset whenever the clip or the document changes (render-time adjustment —
   110→   110→  // the React-endorsed pattern for derived-state resets, no sync effect)
   111→   111→  const [draftParts, setDraftParts] = useState<DraftPart[]>([])
   112→   112→  const syncKey = `${clip.id}:${docVersion}`
   113→   113→  const [lastSyncKey, setLastSyncKey] = useState(syncKey)
   114→   114→  if (lastSyncKey !== syncKey) {
   115→   115→    setLastSyncKey(syncKey)
   116→   116→    setDraftParts(clip.parts.map((p) => ({ name: p.name, layerId: p.layerId, ...(p.rect ? { rect: { ...p.rect } } : {}) })))
   117→   117→    setDirtyDraft(false)
   118→   118→    setAnchorIndex((i) => Math.min(i, clip.frames.length - 1))
   119→   119→    setPlan(null)
   120→   120→  }
   121→   121→
   122→   122→  // --- preset state ----------------------------------------------------------
   123→   123→  const [presetId, setPresetId] = useState<MotionPresetId>('idle')
   124→   124→  const preset: PresetDef = MOTION_PRESETS[presetId]
   125→   125→  const [params, setParams] = useState<MotionParams>({ ...MOTION_PARAMS_DEFAULTS, frameCount: 4 })
   126→   126→  // switching presets snaps frameCount back to the new preset's default
   127→   127→  const [lastPresetId, setLastPresetId] = useState(presetId)
   128→   128→  if (lastPresetId !== presetId) {
   129→   129→    setLastPresetId(presetId)
   130→   130→    setParams((p) => ({ ...p, frameCount: preset.defaultFrameCount }))
   131→   131→  }
   132→   132→
   133→   133→  const sourceCanvasRef = useRef<HTMLCanvasElement | null>(null)
   134→   134→  const previewCanvasRef = useRef<HTMLCanvasElement | null>(null)
   135→   135→  const stripRefs = useRef<Array<HTMLCanvasElement | null>>([])
   136→   136→
   137→   137→  const anchorFrame = clip.frames[Math.min(anchorIndex, clip.frames.length - 1)]
   138→   138→
   139→   139→  // --- preview rendering -----------------------------------------------------
   140→   140→  // plain function — the React Compiler memoizes; no hand-rolled dep arrays
   141→   141→  const renderPlan = () => {
   142→   142→    if (!anchorFrame) return
   143→   143→    let p: MotionPlan
   144→   144→    try {
   145→   145→      p = planMotion(project, clip, { preset: presetId, params, anchorFrameIndex: anchorIndex })
   146→   146→    } catch (err) {
   147→   147→      toast({ title: 'Preview failed', description: err instanceof Error ? err.message : 'Composer error', variant: 'destructive' })
   148→   148→      return
   149→   149→    }
   150→   150→    setPlan(p)
   151→   151→    setIssues(analyzeMotionPlan(project, clip, p))
   152→   152→    setPreviewIdx(0)
   153→   153→    setPlaying(false)
   154→   154→    // the suggested clip name follows the preset; the user can still edit it
   155→   155→    // before accepting (every fresh preview resets it)
   156→   156→    setNewClipName(`${presetId}-${project.clips.length}`)
   157→   157→  }
   158→   158→
   159→   159→  // draw the source canvas (anchor composite + part rects overlay)
   160→   160→  useEffect(() => {
   161→   161→    const canvas = sourceCanvasRef.current
   162→   162→    if (!canvas || !anchorFrame) return
   163→   163→    canvas.width = project.width
   164→   164→    canvas.height = project.height
   165→   165→    const ctx = canvas.getContext('2d')
   166→   166→    if (!ctx) return
   167→   167→    const rgba = composeFrameRgba(project, clip, anchorFrame)
   168→   168→    ctx.putImageData(new ImageData(new Uint8ClampedArray(rgba), project.width, project.height), 0, 0)
   169→   169→    if (!showOverlay) return
   170→   170→    // part rects (1px, color per part) + ground line + center axis
   171→   171→    draftParts.forEach((p, i) => {
   172→   172→      const r = p.rect ?? { x: 0, y: 0, w: project.width, h: project.height }
   173→   173→      ctx.strokeStyle = PART_COLORS[i % PART_COLORS.length]
   174→   174→      ctx.lineWidth = 1
   175→   175→      ctx.strokeRect(r.x + 0.5, r.y + 0.5, r.w - 1, r.h - 1)
   176→   176→    })
   177→   177→    ctx.strokeStyle = 'rgba(45, 212, 191, 0.55)'
   178→   178→    ctx.beginPath()
   179→   179→    ctx.moveTo(0, project.height - 0.5)
   180→   180→    ctx.lineTo(project.width, project.height - 0.5)
   181→   181→    ctx.moveTo(project.width / 2 + 0.5, 0)
   182→   182→    ctx.lineTo(project.width / 2 + 0.5, project.height)
   183→   183→    ctx.stroke()
   184→   184→  }, [project, clip, anchorFrame, draftParts, showOverlay, docVersion])
   185→   185→
   186→   186→  // draw the current preview frame + the strip thumbnails
   187→   187→  useEffect(() => {
   188→   188→    if (!plan || !previewCanvasRef.current) return
   189→   189→    const canvas = previewCanvasRef.current
   190→   190→    canvas.width = project.width
   191→   191→    canvas.height = project.height
   192→   192→    const ctx = canvas.getContext('2d')
   193→   193→    if (!ctx) return
   194→   194→    const rgba = displayBuffer(project, clip, anchorFrame, plan, previewIdx)
   195→   195→    ctx.putImageData(new ImageData(new Uint8ClampedArray(rgba), project.width, project.height), 0, 0)
   196→   196→    // preview frame part rects at their OFFSET positions (visual debugging)
   197→   197→    if (showOverlay) {
   198→   198→      ctx.strokeStyle = 'rgba(45, 212, 191, 0.55)'
   199→   199→      ctx.beginPath()
   200→   200→      ctx.moveTo(0, project.height - 0.5)
   201→   201→      ctx.lineTo(project.width, project.height - 0.5)
   202→   202→      ctx.stroke()
   203→   203→    }
   204→   204→    plan.frames.forEach((f, i) => {
   205→   205→      const c = stripRefs.current[i]
   206→   206→      if (!c) return
   207→   207→      c.width = project.width
   208→   208→      c.height = project.height
   209→   209→      const cctx = c.getContext('2d')
   210→   210→      if (!cctx) return
   211→   211→      const r = displayBuffer(project, clip, anchorFrame, plan, i)
   212→   212→      cctx.putImageData(new ImageData(new Uint8ClampedArray(r), project.width, project.height), 0, 0)
   213→   213→    })
   214→   214→  }, [plan, previewIdx, showOverlay, docVersion])
   215→   215→
   216→   216→  // playback timer (clip fps; play-once presets stop on the last frame)
   217→   217→  useEffect(() => {
   218→   218→    if (!playing || !plan) return
   219→   219→    const ms = Math.max(30, Math.round(1000 / clip.fps))
   220→   220→    const t = setInterval(() => {
   221→   221→      setPreviewIdx((i) => {
   222→   222→        const next = i + 1
   223→   223→        if (next >= plan.frames.length) {
   224→   224→          if (preset.playOnceSuggested) {
   225→   225→            setPlaying(false)
   226→   226→            return i
   227→   227→          }
   228→   228→          return 0
   229→   229→        }
   230→   230→        return next
   231→   231→      })
   232→   232→    }, ms)
   233→   233→    return () => clearInterval(t)
   234→   234→  }, [playing, plan, clip.fps, preset])
   235→   235→
   236→   236→  // --- actions -----------------------------------------------------------------
   237→   237→  const addDraftPart = () => {
   238→   238→    const name = draftName.trim()
   239→   239→    if (!name) return
   240→   240→    if (draftParts.some((p) => roleKey(p.name) === roleKey(name))) {
   241→   241→      toast({ title: 'Duplicate part name', description: `A part named "${name}" already exists — names are roles.`, variant: 'destructive' })
   242→   242→      return
   243→   243→    }
   244→   244→    if (draftParts.length >= 32) {
   245→   245→      toast({ title: 'Too many parts', description: 'At most 32 parts per clip.', variant: 'destructive' })
   246→   246→      return
   247→   247→    }
   248→   248→    setDraftParts((ps) => [...ps, { name, layerId, ...(draftWholeLayer ? {} : { rect: { ...draftRect } }) }])
   249→   249→    setDirtyDraft(true)
   250→   250→  }
   251→   251→
   252→   252→  const useSelectionAsRect = () => {
   253→   253→    if (!selection) {
   254→   254→      toast({ title: 'No selection', description: 'Drag a marquee on the canvas first, then press "Use selection".', variant: 'destructive' })
   255→   255→      return
   256→   256→    }
   257→   257→    setDraftRect({ ...selection })
   258→   258→    setDraftWholeLayer(false)
   259→   259→  }
   260→   260→
   261→   261→  const saveParts = () => {
   262→   262→    if (!setClipParts(clip.id, draftParts.map((p) => ({ ...p })))) return
   263→   263→    setDirtyDraft(false)
   264→   264→    toast({ title: 'Parts saved', description: `${draftParts.length} part${draftParts.length === 1 ? '' : 's'} tagged on "${clip.name}".` })
   265→   265→  }
   266→   266→
   267→   267→  const accept = () => {
   268→   268→    if (!plan) return
   269→   269→    const target =
   270→   270→      targetMode === 'new'
   271→   271→        ? ({ kind: 'new', name: newClipName.trim() || `${presetId}-clip`, fps: clip.fps } as const)
   272→   272→        : ({ kind: 'existing', clipId: clip.id, overwrite: true } as const)
   273→   273→    const createdClipId = applyMotionPreset({
   274→   274→      sourceClipId: clip.id,
   275→   275→      preset: presetId,
   276→   276→      params: { amplitude: params.amplitude, frameCount: plan.report.frameCount, gapFill: params.gapFill, smear: params.smear },
   277→   277→      anchorFrameIndex: anchorIndex,
   278→   278→      target,
   279→   279→    })
   280→   280→    if (!createdClipId) return
   281→   281→    const skipped = plan.report.skipped.length
   282→   282→    toast({
   283→   283→      title: 'Motion applied',
   284→   284→      description:
   285→   285→        targetMode === 'new'
   286→   286→          ? `New clip "${target.kind === 'new' ? target.name : ''}" created with ${plan.report.frameCount} frames.${skipped ? ` ${skipped} absent role(s) skipped.` : ''}`
   287→   287→          : `${plan.report.frameCount} frame${plan.report.frameCount === 1 ? '' : 's'} of "${clip.name}" rewritten (undoable).`,
   288→   288→    })
   289→   289→    setPlan(null)
   290→   290→    setIssues([])
   291→   291→    setPlaying(false)
   292→   292→    if (targetMode === 'new' && createdClipId) {
   293→   293→      const st = useEditor.getState()
   294→   294→      const created = st.project.clips.find((c) => c.id === createdClipId)
   295→   295→      useEditor.setState({
   296→   296→        activeClipId: createdClipId,
   297→   297→        activeFrameIndex: 0,
   298→   298→        activeLayerId: created?.layers.at(-1)?.id ?? st.activeLayerId,
   299→   299→      })
   300→   300→    }
   301→   301→  }
   302→   302→
   303→   303→  // --- dev probe for browser smokes (mirrors window.__pfAi discipline) ----------
   304→   304→  useEffect(() => {
   305→   305→    const w = window as unknown as Record<string, unknown>
   306→   306→    w.__pfMotion = {
   307→   307→      draftParts: () => draftParts.length,
   308→   308→      savedParts: () => clip.parts.map((p) => ({ name: p.name, rect: p.rect ?? null })),
   309→   309→      tagPart: (name: string, rect?: PartRect) =>
   310→   310→        setDraftParts((ps) => [...ps, { name, layerId, ...(rect ? { rect } : {}) }]),
   311→   311→      saveParts,
   312→   312→      selectPreset: (id: string) => {
   313→   313→        if ((PRESET_IDS as string[]).includes(id)) setPresetId(id as MotionPresetId)
   314→   314→      },
   315→   315→      setParam: (k: 'amplitude' | 'frameCount' | 'gapFill' | 'smear', v: number | string | boolean) =>
   316→   316→        setParams((p) => ({ ...p, [k]: v } as MotionParams)),
   317→   317→      preview: renderPlan,
   318→   318→      report: () => plan?.report ?? null,
   319→   319→      issues: () => issues,
   320→   320→      setTarget: (m: 'new' | 'overwrite') => setTargetMode(m),
   321→   321→      accept,
   322→   322→    }
   323→   323→    return () => {
   324→   324→      delete w.__pfMotion
   325→   325→    }
   326→   326→  })
   327→   327→
   328→   328→  // --------------------------------------------------------------------------------
   329→   329→  if (!anchorFrame) {
   330→   330→    return <div className="p-4 text-xs text-zinc-500">This clip has no frames.</div>
   331→   331→  }
   332→   332→
   333→   333→  return (
   334→   334→    <div className="pixelforge-scroll mx-auto min-h-0 w-full max-w-[520px] flex-1 space-y-3 overflow-y-auto p-3" data-testid="motion-panel">
   335→   335→      {/* ---------------------------- source ---------------------------- */}
   336→   336→      <section aria-label="Source" className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-3">
   337→   337→        <header className="mb-2 flex items-center gap-1.5 font-mono text-[10px] uppercase tracking-wider text-teal-300">
   338→   338→          <Film className="h-3 w-3" /> Source
   339→   339→        </header>
   340→   340→        <div className="flex items-center justify-between gap-2">
   341→   341→          <span className="truncate text-xs font-medium text-zinc-200">{clip.name}</span>
   342→   342→          <div className="flex items-center gap-1">
   343→   343→            <span className="font-mono text-[10px] text-zinc-500">anchor</span>
   344→   344→            <Select value={String(anchorIndex)} onValueChange={(v) => setAnchorIndex(Number(v))}>
   345→   345→              <SelectTrigger className="h-6 w-[70px] font-mono text-[11px]" aria-label="Anchor frame">
   346→   346→                <SelectValue />
   347→   347→              </SelectTrigger>
   348→   348→              <SelectContent className="max-h-48">
   349→   349→                {clip.frames.map((f, i) => (
   350→   350→                  <SelectItem key={f.id} value={String(i)} className="font-mono text-[11px]">
   351→   351→                    {i + 1}
   352→   352→                  </SelectItem>
   353→   353→                ))}
   354→   354→              </SelectContent>
   355→   355→            </Select>
   356→   356→          </div>
   357→   357→        </div>
   358→   358→        <p className="mt-1 font-mono text-[10px] text-zinc-500">
   359→   359→          every frame derives from frame {anchorIndex + 1} (R5) · {clip.fps} fps
   360→   360→        </p>
   361→   361→      </section>
   362→   362→
   363→   363→      {/* ---------------------------- parts ---------------------------- */}
   364→   364→      <section aria-label="Body parts" className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-3">
   365→   365→        <header className="mb-2 flex items-center justify-between">
   366→   366→          <span className="flex items-center gap-1.5 font-mono text-[10px] uppercase tracking-wider text-teal-300">
   367→   367→            <Tag className="h-3 w-3" /> Parts ({draftParts.length})
   368→   368→          </span>
   369→   369→          {dirtyDraft && (
   370→   370→            <div className="flex gap-1">
   371→   371→              <Button size="sm" className="h-6 bg-teal-600 px-2 font-mono text-[10px] hover:bg-teal-500" onClick={saveParts}>
   372→   372→                Save parts
   373→   373→              </Button>
   374→   374→              <Button
   375→   375→                size="sm"
   376→   376→                variant="ghost"
   377→   377→                className="h-6 px-2 font-mono text-[10px] text-zinc-400"
   378→   378→                onClick={() => {
   379→   379→                  setDraftParts(clip.parts.map((p) => ({ name: p.name, layerId: p.layerId, ...(p.rect ? { rect: { ...p.rect } } : {}) })))
   380→   380→                  setDirtyDraft(false)
   381→   381→                }}
   382→   382→              >
   383→   383→                Revert
   384→   384→              </Button>
   385→   385→            </div>
   386→   386→          )}
   387→   387→        </header>
   388→   388→
   389→   389→        {/* source preview with part rects */}
   390→   390→        <div className="mb-2 flex gap-2">
   391→   391→          <canvas
   392→   392→            ref={sourceCanvasRef}
   393→   393→            className="h-[96px] w-[96px] shrink-0 rounded border border-zinc-700 bg-zinc-950"
   394→   394→            style={{ imageRendering: 'pixelated' }}
   395→   395→            aria-label="Anchor frame with part rectangles"
   396→   396→          />
   397→   397→          <div className="min-w-0 flex-1 space-y-1">
   398→   398→            <p className="font-mono text-[10px] leading-tight text-zinc-500">
   399→   399→              Regions only — never cuts. Overlaps are owned by the highest-z part (R2).
   400→   400→            </p>
   401→   401→            <div className="flex flex-wrap gap-1">
   402→   402→              {draftParts.map((p, i) => (
   403→   403→                <span
   404→   404→                  key={`${p.name}-${i}`}
   405→   405→                  className="inline-flex items-center gap-1 rounded border border-zinc-700 bg-zinc-950 px-1.5 py-0.5 font-mono text-[10px] text-zinc-300"
   406→   406→                >
   407→   407→                  <span className="h-2 w-2 rounded-sm" style={{ background: PART_COLORS[i % PART_COLORS.length] }} />
