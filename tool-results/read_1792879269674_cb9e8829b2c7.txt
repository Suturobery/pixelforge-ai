     1→// PixelForge AI — editor state store (Zustand) — Task 4 core-backed shim.
     2→//
     3→// Architecture (v3.1 §12 Task 4 "Port renderer onto core"):
     4→//   • @pixelforge/core owns the DOCUMENT: model v2 (per-clip layer stacks +
     5→//     per-frame cels, linked cels, trash), the CommandBus (the single
     6→//     mutation path — UI/AI all execute commands), tile-diff history with a
     7→//     memory budget, and the command journal (crash recovery source).
     8→//   • Zustand keeps UI + VIEW state (tool, zoom, pan, panels, playback,
     9→//     active pointers) and mirrors bus-derived values (project reference,
    10→//     dirty, undo/redo availability) so components stay plain subscribers.
    11→//   • Action names from the prototype store are PRESERVED — components call
    12→//     the same API; each action now maps onto a bus command instead of
    13→//     mutating a v1 document by hand.
    14→//
    15→// The bus lives OUTSIDE React. Every bus change event re-syncs the mirrored
    16→// slice; mid-stroke painting bumps `renderTick` only (no bus traffic).

    96→/** Settings `historyLimit` presets map onto v3.1 byte budgets (§2.5). */
    97→function historyBudgetBytes(limit: number): number {
    98→  switch (limit) {
    99→    case 20: return 16 * 1024 * 1024
   100→    case 40: return 32 * 1024 * 1024
   101→    case 100: return 64 * 1024 * 1024
   102→    case 200: return 128 * 1024 * 1024
   103→    default: return 64 * 1024 * 1024
   104→  }
   105→}
   106→bus.setHistoryBudget(historyBudgetBytes(useAppSettings.getState().historyLimit))
   107→useAppSettings.subscribe((s, prev) => {
   108→  if (s.historyLimit !== prev.historyLimit) {
   109→    bus.setHistoryBudget(historyBudgetBytes(s.historyLimit))
   110→  }

   113→// Shim-registered command: palette presets replace the whole palette in ONE
   114→// undoable step (built-ins only add/remove single colors).
   115→bus.register({
   116→  type: 'palette.replaceAll',
   117→  defaultLabel: 'Apply palette',
   118→  validate(payload) {
   119→    const p = payload as { colors?: unknown }
   120→    const colors = p?.colors
   121→    if (!Array.isArray(colors) || colors.length === 0 || colors.length > 256) {
   122→      throw new CommandError('VALIDATION_FAILED', 'palette.replaceAll: colors must be a 1..256 array')
   123→    }
   124→    for (const c of colors) {
   125→      if (typeof c !== 'string' || !/^#[0-9a-f]{6}([0-9a-f]{2})?$/i.test(c)) {
   126→        throw new CommandError('VALIDATION_FAILED', `palette.replaceAll: bad hex "${String(c)}"`)
   127→      }
   128→    }
   129→  },
   130→  apply(payload, ctx) {
   131→    return {
   132→      project: { ...ctx.project, palette: [...(payload as { colors: string[] }).colors] },
   133→      label: 'Apply palette',
   134→    }
   135→  },
   136→})

   142→interface ActiveRefs {

   144→  frameId: string
   145→  frameIndex: number

   149→/** Resolve the active pointers against the LIVE document (safe fallbacks). */
   150→function activeRefs(s: Pick<EditorState, 'activeClipId' | 'activeFrameIndex' | 'activeLayerId'>): ActiveRefs | null {
   151→  const project = bus.getProject()
   152→  const clip = project.clips.find((c) => c.id === s.activeClipId) ?? project.clips[0]
   153→  if (!clip) return null
   154→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   155→  const frame = clip.frames[frameIndex]
   156→  if (!frame) return null
   157→  const layer = clip.layers.find((l) => l.id === s.activeLayerId) ?? clip.layers[clip.layers.length - 1]
   158→  if (!layer) return null
   159→  return { clipId: clip.id, frameId: frame.id, frameIndex, layerId: layer.id }
   160→}
   161→
   162→function findNewLayerId(before: Set<string>, project: Project, clipId: string): string | null {
   163→  const clip = project.clips.find((c) => c.id === clipId)
   164→  if (!clip) return null
   165→  for (let i = clip.layers.length - 1; i >= 0; i--) {
   166→    const l = clip.layers[i]
   167→    if (!before.has(l.id)) return l.id
   168→  }
   169→  return null
   170→}
   171→
   172→function layerIds(project: Project, clipId: string): Set<string> {
   173→  const clip = project.clips.find((c) => c.id === clipId)
   174→  return new Set(clip?.layers.map((l) => l.id) ?? [])
   175→}

   177→/**
   178→ * Stroke-flow helper for one-shot selection edits (cut/delete/paste):
   179→ * begin → mutate the working buffer → end as ONE tile-diff history entry.
   180→ * Any failure cancels the stroke — the document is never half-mutated.
   181→ */
   182→function withStroke(
   183→  label: string,
   184→  refs: ActiveRefs,
   185→  fn: (handle: CoreStrokeHandle) => boolean
   186→): boolean {
   187→  let changed = false
   188→  try {
   189→    const handle = bus.beginStroke({ clipId: refs.clipId, frameId: refs.frameId, layerId: refs.layerId })
   190→    changed = fn(handle)
   191→    bus.endStroke(label, changed, ORIGIN)
   192→    return changed
   193→  } catch {
   194→    if (bus.activeStroke) {
   195→      try {
   196→        bus.cancelStroke(ORIGIN)
   197→      } catch {
   198→        /* ignore */
   199→      }
   200→    }
   201→    return false
   202→  }
   203→}

   205→/**
   206→ * Structural sanity for documents entering the editor (Task 4 §5 data-loss
   207→ * protection): at least one clip, and every clip needs a frame to draw on
   208→ * and a layer to draw with. The core v1 shape guard is deliberately loose
   209→ * (serialization format check); THIS is the editor-entry gate. Degenerate
   210→ * rows/files are refused with a toast instead of limping or crashing.
   211→ */
   212→export function isUsableProject(p: unknown): p is Project {
   213→  if (!p || typeof p !== 'object') return false
   214→  const doc = p as Project
   215→  if (!Array.isArray(doc.clips) || doc.clips.length === 0) return false
   216→  return doc.clips.every(
   217→    (c) =>
   218→      c &&
   219→      Array.isArray(c.layers) &&
   220→      c.layers.length > 0 &&
   221→      Array.isArray(c.frames) &&
   222→      c.frames.length > 0
   223→  )
   224→}

   365→let docVersionCounter = 0
   366→let loadEpochCounter = 0

   368→function syncFromBus(extra: Partial<EditorState> = {}): void {
   369→  docVersionCounter += 1
   370→  const stats = bus.historyStats()
   371→  useEditor.setState({
   372→    project: bus.getProject(),
   373→    docVersion: docVersionCounter,
   374→    renderTick: docVersionCounter,
   375→    dirty: bus.isDirty(),
   376→    canUndo: bus.canUndo(),
   377→    canRedo: bus.canRedo(),
   378→    historyUndoDepth: stats.undoDepth,
   379→    historyRedoDepth: stats.redoDepth,
   380→    historyBytes: stats.bytes,
   381→    ...extra,
   382→  })
   383→  revalidatePointers()
   384→}
   385→
   386→/** Keep the active pointers resolvable after undo/load/structural edits. */
   387→function revalidatePointers(): void {
   388→  const s = useEditor.getState()
   389→  const project = bus.getProject()
   390→  let clip = project.clips.find((c) => c.id === s.activeClipId)
   391→  let patch: Partial<EditorState> | null = null
   392→  if (!clip) {
   393→    clip = project.clips[0]
   394→    if (!clip) return
   395→    patch = { activeClipId: clip.id }
   396→  }
   397→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   398→  const frame = clip.frames[frameIndex]
   399→  const layerExists = clip.layers.some((l) => l.id === s.activeLayerId)
   400→  const nextLayerId = layerExists ? s.activeLayerId : (clip.layers[clip.layers.length - 1]?.id ?? '')
   401→  if (frameIndex !== s.activeFrameIndex || !layerExists || patch) {
   402→    patch = { ...(patch ?? {}), activeFrameIndex: frameIndex, activeLayerId: nextLayerId }
   403→  }
   404→  if (patch && (patch.activeFrameIndex !== s.activeFrameIndex || patch.activeLayerId !== s.activeLayerId || patch.activeClipId !== s.activeClipId)) {
   405→    useEditor.setState(patch)
   406→  }
   407→}

   564→    if (!selection) return

   571→    set({ clipboard: clip })
   572→  },
   573→
   574→  selectionCut: () => {
   575→    const { selection } = get()
   576→    if (!selection) return
   577→    const refs = activeRefs(get())
   578→    if (!refs) return
   579→    const project = bus.getProject()
   580→    let captured: ClipboardContent | null = null
   581→    const ok = withStroke('Cut selection', refs, (handle) => {

   587→      set({ clipboard: captured, selection: null })
   588→    }
   589→  },

   598→      clearRegionInPlace(handle.buffer, project.width, project.height, selection)
