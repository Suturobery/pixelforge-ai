    54→    54→/** Small PNG thumbnail of a normalized buffer (history strip). */
    55→    55→function makeThumb(buffer: Uint8ClampedArray, w: number, h: number): string {
    56→    56→  const canvas = document.createElement('canvas')
    57→    57→  canvas.width = w
    58→    58→  canvas.height = h
    59→    59→  const ctx = canvas.getContext('2d')
    60→    60→  if (!ctx) return ''
    61→    61→  ctx.putImageData(new ImageData(new Uint8ClampedArray(buffer), w, h), 0, 0)
    62→    62→  return canvas.toDataURL('image/png')
    63→    63→}

    65→    65→function fmtCost(usd: number | null): string {
    66→    66→  if (usd === null) return 'cost n/a'
    67→    67→  if (usd === 0) return 'cost $0 (cache)'
    68→    68→  return `≈ $${usd.toFixed(3)} (estimate)`
    69→    69→}
