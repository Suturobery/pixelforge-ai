   184→export const frameAdd: CommandDefinition<{
   185→  clipId: string;
   186→  afterIndex?: number;
   187→  copy?: boolean;
   188→}> = {
   189→  type: 'frame.add',
   190→  defaultLabel: 'Add frame',
   191→  validate(payload, ctx) {
   192→    const p = assertObject(payload, 'frame.add');
   193→    const { clipId } = parseRef(p);
   194→    const clip = findClip(ctx.project, clipId);
   195→    if (!clip) failNotFound(`frame.add: clip "${clipId}" not found`);
   196→    const afterIndex = getOptionalInt(p, 'afterIndex', 'frame.add', -1, clip.frames.length - 1);
   197→    if (getOptionalInt(p, 'afterIndex', 'frame.add') !== undefined && afterIndex === undefined) {
   198→      fail('frame.add: bad afterIndex');
   199→    }
   200→  },
   201→  apply(payload, ctx) {
   202→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   203→    const insertAt =
   204→      payload.afterIndex !== undefined
   205→        ? Math.min(Math.max(0, payload.afterIndex + 1), clip.frames.length)
   206→        : clip.frames.length;
   207→    const srcIndex =
   208→      payload.afterIndex !== undefined
   209→        ? Math.min(Math.max(0, payload.afterIndex), clip.frames.length - 1)
   210→        : clip.frames.length - 1;
   211→    const frame: Frame = payload.copy && clip.frames[srcIndex]
   212→      ? duplicateFrame(clip.frames[srcIndex], clip.layers) // duplicate preserves cels exactly (§2.7)
   213→      : createFrame(ctx.project.width, ctx.project.height, clip.layers);
   214→    const frames = [...clip.frames.slice(0, insertAt), frame, ...clip.frames.slice(insertAt)];
   215→    return {
   216→      project: mapClip(ctx.project, clip.id, (c) => ({ ...c, frames })),
   217→      label: payload.copy ? 'Duplicate frame' : 'Add frame',
   218→    };
   219→  },

   415→/**
   416→ * Alpha compositing of the top cel onto the one below, per frame (ported
   417→ * from the prototype store's mergeDown). The top layer is removed; the
   418→ * operation is undoable via the structural snapshot, so it is not
   419→ * trash-backed (documented deviation from delete semantics).
   420→ */
   421→export const layerMergeDown: CommandDefinition<{ clipId: string; layerId: string }> = {
   422→  type: 'layer.mergeDown',
   423→  defaultLabel: 'Merge down',
   424→  validate(payload, ctx) {
   425→    const p = assertObject(payload, 'layer.mergeDown');
   426→    const { clipId, layerId } = parseLayerRef(p);
   427→    const clip = findClip(ctx.project, clipId);
   428→    if (!clip) failNotFound(`layer.mergeDown: clip "${clipId}" not found`);
   429→    const found = findLayer(clip, layerId);
   430→    if (!found) failNotFound(`layer.mergeDown: layer "${layerId}" not found`);
   431→    if (found.index === 0) fail('layer.mergeDown: nothing below the bottom layer');
   432→    if (found.layer.linked) {
   433→      fail('layer.mergeDown: merging a linked layer is not supported (unlink first — C2)');
   434→    }
   435→  },
   436→  apply(payload, ctx) {
   437→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   438→    const { index } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   439→    const { width, height } = ctx.project;
   440→    const project = mapClip(ctx.project, clip.id, (c) => ({
   441→      ...c,
   442→      layers: c.layers.filter((_, i) => i !== index),
   443→      frames: c.frames.map((f) => {
   444→        const top = f.cels[payload.layerId];
   445→        const bottomLayer = c.layers[index - 1];
   446→        if (!bottomLayer) return f;
   447→        const bottom = f.cels[bottomLayer.id];
   448→        if (!top && !bottom) return f;
   449→        // Missing cel = transparent (C2) — merge treats it as empty.
   450→        const merged = bottom
   451→          ? new Uint8ClampedArray(bottom.buffer)
   452→          : new Uint8ClampedArray(width * height * 4);
   453→        if (top) {
   454→          for (let i = 0; i < width * height; i++) {
   455→            const ta = (top.buffer[i * 4 + 3] / 255) * (top.linked ? 1 : (c.layers[index]?.opacity ?? 1));
   456→            if (ta >= 1) {
   457→              merged[i * 4] = top.buffer[i * 4];
   458→              merged[i * 4 + 1] = top.buffer[i * 4 + 1];
   459→              merged[i * 4 + 2] = top.buffer[i * 4 + 2];
   460→              merged[i * 4 + 3] = top.buffer[i * 4 + 3];
   461→            } else if (ta > 0) {
   462→              const ba = merged[i * 4 + 3] / 255;
   463→              const outA = ta + ba * (1 - ta);
   464→              if (outA === 0) continue;
   465→              for (let ch = 0; ch < 3; ch++) {
   466→                merged[i * 4 + ch] = Math.round(
   467→                  (top.buffer[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   468→                );
   469→              }
   470→              merged[i * 4 + 3] = Math.round(outA * 255);
   471→            }
   472→          }
   473→        }
   474→        return { ...f, cels: { ...f.cels, [bottomLayer.id]: { buffer: merged, linked: false } } };
   475→      }),
   476→    }));
   477→    // Remove the top layer's cels everywhere.
   478→    const cleaned = mapClip(project, clip.id, (c) => ({
   479→      ...c,
   480→      frames: c.frames.map((f) => {
   481→        if (!f.cels[payload.layerId]) return f;
   482→        const cels = { ...f.cels };
   483→        delete cels[payload.layerId];
   484→        return { ...f, cels };
   485→      }),
   486→    }));
   487→    return { project: cleaned, label: 'Merge down' };
   488→  },
   489→};

   529→    const { clipId, layerId } = parseLayerRef(p);
   530→    const clip = findClip(ctx.project, clipId);
   531→    if (!clip) failNotFound(`layer.setProps: clip "${clipId}" not found`);
   532→    if (!findLayer(clip, layerId)) failNotFound(`layer.setProps: layer "${layerId}" not found`);
   533→    const patch = assertObject(p.patch, 'layer.setProps patch');
   534→    const name = getOptionalString(patch, 'name', 'layer.setProps');
   535→    if (name !== undefined && name.length > 120) fail('layer.setProps: name too long');

   537→    if (opacity !== undefined && (typeof opacity !== 'number' || opacity < 0 || opacity > 1)) {
   538→      fail('layer.setProps: opacity must be 0..1');
   539→    }
   540→  },
   541→  apply(payload, ctx) {
   542→    const { clipId, layerId, patch } = payload;
   543→    return {
   544→      project: mapClip(ctx.project, clipId, (c) => ({
   545→        ...c,
   546→        layers: c.layers.map((l) =>
   547→          l.id === layerId
   548→            ? {
   549→                ...l,
   550→                ...(patch.name !== undefined ? { name: patch.name } : {}),

   562→/**
   563→ * Linked-cel engine switch (C2/C19 — P0 engine support; UI toggle is P2).
   564→ * ON  → pick the anchor frame's cel (first frame holding one) and share it
   565→ *        across ALL frames (content of other frames collapses to the anchor).
   566→ * OFF → break sharing: every frame gets an independent deep copy.
   567→ */
   568→export const layerSetLinked: CommandDefinition<{ clipId: string; layerId: string; linked: boolean }> = {
   569→  type: 'layer.setLinked',
   570→  defaultLabel: 'Linked cel',
   571→  validate(payload, ctx) {
   572→    const p = assertObject(payload, 'layer.setLinked');
   573→    const { clipId, layerId } = parseLayerRef(p);
   574→    const clip = findClip(ctx.project, clipId);
   575→    if (!clip) failNotFound(`layer.setLinked: clip "${clipId}" not found`);
   576→    if (!findLayer(clip, layerId)) failNotFound(`layer.setLinked: layer "${layerId}" not found`);
   577→    if (typeof p.linked !== 'boolean') fail('layer.setLinked: "linked" must be a boolean');
   578→  },
   579→  apply(payload, ctx) {
   580→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   581→    const { layer } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   582→    if (payload.linked === layer.linked) {
   583→      return { project: ctx.project, label: 'Linked cel' };
   584→    }
   585→    if (payload.linked) {
   586→      // Anchor = first frame that has a cel (else first frame → create).
   587→      const anchorFrame = clip.frames.find((f) => f.cels[layer.id]) ?? clip.frames[0];
   588→      if (!anchorFrame) fail('layer.setLinked: clip has no frames');
   589→      const existing = anchorFrame.cels[layer.id];
   590→      const shared: Cel = existing
   591→        ? { buffer: existing.buffer, linked: true }
   592→        : { buffer: new Uint8ClampedArray(ctx.project.width * ctx.project.height * 4), linked: true };
   593→      const withFlag = mapClip(ctx.project, clip.id, (c) => ({
   594→        ...c,
   595→        layers: c.layers.map((l) => (l.id === layer.id ? { ...l, linked: true } : l)),
   596→      }));
   597→      return {
   598→        project: setCelEverywhere(withFlag, clip.id, layer.id, shared),
   599→        label: 'Link cel (shared)',
   600→      };
   601→    }
   602→    // Unlink: give every frame its own copy of the current shared buffer.
   603→    const project = mapClip(ctx.project, clip.id, (c) => ({
   604→      ...c,
   605→      layers: c.layers.map((l) => (l.id === layer.id ? { ...l, linked: false } : l)),
   606→    }));
   607→    const withCopies = mapClip(project, clip.id, (c) => ({
   608→      ...c,
   609→      frames: c.frames.map((f) => {
   610→        const cel = f.cels[layer.id];
   611→        if (!cel) return f;
   612→        return { ...f, cels: { ...f.cels, [layer.id]: { buffer: cloneBuffer(cel.buffer), linked: false } } };
   613→      }),
   614→    }));
   615→    return { project: withCopies, label: 'Unlink cel' };
   616→  },
   617→};

   665→export const trashRestore: CommandDefinition<{ entryId: string }> = {
   666→  type: 'trash.restore',
   667→  defaultLabel: 'Restore from trash',
   668→  validate(payload, ctx) {
   669→    const p = assertObject(payload, 'trash.restore');
   670→    const entryId = getString(p, 'entryId', 'trash.restore');
   671→    if (!ctx.project.trash.some((t) => t.id === entryId)) {
   672→      failNotFound(`trash.restore: entry "${entryId}" not found`);
   673→    }
   674→  },
   675→  apply(payload, ctx) {
   676→    const taken = takeTrash(ctx.project, payload.entryId);
   677→    if (!taken) failNotFound('trash.restore: entry vanished');
   678→    const { entry, project: without } = taken;
   679→    const item = entry.item;
   680→    if (item.kind === 'frame') {
   681→      const clip = findClip(without, item.clipId);
   682→      if (!clip) failNotFound('trash.restore: owning clip no longer exists');
   683→      const index = Math.min(Math.max(0, item.frameIndex), clip.frames.length);
   684→      return {
   685→        project: mapClip(without, clip.id, (c) => ({
   686→          ...c,
   687→          frames: [...c.frames.slice(0, index), item.frame, ...c.frames.slice(index)],
   688→        })),
   689→        label: 'Restore frame',
   690→      };
   691→    }
   692→    if (item.kind === 'layer') {
   693→      const clip = findClip(without, item.clipId);
   694→      if (!clip) failNotFound('trash.restore: owning clip no longer exists');
   695→      const index = Math.min(Math.max(0, item.layerIndex), clip.layers.length);
   696→      const layers = [...clip.layers.slice(0, index), item.layer, ...clip.layers.slice(index)];
   697→      const restored = mapClip(without, clip.id, (c) => ({ ...c, layers }));
   698→      // Restore cels to the frames they came from (missing frame ids skipped).
   699→      const withCels = mapClip(restored, clip.id, (c) => ({
   700→        ...c,
   701→        frames: c.frames.map((f) => {
   702→          const cel = item.celsByFrameId[f.id];
   703→          if (!cel) return f;
   704→          return { ...f, cels: { ...f.cels, [item.layer.id]: cel } };
   705→        }),
   706→      }));
   707→      return { project: withCels, label: 'Restore layer' };
   708→    }
   709→    const index = Math.min(Math.max(0, item.clipIndex), without.clips.length);
   710→    const clips = [...without.clips.slice(0, index), item.clip, ...without.clips.slice(index)];
   711→    return { project: { ...without, clips }, label: 'Restore clip' };
   712→  },
   713→};
