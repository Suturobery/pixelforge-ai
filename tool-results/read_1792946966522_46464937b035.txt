     1→// PixelForge AI — Motion Composer panel (Task 10, v3.1 §4.1).
     2→//
     3→// Deterministic part-based animation workflow:
     4→//   Select source clip + anchor frame → Tag parts (marquee or manual rect,
     5→//   regions never cuts) → Pick preset (Idle/Walk/Run/Attack/Jump) →
     6→//   Adjust parameters → Preview (pure plan, NEVER committed until Accept) →
     7→//   QA warnings → Accept (one undoable bus command) → edit frames as usual.
     8→//
     9→// The preview is the same pure composer the bus command runs — what you
    10→// accept is byte-identical to what you previewed. Nothing here mutates the
    11→// document except the two CommandBus actions (part.set / motion.apply).
    12→

    53→/** Canonical role names offered for tagging (plan §4.1); custom names allowed. */
    54→const CANONICAL_ROLES = ['head', 'body', 'armL', 'armR', 'legL', 'legR', 'hair'] as const

    56→const PART_COLORS = ['#2ce8f5', '#a7f070', '#ffcd75', '#ef7d57', '#f4f4f4', '#73eff7', '#b13e53', '#38b764']

    58→interface DraftPart {
    59→  name: string
    60→  layerId: string
    61→  rect?: PartRect

    64→/** Composited display buffer for one preview frame (plan + static layers). */
    65→function displayBuffer(
    66→  project: Parameters<typeof composeFrameRgba>[0],
    67→  clip: Parameters<typeof composeFrameRgba>[1],
    68→  anchorFrame: Parameters<typeof composeFrameRgba>[2],
    69→  plan: MotionPlan,
    70→  frameIndex: number
    71→): Uint8ClampedArray {
    72→  const cels = { ...anchorFrame.cels }
    73→  for (const [layerId, buf] of Object.entries(plan.frames[frameIndex].cels)) {
    74→    cels[layerId] = { buffer: buf, linked: false }
    75→  }
    76→  return composeFrameRgba(project, clip, { id: anchorFrame.id, durationMs: 0, cels })
    77→}

    79→export function MotionPanel() {
    80→  const project = useEditor((s) => s.project)
    81→  const activeClipId = useEditor((s) => s.activeClipId)
    82→  const selection = useEditor((s) => s.selection)
    83→  const setRightTab = useEditor((s) => s.setRightTab)
    84→  const setClipParts = useEditor((s) => s.setClipParts)
    85→  const applyMotionPreset = useEditor((s) => s.applyMotionPreset)
    86→  const docVersion = useEditor((s) => s.docVersion)
    87→  const { toast } = useToast()
    88→
    89→  const clip = getActiveClip({ project, activeClipId })
    90→  const layerId = clip.layers.at(-1)?.id ?? ''
    91→
    92→  // --- source / parts state ------------------------------------------------
    93→  const [anchorIndex, setAnchorIndex] = useState(0)
    94→  const [draftName, setDraftName] = useState('body')
    95→  const [draftRect, setDraftRect] = useState<PartRect>({ x: 8, y: 8, w: 16, h: 16 })
    96→  const [draftWholeLayer, setDraftWholeLayer] = useState(false)
    97→  const [dirtyDraft, setDirtyDraft] = useState(false)
    98→
    99→  // --- preview state (declared before the sync effect reads setPlan) --------
   100→  const [plan, setPlan] = useState<MotionPlan | null>(null)
   101→  const [issues, setIssues] = useState<QAIssue[]>([])
   102→  const [playing, setPlaying] = useState(false)
   103→  const [previewIdx, setPreviewIdx] = useState(0)
   104→  const [showOverlay, setShowOverlay] = useState(true)
   105→  const [targetMode, setTargetMode] = useState<'new' | 'overwrite'>('new')
   106→  const [newClipName, setNewClipName] = useState('idle-1')
   107→
   108→  // draft parts start as a copy of the committed clip parts; draft state is
   109→  // reset whenever the clip or the document changes (render-time adjustment —
   110→  // the React-endorsed pattern for derived-state resets, no sync effect)
   111→  const [draftParts, setDraftParts] = useState<DraftPart[]>([])
   112→  const syncKey = `${clip.id}:${docVersion}`
   113→  const [lastSyncKey, setLastSyncKey] = useState(syncKey)
   114→  if (lastSyncKey !== syncKey) {
   115→    setLastSyncKey(syncKey)
   116→    setDraftParts(clip.parts.map((p) => ({ name: p.name, layerId: p.layerId, ...(p.rect ? { rect: { ...p.rect } } : {}) })))
   117→    setDirtyDraft(false)
   118→    setAnchorIndex((i) => Math.min(i, clip.frames.length - 1))
   119→    setPlan(null)
   120→  }
   121→
   122→  // --- preset state ----------------------------------------------------------
   123→  const [presetId, setPresetId] = useState<MotionPresetId>('idle')
   124→  const preset: PresetDef = MOTION_PRESETS[presetId]
   125→  const [params, setParams] = useState<MotionParams>({ ...MOTION_PARAMS_DEFAULTS, frameCount: 4 })
   126→  // switching presets snaps frameCount back to the new preset's default
   127→  const [lastPresetId, setLastPresetId] = useState(presetId)
   128→  if (lastPresetId !== presetId) {
   129→    setLastPresetId(presetId)
   130→    setParams((p) => ({ ...p, frameCount: preset.defaultFrameCount }))
   131→  }
   132→
   133→  const sourceCanvasRef = useRef<HTMLCanvasElement | null>(null)
   134→  const previewCanvasRef = useRef<HTMLCanvasElement | null>(null)
   135→  const stripRefs = useRef<Array<HTMLCanvasElement | null>>([])
   136→
   137→  const anchorFrame = clip.frames[Math.min(anchorIndex, clip.frames.length - 1)]
   138→
   139→  // --- preview rendering -----------------------------------------------------
   140→  // plain function — the React Compiler memoizes; no hand-rolled dep arrays
   141→  const renderPlan = () => {
   142→    if (!anchorFrame) return
   143→    let p: MotionPlan
   144→    try {
   145→      p = planMotion(project, clip, { preset: presetId, params, anchorFrameIndex: anchorIndex })
   146→    } catch (err) {
   147→      toast({ title: 'Preview failed', description: err instanceof Error ? err.message : 'Composer error', variant: 'destructive' })
   148→      return
   149→    }
   150→    setPlan(p)
   151→    setIssues(analyzeMotionPlan(project, clip, p))
   152→    setPreviewIdx(0)
   153→    setPlaying(false)
   154→    // the suggested clip name follows the preset; the user can still edit it
   155→    // before accepting (every fresh preview resets it)
   156→    setNewClipName(`${presetId}-${project.clips.length}`)
   157→  }
   158→
   159→  // draw the source canvas (anchor composite + part rects overlay)
   160→  useEffect(() => {
   161→    const canvas = sourceCanvasRef.current
   162→    if (!canvas || !anchorFrame) return
   163→    canvas.width = project.width
   164→    canvas.height = project.height
   165→    const ctx = canvas.getContext('2d')
   166→    if (!ctx) return
   167→    const rgba = composeFrameRgba(project, clip, anchorFrame)
   168→    ctx.putImageData(new ImageData(new Uint8ClampedArray(rgba), project.width, project.height), 0, 0)
   169→    if (!showOverlay) return
   170→    // part rects (1px, color per part) + ground line + center axis
   171→    draftParts.forEach((p, i) => {
   172→      const r = p.rect ?? { x: 0, y: 0, w: project.width, h: project.height }
   173→      ctx.strokeStyle = PART_COLORS[i % PART_COLORS.length]
   174→      ctx.lineWidth = 1
   175→      ctx.strokeRect(r.x + 0.5, r.y + 0.5, r.w - 1, r.h - 1)
   176→    })
   177→    ctx.strokeStyle = 'rgba(45, 212, 191, 0.55)'
   178→    ctx.beginPath()
   179→    ctx.moveTo(0, project.height - 0.5)
   180→    ctx.lineTo(project.width, project.height - 0.5)
   181→    ctx.moveTo(project.width / 2 + 0.5, 0)
   182→    ctx.lineTo(project.width / 2 + 0.5, project.height)
   183→    ctx.stroke()
   184→  }, [project, clip, anchorFrame, draftParts, showOverlay, docVersion])
   185→
   186→  // draw the current preview frame + the strip thumbnails
   187→  useEffect(() => {
   188→    if (!plan || !previewCanvasRef.current) return
   189→    const canvas = previewCanvasRef.current
   190→    canvas.width = project.width
   191→    canvas.height = project.height
   192→    const ctx = canvas.getContext('2d')
   193→    if (!ctx) return
   194→    const rgba = displayBuffer(project, clip, anchorFrame, plan, previewIdx)
   195→    ctx.putImageData(new ImageData(new Uint8ClampedArray(rgba), project.width, project.height), 0, 0)
   196→    // preview frame part rects at their OFFSET positions (visual debugging)
   197→    if (showOverlay) {
   198→      ctx.strokeStyle = 'rgba(45, 212, 191, 0.55)'
   199→      ctx.beginPath()
   200→      ctx.moveTo(0, project.height - 0.5)
   201→      ctx.lineTo(project.width, project.height - 0.5)
   202→      ctx.stroke()
   203→    }
   204→    plan.frames.forEach((f, i) => {
   205→      const c = stripRefs.current[i]
   206→      if (!c) return
   207→      c.width = project.width
   208→      c.height = project.height
   209→      const cctx = c.getContext('2d')
   210→      if (!cctx) return
   211→      const r = displayBuffer(project, clip, anchorFrame, plan, i)
   212→      cctx.putImageData(new ImageData(new Uint8ClampedArray(r), project.width, project.height), 0, 0)
   213→    })
   214→  }, [plan, previewIdx, showOverlay, docVersion])
   215→
   216→  // playback timer (clip fps; play-once presets stop on the last frame)
   217→  useEffect(() => {
   218→    if (!playing || !plan) return
   219→    const ms = Math.max(30, Math.round(1000 / clip.fps))
   220→    const t = setInterval(() => {
   221→      setPreviewIdx((i) => {
   222→        const next = i + 1
   223→        if (next >= plan.frames.length) {
   224→          if (preset.playOnceSuggested) {
   225→            setPlaying(false)
   226→            return i
   227→          }
   228→          return 0
   229→        }
   230→        return next
   231→      })
   232→    }, ms)
   233→    return () => clearInterval(t)
   234→  }, [playing, plan, clip.fps, preset])
   235→
   236→  // --- actions -----------------------------------------------------------------
   237→  const addDraftPart = () => {
   238→    const name = draftName.trim()
   239→    if (!name) return
   240→    if (draftParts.some((p) => roleKey(p.name) === roleKey(name))) {
   241→      toast({ title: 'Duplicate part name', description: `A part named "${name}" already exists — names are roles.`, variant: 'destructive' })
   242→      return
   243→    }
   244→    if (draftParts.length >= 32) {
   245→      toast({ title: 'Too many parts', description: 'At most 32 parts per clip.', variant: 'destructive' })
   246→      return
   247→    }
   248→    setDraftParts((ps) => [...ps, { name, layerId, ...(draftWholeLayer ? {} : { rect: { ...draftRect } }) }])
   249→    setDirtyDraft(true)
   250→  }
   251→
   252→  const useSelectionAsRect = () => {
   253→    if (!selection) {
   254→      toast({ title: 'No selection', description: 'Drag a marquee on the canvas first, then press "Use selection".', variant: 'destructive' })
   255→      return
   256→    }
   257→    setDraftRect({ ...selection })
   258→    setDraftWholeLayer(false)
   259→  }
   260→
   261→  const saveParts = () => {
   262→    if (!setClipParts(clip.id, draftParts.map((p) => ({ ...p })))) return
   263→    setDirtyDraft(false)
   264→    toast({ title: 'Parts saved', description: `${draftParts.length} part${draftParts.length === 1 ? '' : 's'} tagged on "${clip.name}".` })
   265→  }
   266→
   267→  const accept = () => {
   268→    if (!plan) return
   269→    const target =
   270→      targetMode === 'new'
   271→        ? ({ kind: 'new', name: newClipName.trim() || `${presetId}-clip`, fps: clip.fps } as const)
   272→        : ({ kind: 'existing', clipId: clip.id, overwrite: true } as const)
   273→    const createdClipId = applyMotionPreset({
   274→      sourceClipId: clip.id,
   275→      preset: presetId,
   276→      params: { amplitude: params.amplitude, frameCount: plan.report.frameCount, gapFill: params.gapFill, smear: params.smear },
   277→      anchorFrameIndex: anchorIndex,
   278→      target,
   279→    })
   280→    if (!createdClipId) return
   281→    const skipped = plan.report.skipped.length
   282→    toast({
   283→      title: 'Motion applied',
   284→      description:
   285→        targetMode === 'new'
   286→          ? `New clip "${target.kind === 'new' ? target.name : ''}" created with ${plan.report.frameCount} frames.${skipped ? ` ${skipped} absent role(s) skipped.` : ''}`
   287→          : `${plan.report.frameCount} frame${plan.report.frameCount === 1 ? '' : 's'} of "${clip.name}" rewritten (undoable).`,
   288→    })
   289→    setPlan(null)
   290→    setIssues([])
   291→    setPlaying(false)
   292→    if (targetMode === 'new' && createdClipId) {
   293→      const st = useEditor.getState()
   294→      const created = st.project.clips.find((c) => c.id === createdClipId)
   295→      useEditor.setState({
   296→        activeClipId: createdClipId,
   297→        activeFrameIndex: 0,
   298→        activeLayerId: created?.layers.at(-1)?.id ?? st.activeLayerId,
   299→      })
   300→    }
   301→  }
   302→
   303→  // --- dev probe for browser smokes (mirrors window.__pfAi discipline) ----------
   304→  useEffect(() => {
   305→    const w = window as unknown as Record<string, unknown>
   306→    w.__pfMotion = {
   307→      draftParts: () => draftParts.length,
   308→      savedParts: () => clip.parts.map((p) => ({ name: p.name, rect: p.rect ?? null })),
   309→      tagPart: (name: string, rect?: PartRect) =>
   310→        setDraftParts((ps) => [...ps, { name, layerId, ...(rect ? { rect } : {}) }]),
   311→      saveParts,
   312→      selectPreset: (id: string) => {
   313→        if ((PRESET_IDS as string[]).includes(id)) setPresetId(id as MotionPresetId)
   314→      },
   315→      setParam: (k: 'amplitude' | 'frameCount' | 'gapFill' | 'smear', v: number | string | boolean) =>
   316→        setParams((p) => ({ ...p, [k]: v } as MotionParams)),
   317→      preview: renderPlan,
   318→      report: () => plan?.report ?? null,
   319→      issues: () => issues,
   320→      setTarget: (m: 'new' | 'overwrite') => setTargetMode(m),
   321→      accept,
   322→    }
   323→    return () => {
   324→      delete w.__pfMotion
   325→    }
   326→  })
   327→
   328→  // --------------------------------------------------------------------------------
   329→  if (!anchorFrame) {
   330→    return <div className="p-4 text-xs text-zinc-500">This clip has no frames.</div>
   331→  }
   332→
   333→  return (
   334→    <div className="pixelforge-scroll mx-auto min-h-0 w-full max-w-[520px] flex-1 space-y-3 overflow-y-auto p-3" data-testid="motion-panel">
   335→      {/* ---------------------------- source ---------------------------- */}
   336→      <section aria-label="Source" className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-3">
   337→        <header className="mb-2 flex items-center gap-1.5 font-mono text-[10px] uppercase tracking-wider text-teal-300">
   338→          <Film className="h-3 w-3" /> Source
   339→        </header>
   340→        <div className="flex items-center justify-between gap-2">
   341→          <span className="truncate text-xs font-medium text-zinc-200">{clip.name}</span>
   342→          <div className="flex items-center gap-1">
   343→            <span className="font-mono text-[10px] text-zinc-500">anchor</span>
   344→            <Select value={String(anchorIndex)} onValueChange={(v) => setAnchorIndex(Number(v))}>
   345→              <SelectTrigger className="h-6 w-[70px] font-mono text-[11px]" aria-label="Anchor frame">
   346→                <SelectValue />
   347→              </SelectTrigger>
   348→              <SelectContent className="max-h-48">
   349→                {clip.frames.map((f, i) => (
   350→                  <SelectItem key={f.id} value={String(i)} className="font-mono text-[11px]">
   351→                    {i + 1}
   352→                  </SelectItem>
   353→                ))}
   354→              </SelectContent>
   355→            </Select>
   356→          </div>
   357→        </div>
   358→        <p className="mt-1 font-mono text-[10px] text-zinc-500">
   359→          every frame derives from frame {anchorIndex + 1} (R5) · {clip.fps} fps
   360→        </p>
   361→      </section>
   362→
   363→      {/* ---------------------------- parts ---------------------------- */}
   364→      <section aria-label="Body parts" className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-3">
   365→        <header className="mb-2 flex items-center justify-between">
   366→          <span className="flex items-center gap-1.5 font-mono text-[10px] uppercase tracking-wider text-teal-300">
   367→            <Tag className="h-3 w-3" /> Parts ({draftParts.length})
   368→          </span>
   369→          {dirtyDraft && (
   370→            <div className="flex gap-1">
   371→              <Button size="sm" className="h-6 bg-teal-600 px-2 font-mono text-[10px] hover:bg-teal-500" onClick={saveParts}>
   372→                Save parts
   373→              </Button>
   374→              <Button
   375→                size="sm"
   376→                variant="ghost"
   377→                className="h-6 px-2 font-mono text-[10px] text-zinc-400"
   378→                onClick={() => {
   379→                  setDraftParts(clip.parts.map((p) => ({ name: p.name, layerId: p.layerId, ...(p.rect ? { rect: { ...p.rect } } : {}) })))
   380→                  setDirtyDraft(false)
   381→                }}
   382→              >
   383→                Revert
   384→              </Button>
   385→            </div>
   386→          )}
   387→        </header>
   388→
   389→        {/* source preview with part rects */}
   390→        <div className="mb-2 flex gap-2">
   391→          <canvas
   392→            ref={sourceCanvasRef}
   393→            className="h-[96px] w-[96px] shrink-0 rounded border border-zinc-700 bg-zinc-950"
   394→            style={{ imageRendering: 'pixelated' }}
   395→            aria-label="Anchor frame with part rectangles"
   396→          />
   397→          <div className="min-w-0 flex-1 space-y-1">
   398→            <p className="font-mono text-[10px] leading-tight text-zinc-500">
   399→              Regions only — never cuts. Overlaps are owned by the highest-z part (R2).
   400→            </p>
   401→            <div className="flex flex-wrap gap-1">
   402→              {draftParts.map((p, i) => (
   403→                <span
   404→                  key={`${p.name}-${i}`}
   405→                  className="inline-flex items-center gap-1 rounded border border-zinc-700 bg-zinc-950 px-1.5 py-0.5 font-mono text-[10px] text-zinc-300"
   406→                >
   407→                  <span className="h-2 w-2 rounded-sm" style={{ background: PART_COLORS[i % PART_COLORS.length] }} />
   408→                  {p.name}
   409→                  <button
   410→                    className="text-zinc-500 hover:text-red-400"
   411→                    aria-label={`Remove part ${p.name}`}
   412→                    onClick={() => {
   413→                      setDraftParts((ps) => ps.filter((_, j) => j !== i))
   414→                      setDirtyDraft(true)
   415→                    }}
   416→                  >
   417→                    <Trash2 className="h-3 w-3" />
   418→                  </button>
   419→                </span>
   420→              ))}
   421→              {draftParts.length === 0 && (
   422→                <span className="font-mono text-[10px] text-zinc-600">no parts tagged yet</span>
   423→              )}
   424→            </div>
   425→          </div>
   426→        </div>
   427→
   428→        {/* tag controls */}
   429→        <div className="space-y-2 rounded border border-zinc-800 bg-zinc-950/60 p-2">
   430→          <div className="grid grid-cols-2 gap-2">
   431→            <div>
   432→              <Label className="font-mono text-[10px] text-zinc-400">Part name</Label>
   433→              <Select value={draftName} onValueChange={setDraftName}>
   434→                <SelectTrigger className="h-7 font-mono text-[11px]" aria-label="Part name">
   435→                  <SelectValue />
   436→                </SelectTrigger>
   437→                <SelectContent className="max-h-48">
   438→                  {CANONICAL_ROLES.map((r) => (
   439→                    <SelectItem key={r} value={r} className="font-mono text-[11px]">
   440→                      {r}
   441→                    </SelectItem>
   442→                  ))}
   443→                </SelectContent>
   444→              </Select>
   445→            </div>
   446→            <div>
   447→              <Label className="font-mono text-[10px] text-zinc-400">Region</Label>
   448→              <div className="flex gap-1">
   449→                <Input
   450→                  className="h-7 font-mono text-[11px]"
   451→                  aria-label="Rect x"
   452→                  type="number"
   453→                  min={0}
   454→                  max={project.width - 1}
   455→                  value={draftRect.x}
   456→                  disabled={draftWholeLayer}
   457→                  onChange={(e) => setDraftRect((r) => ({ ...r, x: Number(e.target.value) }))}
   458→                />
   459→                <Input
   460→                  className="h-7 font-mono text-[11px]"
   461→                  aria-label="Rect y"
   462→                  type="number"
   463→                  min={0}
   464→                  max={project.height - 1}
   465→                  value={draftRect.y}
   466→                  disabled={draftWholeLayer}
   467→                  onChange={(e) => setDraftRect((r) => ({ ...r, y: Number(e.target.value) }))}
   468→                />
   469→              </div>
   470→            </div>
   471→          </div>
   472→          <div className="grid grid-cols-2 gap-2">
   473→            <div>
   474→              <Label className="font-mono text-[10px] text-zinc-400">Width</Label>
   475→              <Input
   476→                className="h-7 font-mono text-[11px]"
   477→                aria-label="Rect width"
   478→                type="number"
   479→                min={1}
   480→                max={project.width}
   481→                value={draftRect.w}
   482→                disabled={draftWholeLayer}
   483→                onChange={(e) => setDraftRect((r) => ({ ...r, w: Number(e.target.value) }))}
   484→              />
   485→            </div>
   486→            <div>
   487→              <Label className="font-mono text-[10px] text-zinc-400">Height</Label>
   488→              <Input
   489→                className="h-7 font-mono text-[11px]"
   490→                aria-label="Rect height"
