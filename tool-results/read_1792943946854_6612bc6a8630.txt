   179→   179→/**
   180→   180→ * Secure key editor (Task 13a, desktop only). The plaintext is submitted
   181→   181→ * ONCE to the Electron main process (safeStorage → settings file) and kept
   182→   182→ * as a session-only runtime copy for the interim generation path. The
   183→   183→ * renderer can never read a stored key back — only hasKey booleans.
   184→   184→ */
   185→   185→function SecureKeyEditor({
   186→   186→  provider,
   187→   187→  hasKey,
   188→   188→  sessionOnly,
   189→   189→  onSaved,
   190→   190→}: {
   191→   191→  provider: AIProviderConfig
   192→   192→  hasKey: boolean
   193→   193→  sessionOnly: boolean
   194→   194→  onSaved: (ok: boolean, message: string) => void
   195→   195→}) {
   196→   196→  const [editing, setEditing] = useState(false)
   197→   197→  const [draft, setDraft] = useState('')
   198→   198→  const [saving, setSaving] = useState(false)
   199→   199→  const hasSessionKey = provider.apiKey.trim() !== ''
   200→   200→
   201→   201→  const save = async () => {
   202→   202→    if (!draft.trim()) return
   203→   203→    setSaving(true)
   204→   204→    const res = await saveApiKeySecure(provider.id, draft.trim())
   205→   205→    setSaving(false)
   206→   206→    if (res.ok) {
   207→   207→      setDraft('')
   208→   208→      setEditing(false)
   209→   209→      onSaved(true, res.sessionOnly ? 'Key saved — session only (OS keyring unavailable)' : 'Key saved — encrypted via the OS keyring')
   210→   210→    } else {
   211→   211→      onSaved(false, res.error ?? 'Could not store the key')
   212→   212→    }
   213→   213→  }
   214→   214→
   215→   215→  const remove = async () => {
   216→   216→    const removed = await removeApiKeySecure(provider.id)
   217→   217→    onSaved(removed, removed ? 'Stored key removed' : 'Nothing to remove')
   218→   218→  }
   219→   219→
   220→   220→  return (
   221→   221→    <div className="mt-1.5 space-y-1">
   222→   222→      {/* status row — never shows key material */}
   223→   223→      <div className="flex items-center justify-between gap-2">
   224→   224→        <span className="flex min-w-0 items-center gap-1.5 font-mono text-[10px] text-zinc-400">
   225→   225→          <KeyRound className={`h-3 w-3 shrink-0 ${hasKey ? 'text-teal-400' : 'text-zinc-600'}`} aria-hidden />
   226→   226→          {hasKey ? (
   227→   227→            <>
   228→   228→              <span className="text-zinc-300">●●●●●●●●</span>
   229→   229→              <span className="truncate">{sessionOnly ? 'stored — session only' : 'stored — encrypted'}</span>
   230→   230→            </>
   231→   231→          ) : (
   232→   232→            'no key stored'
   233→   233→          )}
   234→   234→          {!hasKey && hasSessionKey && (
   235→   235→            <span className="shrink-0 rounded bg-amber-500/10 px-1 font-mono text-[9px] text-amber-300">entered — unsaved</span>
   236→   236→          )}
   237→   237→          {hasKey && !hasSessionKey && (
   238→   238→            <span className="shrink-0 rounded bg-zinc-800 px-1 font-mono text-[9px] text-zinc-300">locked until re-entered</span>
   239→   239→          )}
   240→   240→        </span>
   241→   241→        <span className="flex shrink-0 items-center gap-1">
   242→   242→          <Button
   243→   243→            variant="ghost"
   244→   244→            size="sm"
   245→   245→            className="h-5 px-1.5 font-mono text-[10px] text-zinc-300 hover:text-teal-300"
   246→   246→            onClick={() => {
   247→   247→              setEditing((v) => !v)
   248→   248→              setDraft('')
   249→   249→            }}
   250→   250→            aria-label={hasKey ? `Replace ${provider.label} API key` : `Set ${provider.label} API key`}
   251→   251→          >
   252→   252→            {hasKey ? 'Replace' : 'Set key'}
   253→   253→          </Button>
   254→   254→          {hasKey && (
   255→   255→            <Button
   256→   256→              variant="ghost"
   257→   257→              size="sm"
   258→   258→              className="h-5 px-1.5 font-mono text-[10px] text-zinc-400 hover:text-red-300"
   259→   259→              onClick={remove}
   260→   260→              aria-label={`Remove ${provider.label} API key`}
   261→   261→            >
   262→   262→              <Trash2 className="h-3 w-3" />
   263→   263→            </Button>
   264→   264→          )}
   265→   265→        </span>
   266→   266→      </div>
   267→   267→      {editing && (
   268→   268→        <div className="flex items-center gap-1">
   269→   269→          <Input
   270→   270→            type="password"
   271→   271→            value={draft}
   272→   272→            onChange={(e) => setDraft(e.target.value)}
   273→   273→            placeholder="paste the API key — it is sent to the OS keyring and never stored in plain text"
   274→   274→            className="h-6 border-zinc-800 bg-zinc-950 px-2 font-mono text-[11px]"
   275→   275→            aria-label={`${provider.label} new API key`}
   276→   276→            spellCheck={false}
   277→   277→            onKeyDown={(e) => {
   278→   278→              if (e.key === 'Enter') void save()
   279→   279→              if (e.key === 'Escape') setEditing(false)
   280→   280→            }}
   281→   281→          />
   282→   282→          <Button
   283→   283→            variant="outline"
   284→   284→            size="sm"
   285→   285→            className="h-6 border-zinc-800 px-1.5 font-mono text-[10px] text-teal-300"
   286→   286→            disabled={saving || !draft.trim()}
   287→   287→            onClick={() => void save()}
   288→   288→            aria-label="Save API key"
   289→   289→          >
   290→   290→            {saving ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />}
   291→   291→          </Button>
   292→   292→          <Button
   293→   293→            variant="ghost"
   294→   294→            size="sm"
   295→   295→            className="h-6 px-1.5 font-mono text-[10px] text-zinc-400"
   296→   296→            onClick={() => setEditing(false)}
   297→   297→            aria-label="Cancel API key edit"
   298→   298→          >
   299→   299→            <X className="h-3 w-3" />
   300→   300→          </Button>
   301→   301→        </div>
   302→   302→      )}
   303→   303→    </div>
   304→   304→  )
   305→   305→}

   307→   307→function ProviderCard({
   308→   308→  provider,
   309→   309→  isActive,
   310→   310→  wide,
   311→   311→  secureMode,
   312→   312→  onUpdate,
   313→   313→  onSetActive,

   315→   315→  provider: AIProviderConfig
   316→   316→  isActive: boolean
   317→   317→  wide?: boolean
   318→   318→  secureMode: boolean
   319→   319→  onUpdate: (patch: Partial<AIProviderConfig>) => void
   320→   320→  onSetActive: () => void

   323→   323→  const [testing, setTesting] = useState(false)
   324→   324→  const [test, setTest] = useState<TestResult | null>(null)
   325→   325→
   326→   326→  const isBuiltin = provider.kind === 'zai'
   327→   327→  const isAnthropic = provider.kind === 'anthropic'

   331→   331→  // but-locked key cannot drive the interim generation path (Task 9 moves

   338→   338→    setTesting(true)
   339→   339→    setTest(null)
   340→   340→    try {
   341→   341→      const res = await fetch('/api/ai/test', {
   342→   342→        method: 'POST',
   343→   343→        headers: { 'Content-Type': 'application/json', ...AGENT_FETCH_HEADERS },

   345→   345→          kind: provider.kind,

   353→   353→        ok: json.ok,
   354→   354→        text: json.ok
   355→   355→          ? `${json.note ?? 'Connection OK'} · ${json.latencyMs ?? 0} ms`
   356→   356→          : describeApiErrorInline(json.error, 'Connection failed'),
   357→   357→      })
   358→   358→    } catch (err) {
   359→   359→      setTest({ ok: false, text: err instanceof Error ? err.message : 'Test failed' })
   360→   360→    } finally {
   361→   361→      setTesting(false)
   362→   362→    }
   363→   363→  }
   364→   364→
   365→   365→  return (

   369→   369→        <div className="flex min-w-0 items-center gap-1.5">
   370→   370→          <span
   371→   371→            className={`inline-block h-1.5 w-1.5 shrink-0 rounded-full ${
   372→   372→              provider.enabled ? 'bg-teal-400' : 'bg-zinc-600'
   373→   373→            }`}
   374→   374→            aria-hidden
   375→   375→          />
   376→   376→          <span className="truncate font-mono text-xs text-zinc-100">{provider.label}</span>
   377→   377→          <span className="shrink-0 rounded border border-zinc-800 bg-zinc-950 px-1 py-px font-mono text-[9px] uppercase tracking-wider text-zinc-400">
   378→   378→            {provider.kind === 'custom' ? 'openai-compat' : provider.kind}
   379→   379→          </span>
   380→   380→        </div>
   381→   381→        <div className="flex shrink-0 items-center gap-2">
   382→   382→          {isActive ? (
   383→   383→            <span className="rounded bg-teal-600/20 px-1.5 py-0.5 font-mono text-[9px] uppercase tracking-wider text-teal-300">
   384→   384→              engine

   387→   387→            <Button
   388→   388→              variant="ghost"
   389→   389→              size="sm"

   394→   394→                isAnthropic

   397→   397→                    ? 'Enter the key for this session to use this provider'
   398→   398→                    : undefined

   400→   400→            >
   401→   401→              Use

   427→   427→                spellCheck={false}
   428→   428→              />
   429→   429→            </label>
   430→   430→            {secureMode ? (
   431→   431→              <div className="pl-[4.5rem]">
   432→   432→                <SecureKeyEditor
   433→   433→                  provider={provider}
   434→   434→                  hasKey={keyStatus?.hasKey ?? false}

   437→   437→                    setTest({ ok, text: message })

   447→   447→                    value={provider.apiKey}
   448→   448→                    onChange={(e) => onUpdate({ apiKey: e.target.value })}
   449→   449→                    placeholder="sk-…"
   450→   450→                    className="h-6 border-zinc-800 bg-zinc-950 px-2 pr-7 font-mono text-[11px]"
