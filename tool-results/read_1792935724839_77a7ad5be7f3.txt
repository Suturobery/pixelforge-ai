    56→/** Effective duration of a frame for badges/scrubber (mirrors core math). */
    57→function frameMs(clip: Clip, index: number): number {
    58→  const f = clip.frames[index]
    59→  if (!f) return 0
    60→  if (f.durationMs > 0) return f.durationMs
    61→  const fps = Math.min(60, Math.max(1, Math.round(clip.fps || 8)))
    62→  return Math.max(1, Math.round(1000 / fps))
    63→}
    64→
    65→function formatMs(ms: number): string {
    66→  return ms >= 1000 ? `${(ms / 1000).toFixed(ms % 1000 === 0 ? 0 : 2)}s` : `${ms}ms`
    67→}

   145→            type="button"

   151→            aria-label={`Edit duration of frame ${index + 1}`}
   152→            title="Frame duration (0 = inherit clip FPS)"

   186→              className="mt-1 h-6 w-full text-[10px] text-zinc-400 hover:text-teal-300"
   187→              onClick={() => setFrameDuration(index, 0)}
   188→            >
   189→              Reset to auto ({clip.fps}fps)
   190→            </Button>
   191→          )}
   192→        </PopoverContent>
   193→      </Popover>
   194→    </div>
   195→  )
   196→}

   200→function Scrubber({ clip, activeIndex }: { clip: Clip; activeIndex: number }) {
   201→  const trackRef = useRef<HTMLDivElement>(null)
   202→  const markerRef = useRef<HTMLDivElement>(null)
   203→  const [hoverMs, setHoverMs] = useState<number | null>(null)
   204→
   205→  // Playhead marker follows the shared engine via direct DOM writes — no
   206→  // React re-render per animation tick (smooth at any fps).
   207→  useEffect(
   208→    () =>
   209→      editorPlayback.onPosition((pos, total) => {
   210→        const el = markerRef.current
   211→        if (el) el.style.left = `${total > 0 ? Math.min(100, (pos / total) * 100) : 0}%`
   212→      }),
   213→    []
   214→  )
   215→
   216→  const msFromEvent = (clientX: number): number => {
   217→    const track = trackRef.current
   218→    if (!track) return 0
   219→    const rect = track.getBoundingClientRect()
   220→    const total = clip.frames.reduce((s, _, i) => s + frameMs(clip, i), 0)
   221→    const ratio = Math.min(1, Math.max(0, (clientX - rect.left) / rect.width))
   222→    return Math.round(ratio * total)
   223→  }
   224→
   225→  const scrub = (clientX: number) => editorPlayback.scrubToMs(msFromEvent(clientX))
   226→
   227→  return (
   228→    <div className="px-2 pb-1">
   229→      <div
   230→        ref={trackRef}
   231→        role="slider"
   232→        aria-label="Timeline position"
   233→        aria-valuemin={1}
   234→        aria-valuemax={clip.frames.length}
   235→        aria-valuenow={activeIndex + 1}
   236→        aria-valuetext={`Frame ${activeIndex + 1} of ${clip.frames.length}`}
   237→        tabIndex={0}
   238→        className="relative flex h-5 cursor-pointer touch-none items-stretch overflow-hidden rounded-md border border-zinc-800 bg-zinc-900"
   239→        onPointerDown={(e) => {
   240→          e.currentTarget.setPointerCapture(e.pointerId)
   241→          scrub(e.clientX)
   242→        }}
   243→        onPointerMove={(e) => {
   244→          if (e.buttons & 1) scrub(e.clientX)
   245→          setHoverMs(msFromEvent(e.clientX))
   246→        }}
   247→        onPointerLeave={() => setHoverMs(null)}
   248→        onKeyDown={(e) => {
   249→          if (e.key === 'ArrowRight') editorPlayback.step(1)
   250→          else if (e.key === 'ArrowLeft') editorPlayback.step(-1)
   251→        }}
   252→      >
   253→        {/* proportional frame segments (width = effective duration) */}
   254→        {clip.frames.map((f, i) => (
   255→          <div
   256→            key={f.id}
   257→            title={`Frame ${i + 1} · ${frameMs(clip, i)}ms`}
   258→            style={{ flexGrow: frameMs(clip, i), flexBasis: 0 }}
   259→            className={`min-w-[3px] border-r border-zinc-950/80 last:border-r-0 ${
   260→              i === activeIndex ? 'bg-teal-700/50' : i % 2 === 0 ? 'bg-zinc-800/70' : 'bg-zinc-800/40'
   261→            }`}
   262→          />
   263→        ))}
   264→        {/* playhead marker */}
   265→        <div
   266→          ref={markerRef}
   267→          aria-hidden
   268→          className="pointer-events-none absolute inset-y-0 left-0 w-0.5 bg-teal-400 shadow-[0_0_6px_rgba(45,212,191,0.8)]"
   269→        />
   270→      </div>
   271→      <div className="mt-0.5 flex justify-between font-mono text-[9px] text-zinc-500">
   272→        <span>
   273→          {hoverMs != null ? `scrub ${formatMs(hoverMs)}` : `${clip.frames.length} frames`}
   274→        </span>
   275→        <span>{formatMs(clip.frames.reduce((s, _, i) => s + frameMs(clip, i), 0))} cycle</span>
   276→      </div>
   277→    </div>
   278→  )
   279→}

   513→        <TooltipTrigger asChild>
   514→          <Button
   515→            variant={playing ? 'secondary' : 'ghost'}
   516→            size="icon"
   517→            className={`h-7 w-7 ${playing ? 'bg-teal-500/15 text-teal-300 ring-1 ring-teal-500/40' : 'text-zinc-300'}`}

   519→            aria-label={playing ? 'Pause animation' : 'Play animation'}
   520→          >
   521→            {playing ? <Pause className="h-4 w-4" /> : <Play className="h-4 w-4" />}
   522→          </Button>
   523→        </TooltipTrigger>
   524→        <TooltipContent>Play / pause (Enter)</TooltipContent>
   525→      </Tooltip>
   526→      <Tooltip>
   527→        <TooltipTrigger asChild>
   528→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400" onClick={() => editorPlayback.step(1)} aria-label="Next frame">
   529→            <ArrowRight className="h-4 w-4" />
   530→          </Button>
   531→        </TooltipTrigger>
   532→        <TooltipContent>Next frame (.)</TooltipContent>
   533→      </Tooltip>
   534→      <Tooltip>
   535→        <TooltipTrigger asChild>
   536→          <Button variant="ghost" size="icon" className="h-7 w-7 text-zinc-400" onClick={() => editorPlayback.stop()} aria-label="Stop and rewind">
   537→            <Square className="h-3.5 w-3.5" />
   538→          </Button>
   539→        </TooltipTrigger>
   540→        <TooltipContent>Stop &amp; rewind</TooltipContent>
   541→      </Tooltip>
   542→    </>
   543→  )
   544→
   545→  return (
   546→    <TooltipProvider delayDuration={200}>
   547→      <section aria-label="Animation timeline" className="shrink-0 border-t border-zinc-800 bg-zinc-950/80">
   548→        {/* clips + transport row */}
   549→        <div className="flex h-9 items-center gap-1 overflow-x-auto border-b border-zinc-800 px-2 pixelforge-scroll">
   550→          {project.clips.map((c) => (
   551→            <div key={c.id} className="flex shrink-0 items-center">
   552→              {renamingClip === c.id ? (
   553→                <Input
   554→                  autoFocus
   555→                  defaultValue={c.name}
   556→                  className="h-6 w-24 border-zinc-700 bg-zinc-900 font-mono text-xs"
   557→                  onBlur={(e) => {
   558→                    setClipProp(c.id, { name: e.target.value || c.name })
   559→                    setRenamingClip(null)
   560→                  }}
   561→                  onKeyDown={(e) => {
   562→                    if (e.key === 'Enter') (e.target as HTMLInputElement).blur()
   563→                    if (e.key === 'Escape') setRenamingClip(null)
   564→                  }}
   565→                />
   566→              ) : (
   567→                <button
   568→                  type="button"
   569→                  onDoubleClick={() => setRenamingClip(c.id)}
   570→                  onClick={() => selectClip(c.id)}
   571→                  className={`rounded-t-md border-b-2 px-3 py-1 font-mono text-xs transition-colors ${
   572→                    c.id === clip.id
   573→                      ? 'border-teal-400 bg-zinc-900 text-teal-300'
   574→                      : 'border-transparent text-zinc-400 hover:text-zinc-300'
   575→                  }`}
   576→                  title={`${c.name} — ${c.frames.length} frames · double-click to rename`}
   577→                >
   578→                  {c.name}
   579→                  <span className="ml-2 text-[10px] text-zinc-400">{c.frames.length}f</span>
   580→                </button>
   581→              )}
   582→            </div>
   583→          ))}
   584→          <Tooltip>
   585→            <TooltipTrigger asChild>
   586→              <Button
   587→                variant="ghost"
   588→                size="icon"
   589→                className="h-6 w-6 shrink-0 text-zinc-400"
   590→                onClick={() => addClip(`clip ${project.clips.length + 1}`)}
   591→                aria-label="Add clip"
   592→              >
   593→                <Plus className="h-3.5 w-3.5" />
   594→              </Button>
   595→            </TooltipTrigger>
   596→            <TooltipContent>New clip</TooltipContent>
   597→          </Tooltip>
   598→          {project.clips.length > 1 && (
   599→            <Tooltip>
   600→              <TooltipTrigger asChild>
   601→                <Button
   602→                  variant="ghost"
   603→                  size="icon"
   604→                  className="h-6 w-6 shrink-0 text-red-400/70 hover:text-red-300"
   605→                  onClick={() => deleteClip(clip.id)}
   606→                  aria-label="Delete current clip"
   607→                >
   608→                  <Trash2 className="h-3.5 w-3.5" />
   609→                </Button>
   610→              </TooltipTrigger>
   611→              <TooltipContent>Delete clip</TooltipContent>
   612→            </Tooltip>
   613→          )}
   614→
   615→          <div className="ml-auto flex shrink-0 items-center gap-1 pl-3">
   616→            {/* playback state chip (Task 6 §5 visible playback state) */}
   617→            <span
   618→              role="status"
   619→              className={`mr-1 hidden shrink-0 rounded px-1.5 py-0.5 font-mono text-[9px] uppercase tracking-wider sm:inline ${
   620→                playing ? 'bg-teal-500/15 text-teal-300' : 'bg-zinc-800 text-zinc-500'
   621→              }`}
   622→              aria-live="polite"
   623→            >
   624→              {playing ? `▶ ${frameIndex + 1}/${clip.frames.length}` : `paused ${frameIndex + 1}/${clip.frames.length}`}
   625→            </span>
   626→            {transport}
   627→            {/* loop mode (Task 6 §3): forward · reverse · ping-pong */}
   628→            <Select
   629→              value={clip.loopMode}
   630→              onValueChange={(v) => setClipProp(clip.id, { loopMode: v as Clip['loopMode'] })}
   631→            >
   632→              <SelectTrigger
   633→                // NOTE: the base trigger ships `data-[size=default]:h-9`, whose
   634→                // attribute-variant specificity beats a plain `h-7` — the same-variant
   635→                // override here is what actually pins it to the 28px transport row.
   636→                className="data-[size=default]:h-7 w-[96px] gap-1 border-zinc-800 bg-zinc-900 dark:bg-zinc-900 dark:hover:bg-zinc-800 px-2 font-mono text-[10px] text-zinc-300"
   637→                aria-label="Playback mode"
   638→              >
   639→                <SelectValue />
   640→              </SelectTrigger>
   641→              <SelectContent>
   642→                <SelectItem value="forward">Forward</SelectItem>
   643→                <SelectItem value="reverse">Reverse</SelectItem>
   644→                <SelectItem value="pingpong">Ping-pong</SelectItem>
   645→              </SelectContent>
   646→            </Select>
   647→            <Tooltip>
   648→              <TooltipTrigger asChild>
   649→                <Button
   650→                  variant="ghost"
   651→                  size="icon"
   652→                  className={`h-7 w-7 ${clip.loop ? 'text-teal-300' : 'text-zinc-400'}`}
   653→                  onClick={() => setClipProp(clip.id, { loop: !clip.loop })}
   654→                  aria-label="Loop clip"
   655→                  aria-pressed={clip.loop}
   656→                >
   657→                  <Repeat className="h-4 w-4" />
   658→                </Button>
   659→              </TooltipTrigger>
   660→              <TooltipContent>{clip.loop ? 'Looping' : 'Play once'}</TooltipContent>
   661→            </Tooltip>
   662→            <OnionPopover />
   663→            <label className="flex items-center gap-1.5 font-mono text-[10px] text-zinc-400">
   664→              FPS
   665→              <StepperInput
   666→                size="sm"
   667→                value={clip.fps}
   668→                onChange={(v) => setClipProp(clip.id, { fps: Math.min(60, Math.max(1, v)) })}
   669→                min={1}
   670→                max={60}
   671→                ariaLabel="Frames per second"
   672→              />
   673→            </label>
   674→            <Tooltip>
   675→              <TooltipTrigger asChild>
   676→                <Button
   677→                  variant="ghost"
   678→                  size="icon"
   679→                  className="h-7 w-7 text-zinc-400"
   680→                  onClick={() => setTimelineOpen(!timelineOpen)}
   681→                  aria-label={timelineOpen ? 'Collapse frames strip' : 'Expand frames strip'}
   682→                  aria-pressed={timelineOpen}
   683→                >
   684→                  {timelineOpen ? <ChevronDown className="h-4 w-4" /> : <ChevronUp className="h-4 w-4" />}
   685→                </Button>
   686→              </TooltipTrigger>
   687→              <TooltipContent>{timelineOpen ? 'Collapse frames' : 'Expand frames'}</TooltipContent>
   688→            </Tooltip>
   689→          </div>
   690→        </div>
   691→
   692→        {/* frames strip (collapsible to give the canvas more vertical space) */}
   693→        {timelineOpen && (
   694→          <>
   695→            <Scrubber clip={clip} activeIndex={frameIndex} />
   696→            <div className="flex items-center gap-1 border-b border-zinc-800/60 px-2 py-1">
   697→              <Tooltip>
   698→                <TooltipTrigger asChild>
   699→                  <Button variant="ghost" size="icon" className="h-6 w-6 text-zinc-400" onClick={() => addFrame(false)} aria-label="Add empty frame after current">
   700→                    <Plus className="h-3.5 w-3.5" />
   701→                  </Button>
   702→                </TooltipTrigger>
   703→                <TooltipContent>Add frame (Alt+N)</TooltipContent>
   704→              </Tooltip>
   705→              <Tooltip>
   706→                <TooltipTrigger asChild>
   707→                  <Button variant="ghost" size="icon" className="h-6 w-6 text-zinc-400" onClick={() => addFrame(true)} aria-label="Duplicate current frame">
   708→                    <Copy className="h-3.5 w-3.5" />
   709→                  </Button>
   710→                </TooltipTrigger>
   711→                <TooltipContent>Duplicate frame</TooltipContent>
   712→              </Tooltip>
   713→              <Tooltip>
   714→                <TooltipTrigger asChild>
   715→                  <Button
   716→                    variant="ghost"
   717→                    size="icon"
   718→                    className="h-6 w-6 text-zinc-400"
   719→                    onClick={() => moveFrame(frameIndex, frameIndex - 1)}
   720→                    disabled={frameIndex === 0}
   721→                    aria-label="Move frame left"
   722→                  >
   723→                    <ArrowLeft className="h-3.5 w-3.5" />
   724→                  </Button>
   725→                </TooltipTrigger>
   726→                <TooltipContent>Move left</TooltipContent>
   727→              </Tooltip>
   728→              <Tooltip>
   729→                <TooltipTrigger asChild>
   730→                  <Button
   731→                    variant="ghost"
   732→                    size="icon"
   733→                    className="h-6 w-6 text-zinc-400"
   734→                    onClick={() => moveFrame(frameIndex, frameIndex + 1)}
   735→                    disabled={frameIndex >= clip.frames.length - 1}
   736→                    aria-label="Move frame right"
   737→                  >
   738→                    <ArrowRight className="h-3.5 w-3.5" />
   739→                  </Button>
   740→                </TooltipTrigger>
   741→                <TooltipContent>Move right</TooltipContent>
   742→              </Tooltip>
   743→              <Tooltip>
   744→                <TooltipTrigger asChild>
   745→                  <Button
   746→                    variant="ghost"
   747→                    size="icon"
   748→                    className="h-6 w-6 text-red-400/80 hover:text-red-300"
   749→                    onClick={() => deleteFrame(frameIndex)}
   750→                    disabled={clip.frames.length <= 1}
   751→                    aria-label="Delete current frame"
   752→                  >
   753→                    <Trash2 className="h-3.5 w-3.5" />
   754→                  </Button>
   755→                </TooltipTrigger>
   756→                <TooltipContent>Delete frame</TooltipContent>
   757→              </Tooltip>
   758→              <span className="ml-2 flex min-w-0 items-center gap-2 truncate font-mono text-[10px] text-zinc-400">
   759→                <span>
   760→                  frame {frameIndex + 1}/{clip.frames.length} · {clip.name} · {clip.fps}fps ·{' '}
   761→                  {clip.loopMode}
   762→                  {!clip.loop && ' · once'}
   763→                </span>
   764→                {playing && (
   765→                  <span className="inline-flex items-center gap-1 text-teal-300">
   766→                    <ArrowLeftRight className="h-3 w-3 animate-pulse" aria-hidden />
   767→                    playing
   768→                  </span>
   769→                )}
   770→              </span>
   771→            </div>
   772→            <div ref={stripRef} className="pixelforge-scroll flex items-center gap-2 overflow-x-auto px-2 py-2">
   773→              {clip.frames.map((f, i) => (
   774→                <div key={f.id} data-frame-index={i} className="shrink-0">
   775→                  <FrameThumb
   776→                    clip={clip}
   777→                    frame={f}
   778→                    index={i}
   779→                    active={i === frameIndex}
   780→                    dropBefore={dropTarget === i && dragFromRef.current !== i}
   781→                    w={project.width}
   782→                    h={project.height}
   783→                    tick={renderTick}
   784→                    playing={playing}
   785→                    onSelect={() => selectFrame(i)}
   786→                    onDragStart={() => {
   787→                      dragFromRef.current = i
   788→                    }}
   789→                    onDragOver={(e) => {
   790→                      e.preventDefault()
   791→                      if (dragFromRef.current != null && dropTarget !== i) setDropTarget(i)
   792→                    }}
   793→                    onDrop={() => commitDrop(i)}
   794→                    onDragEnd={() => {
   795→                      dragFromRef.current = null
   796→                      setDropTarget(null)
   797→                    }}
   798→                  />
   799→                </div>
   800→              ))}
   801→            </div>
   802→          </>
   803→        )}
   804→      </section>
   805→    </TooltipProvider>
   806→  )
