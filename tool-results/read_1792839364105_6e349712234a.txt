    30→interface ActiveLoc {
    31→  clip: Clip
    32→  frame: Frame
    33→  frameIndex: number
    34→  layer: Layer
    35→}
    36→
    37→function locate(
    38→  project: Project,
    39→  activeClipId: string,
    40→  activeFrameIndex: number,
    41→  activeLayerId: string
    42→): ActiveLoc | null {
    43→  const clip =
    44→    project.clips.find((c) => c.id === activeClipId) ?? project.clips[0]
    45→  if (!clip || clip.frames.length === 0) return null
    46→  const frameIndex = Math.min(Math.max(0, activeFrameIndex), clip.frames.length - 1)
    47→  const frame = clip.frames[frameIndex]
    48→  if (!frame || frame.layers.length === 0) return null
    49→  const layer =
    50→    frame.layers.find((l) => l.id === activeLayerId) ?? frame.layers[frame.layers.length - 1]
    51→  return { clip, frame, frameIndex, layer }
    52→}

    54→/** Immutably replace a layer's buffer deep inside the project. */
    55→function withLayerData(
    56→  project: Project,
    57→  clipId: string,
    58→  frameId: string,
    59→  layerId: string,
    60→  data: Uint8ClampedArray
    61→): Project {
    62→  return {
    63→    ...project,
    64→    clips: project.clips.map((c) =>
    65→      c.id !== clipId
    66→        ? c
    67→        : {
    68→            ...c,
    69→            frames: c.frames.map((f) =>
    70→              f.id !== frameId
    71→                ? f
    72→                : { ...f, layers: f.layers.map((l) => (l.id !== layerId ? l : { ...l, data })) }
    73→            ),
    74→          }
    75→    ),
    76→  }
    77→}
    78→
    79→/**
    80→ * Layers are CLIP-WIDE: every frame in a clip shares the same layer stack
    81→ * (same names, same order — Aseprite-style). Each frame keeps its own cell
    82→ * buffers/ids, so structural layer ops must apply to ALL frames of the clip,
    83→ * not just the visible one (previously adding a layer on frame 1 made it
    84→ * disappear when switching to frame 2 — the "why is there only 1 layer?" bug).
    85→ */
    86→function mapClipFrames(
    87→  project: Project,
    88→  clipId: string,
    89→  fn: (frame: Frame, index: number) => Frame
    90→): Project {
    91→  return {
    92→    ...project,
    93→    clips: project.clips.map((c) =>
    94→      c.id !== clipId ? c : { ...c, frames: c.frames.map((f, i) => fn(f, i)) }
    95→    ),
    96→  }
    97→}

   224→export const useEditor = create<EditorState>()((set, get) => ({
   225→  project: initialProject,
   226→  activeClipId: 'cl_initial',
   227→  activeFrameIndex: 0,
   228→  activeLayerId: 'ly_initial',
   229→  tool: 'pencil',
   230→  primary: '#2ce8f5',
   231→  secondary: '#1a1c2c',
   232→  brushSize: 1,
   233→  mirrorX: false,
   234→  zoom: 12,
   235→  panX: 0,
   236→  panY: 0,
   237→  showGrid: true,
   238→  onionSkin: true,
   239→  hover: null,
   240→  playing: false,
   241→  past: [],
   242→  future: [],
   243→  dirty: false,
   244→  savedAt: null,
   245→  autosave: true,
   246→  rightTab: 'layers',
   247→  paletteOpen: true,
   248→  rightOpen: true,
   249→  rightWidth: 320,
   250→  timelineOpen: true,
   251→  renderTick: 0,
   252→
   253→  newProject: (p) =>
   254→    set({
   255→      project: p,
   256→      activeClipId: p.clips[0]?.id ?? '',
   257→      activeFrameIndex: 0,
   258→      activeLayerId: p.clips[0]?.frames[0]?.layers.at(-1)?.id ?? '',
   259→      past: [],
   260→      future: [],
   261→      dirty: true,
   262→      savedAt: null,
   263→      playing: false,
   264→      zoom: 12,
   265→      panX: 0,
   266→      panY: 0,
   267→    }),
   268→
   269→  loadProject: (p) =>
   270→    set({
   271→      project: p,
   272→      activeClipId: p.clips[0]?.id ?? '',
   273→      activeFrameIndex: 0,
   274→      activeLayerId: p.clips[0]?.frames[0]?.layers.at(-1)?.id ?? '',
   275→      past: [],
   276→      future: [],
   277→      dirty: false,
   278→      savedAt: Date.now(),
   279→      playing: false,
   280→      zoom: 12,
   281→      panX: 0,
   282→      panY: 0,
   283→    }),
   284→
   285→  setName: (name) => {
   286→    const s = get()
   287→    set({ project: { ...s.project, name }, dirty: true })
   288→  },
   289→
   290→  update: (mutator, label) => {
   291→    const s = get()
   292→    const before = s.project
   293→    const project = { ...before }
   294→    mutator(project)
   295→    const entry: HistoryEntry = { kind: 'project', label, before, after: project }
   296→    set({
   297→      project,
   298→      past: [...s.past, entry].slice(-useAppSettings.getState().historyLimit),
   299→      future: [],
   300→      dirty: true,
   301→    })
   302→  },
   303→
   304→  beginStroke: () => {
   305→    const s = get()
   306→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   307→    if (!loc) return null
   308→    const { clip, frame, frameIndex, layer } = loc
   309→    const before = layer.data
   310→    const working = new Uint8ClampedArray(before)
   311→    const newFrame: Frame = {
   312→      ...frame,
   313→      layers: frame.layers.map((l) => (l.id === layer.id ? { ...l, data: working } : l)),
   314→    }
   315→    const project: Project = {
   316→      ...s.project,
   317→      clips: s.project.clips.map((c) =>
   318→        c.id !== clip.id
   319→          ? c
   320→          : { ...c, frames: c.frames.map((f, i) => (i === frameIndex ? newFrame : f)) }
   321→      ),
   322→    }
   323→    stroke = { clipId: clip.id, frameId: frame.id, layerId: layer.id, before }
   324→    set({ project, renderTick: s.renderTick + 1 })
   325→    return { width: s.project.width, height: s.project.height, buffer: working }
   326→  },
   327→
   328→  endStroke: (label, changed) => {
   329→    if (!stroke) return
   330→    const st = stroke
   331→    stroke = null
   332→    if (!changed) return
   333→    const s = get()
   334→    const entry: HistoryEntry = { kind: 'pixels', label, ...st, after: st.before }
   335→    // Locate the working buffer in the *current* project (it may have moved if
   336→    // a structural change happened mid-stroke — rare, but stay safe)
   337→    const loc = locate(s.project, st.clipId, s.activeFrameIndex, st.layerId)
   338→    let after = st.before
   339→    if (loc && loc.clip.id === st.clipId && loc.frame.id === st.frameId && loc.layer.id === st.layerId) {
   340→      after = loc.layer.data
   341→    } else {
   342→      const clip = s.project.clips.find((c) => c.id === st.clipId)
   343→      const frame = clip?.frames.find((f) => f.id === st.frameId)
   344→      const layer = frame?.layers.find((l) => l.id === st.layerId)
   345→      if (layer) after = layer.data
   346→      else return
   347→    }
   348→    entry.after = after
   349→    set({
   350→      past: [...s.past, entry].slice(-useAppSettings.getState().historyLimit),
   351→      future: [],
   352→      dirty: true,
   353→    })
   354→  },
   355→
   356→  cancelStroke: () => {
   357→    if (!stroke) return
   358→    const st = stroke
   359→    stroke = null
   360→    const s = get()
   361→    set({
   362→      project: withLayerData(s.project, st.clipId, st.frameId, st.layerId, st.before),
   363→      renderTick: s.renderTick + 1,
   364→    })
   365→  },
   366→
   367→  bump: () => set((s) => ({ renderTick: s.renderTick + 1 })),
   368→
   369→  undo: () => {
   370→    const s = get()
   371→    if (stroke) return
   372→    const past = [...s.past]
   373→    while (past.length > 0) {
   374→      const entry = past.pop() as HistoryEntry
   375→      if (entry.kind === 'pixels') {
   376→        const cur = locate(s.project, entry.clipId, s.activeFrameIndex, entry.layerId)
   377→        const layer = cur && cur.clip.id === entry.clipId && cur.frame.id === entry.frameId && cur.layer.id === entry.layerId
   378→          ? cur.layer
   379→          : s.project.clips
   380→              .find((c) => c.id === entry.clipId)
   381→              ?.frames.find((f) => f.id === entry.frameId)
   382→              ?.layers.find((l) => l.id === entry.layerId)
   383→        if (!layer) continue // structure changed underneath; keep unwinding
   384→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.before)
   385→        const redoEntry: HistoryEntry = { ...entry, before: layer.data, after: entry.after }
   386→        set({ project, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   387→        return
   388→      }
   389→      const redoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: s.project, after: entry.after }
   390→      set({ project: entry.before, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   391→      return
   392→    }
   393→  },
   394→
   395→  redo: () => {
   396→    const s = get()
   397→    if (stroke) return
   398→    const future = [...s.future]
   399→    while (future.length > 0) {
   400→      const entry = future.pop() as HistoryEntry
   401→      if (entry.kind === 'pixels') {
   402→        const layer = s.project.clips
   403→          .find((c) => c.id === entry.clipId)
   404→          ?.frames.find((f) => f.id === entry.frameId)
   405→          ?.layers.find((l) => l.id === entry.layerId)
   406→        if (!layer) continue
   407→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.after)
   408→        const undoEntry: HistoryEntry = { ...entry, before: entry.before, after: layer.data }
   409→        set({ project, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   410→        return
   411→      }
   412→      const undoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: entry.before, after: s.project }
   413→      set({ project: entry.after, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   414→      return
   415→    }
   416→  },
   417→
   418→  // --- structural commands ---
   419→
   420→  addFrame: (copyContent) => {
   421→    const s = get()
   422→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   423→    if (!loc) return
   424→    const insertAt = s.activeFrameIndex + 1
   425→    const newFrame: Frame = {
   426→      id: uid('fr'),
   427→      layers: loc.frame.layers.map((l) => ({
   428→        id: uid('ly'),
   429→        name: l.name,
   430→        visible: l.visible,
   431→        opacity: l.opacity,
   432→        data: copyContent
   433→          ? new Uint8ClampedArray(l.data)
   434→          : new Uint8ClampedArray(s.project.width * s.project.height * 4),
   435→      })),
   436→    }
   437→    get().update((p) => {
   438→      p.clips = p.clips.map((c) =>
   439→        c.id !== loc.clip.id
   440→          ? c
   441→          : { ...c, frames: [...c.frames.slice(0, insertAt), newFrame, ...c.frames.slice(insertAt)] }
   442→      )
   443→    }, copyContent ? 'Duplicate frame' : 'Add frame')
   444→    set({ activeFrameIndex: insertAt, activeLayerId: newFrame.layers.at(-1)?.id ?? '' })
   445→    toast(
   446→      copyContent
   447→        ? { title: `Frame ${insertAt + 1} duplicated` }
   448→        : {
   449→            title: `Frame ${insertAt + 1} added (empty)`,
   450→            description: `The canvas now shows the new frame — frame ${s.activeFrameIndex + 1} and its pixels are safe on the timeline below.`,
   451→          }
   452→    )
   453→  },
   454→
   455→  deleteFrame: (index) => {
   456→    const s = get()
   457→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   458→    if (!loc || loc.clip.frames.length <= 1) return
   459→    get().update((p) => {
   460→      p.clips = p.clips.map((c) =>
   461→        c.id !== loc.clip.id ? c : { ...c, frames: c.frames.filter((_, i) => i !== index) }
   462→      )
   463→    }, 'Delete frame')
   464→    const len = loc.clip.frames.length - 1
   465→    set({ activeFrameIndex: Math.min(index, len - 1), activeLayerId: loc.layer.id })
   466→  },
   467→
   468→  moveFrame: (from, to) => {
   469→    const s = get()
   470→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   471→    if (!loc) return
   472→    const len = loc.clip.frames.length
   473→    if (to < 0 || to >= len || from === to) return
   474→    get().update((p) => {
   475→      p.clips = p.clips.map((c) => {
   476→        if (c.id !== loc.clip.id) return c
   477→        const frames = [...c.frames]
   478→        const [f] = frames.splice(from, 1)
   479→        frames.splice(to, 0, f)
   480→        return { ...c, frames }
   481→      })
   482→    }, 'Reorder frame')
   483→    set({ activeFrameIndex: to })
   484→  },
   485→
   486→  addLayer: () => {
   487→    const s = get()
   488→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   489→    if (!loc) return
   490→    const name = `layer ${loc.frame.layers.length + 1}`
   491→    // CLIP-WIDE: every frame gets its own cell of the new layer (same name,
   492→    // independent buffer) so the stack looks identical on every frame
   493→    get().update((p) => {
   494→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   495→        ...f,
   496→        layers: [...f.layers, createLayer(s.project.width, s.project.height, name)],
   497→      }))
   498→    }, 'Add layer')
   499→    // grab the id on the *active* frame (ids are per-frame)
   500→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   501→    const newLayerIdOnActiveFrame = cur?.frame.layers.at(-1)?.id ?? ''
   502→    set({ activeLayerId: newLayerIdOnActiveFrame })
   503→    toast({
   504→      title: `Layer "${name}" added`,
   505→      description: `Added to all ${loc.clip.frames.length} frame${loc.clip.frames.length === 1 ? '' : 's'} in this clip.`,
   506→    })
   507→  },
   508→
   509→  deleteLayer: (layerId) => {
   510→    const s = get()
   511→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   512→    if (!loc || loc.frame.layers.length <= 1) return
   513→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   514→    if (idx < 0) return
   515→    // CLIP-WIDE: drop the same slot from every frame
   516→    get().update((p) => {
   517→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   518→        ...f,
   519→        layers: f.layers.filter((_, i) => i !== idx),
   520→      }))
   521→    }, 'Delete layer')
   522→    const next = loc.frame.layers[Math.min(idx, loc.frame.layers.length - 2)]
   523→    set({ activeLayerId: next?.id ?? '' })
   524→  },
   525→
   526→  duplicateLayer: (layerId) => {
   527→    const s = get()
   528→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   529→    if (!loc) return
   530→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   531→    if (idx < 0) return
   532→    const srcName = loc.frame.layers[idx].name
   533→    // CLIP-WIDE: duplicate the slot in every frame (each frame copies its own
   534→    // cell content), inserted directly above the original
   535→    get().update((p) => {
   536→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   537→        const src = f.layers[idx]
   538→        if (!src) return f
   539→        const copy: Layer = {
   540→          ...src,
   541→          id: uid('ly'),
   542→          name: `${srcName} copy`,
   543→          data: new Uint8ClampedArray(src.data),
   544→        }
   545→        return {
   546→          ...f,
   547→          layers: [...f.layers.slice(0, idx + 1), copy, ...f.layers.slice(idx + 1)],
   548→        }
   549→      })
   550→    }, 'Duplicate layer')
   551→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   552→    const copyIdOnActiveFrame = cur?.frame.layers[idx + 1]?.id ?? ''
   553→    set({ activeLayerId: copyIdOnActiveFrame })
   554→  },
   555→
   556→  mergeDown: (layerId) => {
   557→    const s = get()
   558→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   559→    if (!loc) return
   560→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   561→    if (idx <= 0) return // nothing below
   562→    const { width: w, height: h } = s.project
   563→    const bottomId = loc.frame.layers[idx - 1].id
   564→    // CLIP-WIDE: composite the same slot pair on every frame
   565→    get().update((p) => {
   566→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   567→        if (idx >= f.layers.length) return f
   568→        const fTop = f.layers[idx]
   569→        const fBottom = f.layers[idx - 1]
   570→        if (!fTop || !fBottom) return f
   571→        const merged = new Uint8ClampedArray(fBottom.data)
   572→        for (let i = 0; i < w * h; i++) {
   573→          const ta = (fTop.data[i * 4 + 3] / 255) * fTop.opacity
   574→          if (ta >= 1) {
   575→            merged[i * 4] = fTop.data[i * 4]
   576→            merged[i * 4 + 1] = fTop.data[i * 4 + 1]
   577→            merged[i * 4 + 2] = fTop.data[i * 4 + 2]
   578→            merged[i * 4 + 3] = fTop.data[i * 4 + 3]
   579→          } else if (ta > 0) {
   580→            const ba = merged[i * 4 + 3] / 255
   581→            const outA = ta + ba * (1 - ta)
   582→            if (outA === 0) continue
   583→            for (let ch = 0; ch < 3; ch++) {
   584→              merged[i * 4 + ch] = Math.round(
   585→                (fTop.data[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   586→              )
   587→            }
   588→            merged[i * 4 + 3] = Math.round(outA * 255)
   589→          }
   590→        }
   591→        return {
   592→          ...f,
   593→          layers: f.layers.filter((_, i) => i !== idx).map((l, i) => (i === idx - 1 ? { ...l, data: merged } : l)),
   594→        }
   595→      })
   596→    }, 'Merge down')
   597→    set({ activeLayerId: bottomId })
   598→  },
   599→
   600→  moveLayer: (layerId, dir) => {
   601→    const s = get()
   602→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   603→    if (!loc) return
   604→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   605→    const to = idx + dir
   606→    if (idx < 0 || to < 0 || to >= loc.frame.layers.length) return
   607→    // CLIP-WIDE: reorder the same slot on every frame
   608→    get().update((p) => {
   609→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   610→        if (to >= f.layers.length) return f
   611→        const layers = [...f.layers]
   612→        const [l] = layers.splice(idx, 1)
   613→        if (!l) return f
   614→        layers.splice(to, 0, l)
   615→        return { ...f, layers }
   616→      })
   617→    }, 'Reorder layer')
   618→    // the moved layer keeps its id on the active frame — selection unchanged
   619→  },
   620→
   621→  setLayerProp: (layerId, patch) => {
   622→    const s = get()
   623→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   624→    if (!loc) return
   625→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   626→    if (idx < 0) return
   627→    // CLIP-WIDE: name/visibility/opacity apply to the slot on every frame so
   628→    // the panel shows a consistent stack everywhere (eye toggle included)
   629→    get().update((p) => {
   630→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   631→        ...f,
   632→        layers: f.layers.map((l, i) => (i === idx ? { ...l, ...patch } : l)),
   633→      }))
   634→    }, 'Layer property')
   635→  },
   636→
   637→  addClip: (name) => {
   638→    const s = get()
   639→    const clip: Clip = {
   640→      id: uid('cl'),
   641→      name,
   642→      fps: Math.min(60, Math.max(1, useAppSettings.getState().defaultClipFps)),
   643→      loop: true,
   644→      frames: [createFrame(s.project.width, s.project.height, ['layer 1'])],
   645→    }
   646→    get().update((p) => {
   647→      p.clips = [...p.clips, clip]
   648→    }, 'Add clip')
   649→    set({ activeClipId: clip.id, activeFrameIndex: 0, activeLayerId: clip.frames[0].layers[0].id })
   650→  },
   651→
   652→  deleteClip: (clipId) => {
   653→    const s = get()
   654→    if (s.project.clips.length <= 1) return
   655→    const idx = s.project.clips.findIndex((c) => c.id === clipId)
   656→    if (idx < 0) return
   657→    get().update((p) => {
   658→      p.clips = p.clips.filter((c) => c.id !== clipId)
   659→    }, 'Delete clip')
   660→    const next = get().project.clips[Math.min(idx, get().project.clips.length - 1)]
   661→    set({ activeClipId: next.id, activeFrameIndex: 0, activeLayerId: next.frames[0]?.layers.at(-1)?.id ?? '' })
   662→  },
   663→
   664→  setClipProp: (clipId, patch) => {
   665→    const s = get()
   666→    get().update((p) => {
   667→      p.clips = p.clips.map((c) => (c.id === clipId ? { ...c, ...patch } : c))
   668→    }, 'Clip property')
   669→  },
   670→
   671→  addPaletteColor: (hex) => {
   672→    const s = get()
   673→    if (s.project.palette.includes(hex)) return
   674→    get().update((p) => {
   675→      p.palette = [...p.palette, hex]
   676→    }, 'Add palette color')
   677→  },
   678→
   679→  removePaletteColor: (hex) => {
   680→    get().update((p) => {
   681→      p.palette = p.palette.filter((c) => c !== hex)
   682→    }, 'Remove palette color')
   683→  },
   684→
   685→  // --- selection & ui ---
   686→
   687→  selectClip: (id) => {
   688→    const s = get()
   689→    const clip = s.project.clips.find((c) => c.id === id)
   690→    if (!clip) return
   691→    set({
   692→      activeClipId: id,
   693→      activeFrameIndex: Math.min(s.activeFrameIndex, clip.frames.length - 1),
   694→      activeLayerId: clip.frames[Math.min(s.activeFrameIndex, clip.frames.length - 1)]?.layers.at(-1)?.id ?? '',
   695→      playing: false,
   696→    })
   697→  },
   698→
   699→  selectFrame: (index) => {
   700→    const s = get()
   701→    const clip = s.project.clips.find((c) => c.id === s.activeClipId) ?? s.project.clips[0]
   702→    if (!clip) return
   703→    const i = Math.min(Math.max(0, index), clip.frames.length - 1)
   704→    // keep the user's layer context across frames: layers are clip-wide and
   705→    // share names, so match the active layer by name (fall back to the top)
   706→    const prev = clip.frames[Math.min(s.activeFrameIndex, clip.frames.length - 1)]
   707→    const prevName = prev?.layers.find((l) => l.id === s.activeLayerId)?.name
   708→    const nextFrame = clip.frames[i]
   709→    const matched = prevName ? nextFrame?.layers.find((l) => l.name === prevName) : undefined
   710→    set({ activeFrameIndex: i, activeLayerId: matched?.id ?? nextFrame?.layers.at(-1)?.id ?? s.activeLayerId })
   711→  },

   746→export function getActiveClip(s: Pick<EditorState, 'project' | 'activeClipId'>): Clip {
   747→  return s.project.clips.find((c) => c.id === s.activeClipId) ?? s.project.clips[0]
   748→}

   750→export function getActiveFrame(s: Pick<EditorState, 'project' | 'activeClipId' | 'activeFrameIndex'>): Frame {
   751→  const clip = getActiveClip(s)
   752→  return clip.frames[Math.min(s.activeFrameIndex, clip.frames.length - 1)] ?? clip.frames[0]
   753→}
