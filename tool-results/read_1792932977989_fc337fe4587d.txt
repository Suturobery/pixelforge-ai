     3→// Architecture (v3.1 §12 Task 4 "Port renderer onto core"):
     4→//   • @pixelforge/core owns the DOCUMENT: model v2 (per-clip layer stacks +
     5→//     per-frame cels, linked cels, trash), the CommandBus (the single
     6→//     mutation path — UI/AI all execute commands), tile-diff history with a

    11→//   • Action names from the prototype store are PRESERVED — components call
    12→//     the same API; each action now maps onto a bus command instead of
    13→//     mutating a v1 document by hand.
    14→//
    15→// The bus lives OUTSIDE React. Every bus change event re-syncs the mirrored
    16→// slice; mid-stroke painting bumps `renderTick` only (no bus traffic).

    98→/** Settings `historyLimit` presets map onto v3.1 byte budgets (§2.5). */
    99→function historyBudgetBytes(limit: number): number {
   100→  switch (limit) {
   101→    case 20: return 16 * 1024 * 1024
   102→    case 40: return 32 * 1024 * 1024
   103→    case 100: return 64 * 1024 * 1024
   104→    case 200: return 128 * 1024 * 1024
   105→    default: return 64 * 1024 * 1024
   106→  }
   107→}
   108→bus.setHistoryBudget(historyBudgetBytes(useAppSettings.getState().historyLimit))
   109→useAppSettings.subscribe((s, prev) => {
   110→  if (s.historyLimit !== prev.historyLimit) {
   111→    bus.setHistoryBudget(historyBudgetBytes(s.historyLimit))
   112→  }

   115→// Shim-registered command: palette presets replace the whole palette in ONE
   116→// undoable step (built-ins only add/remove single colors).
   117→bus.register({
   118→  type: 'palette.replaceAll',
   119→  defaultLabel: 'Apply palette',
   120→  validate(payload) {
   121→    const p = payload as { colors?: unknown }
   122→    const colors = p?.colors
   123→    if (!Array.isArray(colors) || colors.length === 0 || colors.length > 256) {
   124→      throw new CommandError('VALIDATION_FAILED', 'palette.replaceAll: colors must be a 1..256 array')
   125→    }
   126→    for (const c of colors) {
   127→      if (typeof c !== 'string' || !/^#[0-9a-f]{6}([0-9a-f]{2})?$/i.test(c)) {
   128→        throw new CommandError('VALIDATION_FAILED', `palette.replaceAll: bad hex "${String(c)}"`)
   129→      }
   130→    }
   131→  },
   132→  apply(payload, ctx) {
   133→    return {
   134→      project: { ...ctx.project, palette: [...(payload as { colors: string[] }).colors] },
   135→      label: 'Apply palette',
   136→    }
   137→  },
   138→})

   144→interface ActiveRefs {
   145→  clipId: string
   146→  frameId: string
   147→  frameIndex: number
   148→  layerId: string

   151→/** Resolve the active pointers against the LIVE document (safe fallbacks). */
   152→function activeRefs(s: Pick<EditorState, 'activeClipId' | 'activeFrameIndex' | 'activeLayerId'>): ActiveRefs | null {
   153→  const project = bus.getProject()
   154→  const clip = project.clips.find((c) => c.id === s.activeClipId) ?? project.clips[0]
   155→  if (!clip) return null
   156→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   157→  const frame = clip.frames[frameIndex]
   158→  if (!frame) return null
   159→  const layer = clip.layers.find((l) => l.id === s.activeLayerId) ?? clip.layers[clip.layers.length - 1]
   160→  if (!layer) return null
   161→  return { clipId: clip.id, frameId: frame.id, frameIndex, layerId: layer.id }
   162→}
   163→
   164→function findNewLayerId(before: Set<string>, project: Project, clipId: string): string | null {
   165→  const clip = project.clips.find((c) => c.id === clipId)
   166→  if (!clip) return null
   167→  for (let i = clip.layers.length - 1; i >= 0; i--) {
   168→    const l = clip.layers[i]
   169→    if (!before.has(l.id)) return l.id
   170→  }
   171→  return null
   172→}
   173→
   174→function layerIds(project: Project, clipId: string): Set<string> {
   175→  const clip = project.clips.find((c) => c.id === clipId)
   176→  return new Set(clip?.layers.map((l) => l.id) ?? [])
   177→}

   179→/**
   180→ * Stroke-flow helper for one-shot selection edits (cut/delete/paste):
   181→ * begin → mutate the working buffer → end as ONE tile-diff history entry.
   182→ * Any failure cancels the stroke — the document is never half-mutated.
   183→ */
   184→function withStroke(
   185→  label: string,
   186→  refs: ActiveRefs,
   187→  fn: (handle: CoreStrokeHandle) => boolean
   188→): boolean {
   189→  let changed = false
   190→  try {
   191→    const handle = bus.beginStroke({ clipId: refs.clipId, frameId: refs.frameId, layerId: refs.layerId })
   192→    changed = fn(handle)
   193→    bus.endStroke(label, changed, ORIGIN)
   194→    return changed
   195→  } catch {
   196→    if (bus.activeStroke) {
   197→      try {
   198→        bus.cancelStroke(ORIGIN)
   199→      } catch {
   200→        /* ignore */
   201→      }
   202→    }
   203→    return false
   204→  }
   205→}

   207→/**
   208→ * Structural sanity for documents entering the editor (Task 4 §5 data-loss
   209→ * protection): at least one clip, and every clip needs a frame to draw on
   210→ * and a layer to draw with. The core v1 shape guard is deliberately loose
   211→ * (serialization format check); THIS is the editor-entry gate. Degenerate
   212→ * rows/files are refused with a toast instead of limping or crashing.
   213→ */
   214→export function isUsableProject(p: unknown): p is Project {
   215→  if (!p || typeof p !== 'object') return false
   216→  const doc = p as Project
   217→  if (!Array.isArray(doc.clips) || doc.clips.length === 0) return false
   218→  return doc.clips.every(
   219→    (c) =>
   220→      c &&
   221→      Array.isArray(c.layers) &&
   222→      c.layers.length > 0 &&
   223→      Array.isArray(c.frames) &&
   224→      c.frames.length > 0
   225→  )
   226→}

   377→let docVersionCounter = 0
   378→let loadEpochCounter = 0

   380→function syncFromBus(extra: Partial<EditorState> = {}): void {
   381→  docVersionCounter += 1
   382→  const stats = bus.historyStats()
   383→  useEditor.setState({
   384→    project: bus.getProject(),
   385→    docVersion: docVersionCounter,
   386→    renderTick: docVersionCounter,
   387→    dirty: bus.isDirty(),
   388→    canUndo: bus.canUndo(),
   389→    canRedo: bus.canRedo(),
   390→    historyUndoDepth: stats.undoDepth,
   391→    historyRedoDepth: stats.redoDepth,
   392→    historyBytes: stats.bytes,
   393→    ...extra,
   394→  })
   395→  revalidatePointers()
   396→}
   397→
   398→/** Keep the active pointers resolvable after undo/load/structural edits. */
   399→function revalidatePointers(): void {
   400→  const s = useEditor.getState()
   401→  const project = bus.getProject()
   402→  let clip = project.clips.find((c) => c.id === s.activeClipId)
   403→  let patch: Partial<EditorState> | null = null
   404→  if (!clip) {
   405→    clip = project.clips[0]
   406→    if (!clip) return
   407→    patch = { activeClipId: clip.id }
   408→  }
   409→  const frameIndex = Math.min(Math.max(0, s.activeFrameIndex), clip.frames.length - 1)
   410→  const frame = clip.frames[frameIndex]
   411→  const layerExists = clip.layers.some((l) => l.id === s.activeLayerId)
   412→  const nextLayerId = layerExists ? s.activeLayerId : (clip.layers[clip.layers.length - 1]?.id ?? '')
   413→  if (frameIndex !== s.activeFrameIndex || !layerExists || patch) {
   414→    patch = { ...(patch ?? {}), activeFrameIndex: frameIndex, activeLayerId: nextLayerId }
   415→  }
   416→  if (patch && (patch.activeFrameIndex !== s.activeFrameIndex || patch.activeLayerId !== s.activeLayerId || patch.activeClipId !== s.activeClipId)) {
   417→    useEditor.setState(patch)
   418→  }
   419→}
