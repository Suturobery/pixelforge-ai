    30→interface ActiveLoc {
    31→  clip: Clip
    32→  frame: Frame
    33→  frameIndex: number
    34→  layer: Layer
    35→}
    36→
    37→function locate(
    38→  project: Project,
    39→  activeClipId: string,
    40→  activeFrameIndex: number,
    41→  activeLayerId: string
    42→): ActiveLoc | null {
    43→  const clip =
    44→    project.clips.find((c) => c.id === activeClipId) ?? project.clips[0]
    45→  if (!clip || clip.frames.length === 0) return null
    46→  const frameIndex = Math.min(Math.max(0, activeFrameIndex), clip.frames.length - 1)
    47→  const frame = clip.frames[frameIndex]
    48→  if (!frame || frame.layers.length === 0) return null
    49→  const layer =
    50→    frame.layers.find((l) => l.id === activeLayerId) ?? frame.layers[frame.layers.length - 1]
    51→  return { clip, frame, frameIndex, layer }
    52→}

    54→/** Immutably replace a layer's buffer deep inside the project. */
    55→function withLayerData(
    56→  project: Project,
    57→  clipId: string,
    58→  frameId: string,
    59→  layerId: string,
    60→  data: Uint8ClampedArray
    61→): Project {
    62→  return {
    63→    ...project,
    64→    clips: project.clips.map((c) =>
    65→      c.id !== clipId
    66→        ? c
    67→        : {
    68→            ...c,
    69→            frames: c.frames.map((f) =>
    70→              f.id !== frameId
    71→                ? f
    72→                : { ...f, layers: f.layers.map((l) => (l.id !== layerId ? l : { ...l, data })) }
    73→            ),
    74→          }
    75→    ),
    76→  }
    77→}

    79→/**
    80→ * Layers are CLIP-WIDE: every frame in a clip shares the same layer stack
    81→ * (same names, same order — Aseprite-style). Each frame keeps its own cell
    82→ * buffers/ids, so structural layer ops must apply to ALL frames of the clip,
    83→ * not just the visible one. Returns the NEW clips array (assign it back to
    84→ * `p.clips` — returning a whole Project here once made `p.clips` a Project
    85→ * object instead of an array, crashing every layer op with
    86→ * "project.clips.find is not a function").
    87→ */
    88→function mapClipFrames(
    89→  project: Project,
    90→  clipId: string,
    91→  fn: (frame: Frame, index: number) => Frame
    92→): Clip[] {
    93→  return project.clips.map((c) =>
    94→    c.id !== clipId ? c : { ...c, frames: c.frames.map((f, i) => fn(f, i)) }
    95→  )
    96→}

   224→  project: initialProject,
   225→  activeClipId: 'cl_initial',
   226→  activeFrameIndex: 0,
   227→  activeLayerId: 'ly_initial',
   228→  tool: 'pencil',
   229→  primary: '#2ce8f5',
   230→  secondary: '#1a1c2c',
   231→  brushSize: 1,
   232→  mirrorX: false,
   233→  zoom: 12,
   234→  panX: 0,
   235→  panY: 0,
   236→  showGrid: true,
   237→  onionSkin: true,
   238→  hover: null,
   239→  playing: false,
   240→  past: [],
   241→  future: [],
   242→  dirty: false,
   243→  savedAt: null,
   244→  autosave: true,
   245→  rightTab: 'layers',
   246→  paletteOpen: true,
   247→  rightOpen: true,
   248→  rightWidth: 320,
   249→  timelineOpen: true,
   250→  renderTick: 0,
   251→
   252→  newProject: (p) =>
   253→    set({
   254→      project: p,
   255→      activeClipId: p.clips[0]?.id ?? '',
   256→      activeFrameIndex: 0,
   257→      activeLayerId: p.clips[0]?.frames[0]?.layers.at(-1)?.id ?? '',
   258→      past: [],
   259→      future: [],
   260→      dirty: true,
   261→      savedAt: null,
   262→      playing: false,
   263→      zoom: 12,
   264→      panX: 0,
   265→      panY: 0,
   266→    }),
   267→
   268→  loadProject: (p) =>
   269→    set({
   270→      project: p,
   271→      activeClipId: p.clips[0]?.id ?? '',
   272→      activeFrameIndex: 0,
   273→      activeLayerId: p.clips[0]?.frames[0]?.layers.at(-1)?.id ?? '',
   274→      past: [],
   275→      future: [],
   276→      dirty: false,
   277→      savedAt: Date.now(),
   278→      playing: false,
   279→      zoom: 12,
   280→      panX: 0,
   281→      panY: 0,
   282→    }),
   283→
   284→  setName: (name) => {
   285→    const s = get()
   286→    set({ project: { ...s.project, name }, dirty: true })
   287→  },
   288→
   289→  update: (mutator, label) => {
   290→    const s = get()
   291→    const before = s.project
   292→    const project = { ...before }
   293→    mutator(project)
   294→    const entry: HistoryEntry = { kind: 'project', label, before, after: project }
   295→    set({
   296→      project,
   297→      past: [...s.past, entry].slice(-useAppSettings.getState().historyLimit),
   298→      future: [],
   299→      dirty: true,
   300→    })
   301→  },
   302→
   303→  beginStroke: () => {
   304→    const s = get()
   305→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   306→    if (!loc) return null
   307→    const { clip, frame, frameIndex, layer } = loc
   308→    const before = layer.data
   309→    const working = new Uint8ClampedArray(before)
   310→    const newFrame: Frame = {
   311→      ...frame,
   312→      layers: frame.layers.map((l) => (l.id === layer.id ? { ...l, data: working } : l)),
   313→    }
   314→    const project: Project = {
   315→      ...s.project,
   316→      clips: s.project.clips.map((c) =>
   317→        c.id !== clip.id
   318→          ? c
   319→          : { ...c, frames: c.frames.map((f, i) => (i === frameIndex ? newFrame : f)) }
   320→      ),
   321→    }
   322→    stroke = { clipId: clip.id, frameId: frame.id, layerId: layer.id, before }
   323→    set({ project, renderTick: s.renderTick + 1 })
   324→    return { width: s.project.width, height: s.project.height, buffer: working }
   325→  },
   326→
   327→  endStroke: (label, changed) => {
   328→    if (!stroke) return
   329→    const st = stroke
   330→    stroke = null
   331→    if (!changed) return
   332→    const s = get()
   333→    const entry: HistoryEntry = { kind: 'pixels', label, ...st, after: st.before }
   334→    // Locate the working buffer in the *current* project (it may have moved if
   335→    // a structural change happened mid-stroke — rare, but stay safe)
   336→    const loc = locate(s.project, st.clipId, s.activeFrameIndex, st.layerId)
   337→    let after = st.before
   338→    if (loc && loc.clip.id === st.clipId && loc.frame.id === st.frameId && loc.layer.id === st.layerId) {
   339→      after = loc.layer.data
   340→    } else {
   341→      const clip = s.project.clips.find((c) => c.id === st.clipId)
   342→      const frame = clip?.frames.find((f) => f.id === st.frameId)
   343→      const layer = frame?.layers.find((l) => l.id === st.layerId)
   344→      if (layer) after = layer.data
   345→      else return
   346→    }
   347→    entry.after = after
   348→    set({
   349→      past: [...s.past, entry].slice(-useAppSettings.getState().historyLimit),
   350→      future: [],
   351→      dirty: true,
   352→    })
   353→  },
   354→
   355→  cancelStroke: () => {
   356→    if (!stroke) return
   357→    const st = stroke
   358→    stroke = null
   359→    const s = get()
   360→    set({
   361→      project: withLayerData(s.project, st.clipId, st.frameId, st.layerId, st.before),
   362→      renderTick: s.renderTick + 1,
   363→    })
   364→  },
   365→
   366→  bump: () => set((s) => ({ renderTick: s.renderTick + 1 })),
   367→
   368→  undo: () => {
   369→    const s = get()
   370→    if (stroke) return
   371→    const past = [...s.past]
   372→    while (past.length > 0) {
   373→      const entry = past.pop() as HistoryEntry
   374→      if (entry.kind === 'pixels') {
   375→        const cur = locate(s.project, entry.clipId, s.activeFrameIndex, entry.layerId)
   376→        const layer = cur && cur.clip.id === entry.clipId && cur.frame.id === entry.frameId && cur.layer.id === entry.layerId
   377→          ? cur.layer
   378→          : s.project.clips
   379→              .find((c) => c.id === entry.clipId)
   380→              ?.frames.find((f) => f.id === entry.frameId)
   381→              ?.layers.find((l) => l.id === entry.layerId)
   382→        if (!layer) continue // structure changed underneath; keep unwinding
   383→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.before)
   384→        const redoEntry: HistoryEntry = { ...entry, before: layer.data, after: entry.after }
   385→        set({ project, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   386→        return
   387→      }
   388→      const redoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: s.project, after: entry.after }
   389→      set({ project: entry.before, past, future: [...s.future, redoEntry].slice(-useAppSettings.getState().historyLimit), dirty: true })
   390→      return
   391→    }
   392→  },
   393→
   394→  redo: () => {
   395→    const s = get()
   396→    if (stroke) return
   397→    const future = [...s.future]
   398→    while (future.length > 0) {
   399→      const entry = future.pop() as HistoryEntry
   400→      if (entry.kind === 'pixels') {
   401→        const layer = s.project.clips
   402→          .find((c) => c.id === entry.clipId)
   403→          ?.frames.find((f) => f.id === entry.frameId)
   404→          ?.layers.find((l) => l.id === entry.layerId)
   405→        if (!layer) continue
   406→        const project = withLayerData(s.project, entry.clipId, entry.frameId, entry.layerId, entry.after)
   407→        const undoEntry: HistoryEntry = { ...entry, before: entry.before, after: layer.data }
   408→        set({ project, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   409→        return
   410→      }
   411→      const undoEntry: HistoryEntry = { kind: 'project', label: entry.label, before: entry.before, after: s.project }
   412→      set({ project: entry.after, past: [...s.past, undoEntry].slice(-useAppSettings.getState().historyLimit), future, dirty: true })
   413→      return
   414→    }
   415→  },
   416→
   417→  // --- structural commands ---
   418→
   419→  addFrame: (copyContent) => {
   420→    const s = get()
   421→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   422→    if (!loc) return
   423→    const insertAt = s.activeFrameIndex + 1
   424→    const newFrame: Frame = {
   425→      id: uid('fr'),
   426→      layers: loc.frame.layers.map((l) => ({
   427→        id: uid('ly'),
   428→        name: l.name,
   429→        visible: l.visible,
   430→        opacity: l.opacity,
   431→        data: copyContent
   432→          ? new Uint8ClampedArray(l.data)
   433→          : new Uint8ClampedArray(s.project.width * s.project.height * 4),
   434→      })),
   435→    }
   436→    get().update((p) => {
   437→      p.clips = p.clips.map((c) =>
   438→        c.id !== loc.clip.id
   439→          ? c
   440→          : { ...c, frames: [...c.frames.slice(0, insertAt), newFrame, ...c.frames.slice(insertAt)] }
   441→      )
   442→    }, copyContent ? 'Duplicate frame' : 'Add frame')
   443→    set({ activeFrameIndex: insertAt, activeLayerId: newFrame.layers.at(-1)?.id ?? '' })
   444→    toast(
   445→      copyContent
   446→        ? { title: `Frame ${insertAt + 1} duplicated` }
   447→        : {
   448→            title: `Frame ${insertAt + 1} added (empty)`,
   449→            description: `The canvas now shows the new frame — frame ${s.activeFrameIndex + 1} and its pixels are safe on the timeline below.`,
   450→          }
   451→    )
   452→  },
   453→
   454→  deleteFrame: (index) => {
   455→    const s = get()
   456→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   457→    if (!loc || loc.clip.frames.length <= 1) return
   458→    const keptName = loc.layer.name // layer ids are per-frame — match by name
   459→    get().update((p) => {
   460→      p.clips = p.clips.map((c) =>
   461→        c.id !== loc.clip.id ? c : { ...c, frames: c.frames.filter((_, i) => i !== index) }
   462→      )
   463→    }, 'Delete frame')
   464→    const len = loc.clip.frames.length - 1
   465→    const nextIdx = Math.min(index, len - 1)
   466→    const nextFrame = loc.clip.frames.filter((_, i) => i !== index)[nextIdx]
   467→    const nextLayer =
   468→      nextFrame?.layers.find((l) => l.name === keptName) ?? nextFrame?.layers.at(-1)
   469→    set({ activeFrameIndex: nextIdx, activeLayerId: nextLayer?.id ?? '' })
   470→  },
   471→
   472→  moveFrame: (from, to) => {
   473→    const s = get()
   474→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   475→    if (!loc) return
   476→    const len = loc.clip.frames.length
   477→    if (to < 0 || to >= len || from === to) return
   478→    get().update((p) => {
   479→      p.clips = p.clips.map((c) => {
   480→        if (c.id !== loc.clip.id) return c
   481→        const frames = [...c.frames]
   482→        const [f] = frames.splice(from, 1)
   483→        frames.splice(to, 0, f)
   484→        return { ...c, frames }
   485→      })
   486→    }, 'Reorder frame')
   487→    set({ activeFrameIndex: to })
   488→  },
   489→
   490→  addLayer: () => {
   491→    const s = get()
   492→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   493→    if (!loc) return
   494→    const name = `layer ${loc.frame.layers.length + 1}`
   495→    // CLIP-WIDE: every frame gets its own cell of the new layer (same name,
   496→    // independent buffer) so the stack looks identical on every frame
   497→    get().update((p) => {
   498→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   499→        ...f,
   500→        layers: [...f.layers, createLayer(s.project.width, s.project.height, name)],
   501→      }))
   502→    }, 'Add layer')
   503→    // grab the id on the *active* frame (ids are per-frame)
   504→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   505→    const newLayerIdOnActiveFrame = cur?.frame.layers.at(-1)?.id ?? ''
   506→    set({ activeLayerId: newLayerIdOnActiveFrame })
   507→    toast({
   508→      title: `Layer "${name}" added`,
   509→      description: `Added to all ${loc.clip.frames.length} frame${loc.clip.frames.length === 1 ? '' : 's'} in this clip.`,
   510→    })
   511→  },
   512→
   513→  deleteLayer: (layerId) => {
   514→    const s = get()
   515→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   516→    if (!loc || loc.frame.layers.length <= 1) return
   517→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   518→    if (idx < 0) return
   519→    // CLIP-WIDE: drop the same slot from every frame
   520→    get().update((p) => {
   521→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   522→        ...f,
   523→        layers: f.layers.filter((_, i) => i !== idx),
   524→      }))
   525→    }, 'Delete layer')
   526→    const next = loc.frame.layers[Math.min(idx, loc.frame.layers.length - 2)]
   527→    set({ activeLayerId: next?.id ?? '' })
   528→  },
   529→
   530→  duplicateLayer: (layerId) => {
   531→    const s = get()
   532→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   533→    if (!loc) return
   534→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   535→    if (idx < 0) return
   536→    const srcName = loc.frame.layers[idx].name
   537→    // CLIP-WIDE: duplicate the slot in every frame (each frame copies its own
   538→    // cell content), inserted directly above the original
   539→    get().update((p) => {
   540→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   541→        const src = f.layers[idx]
   542→        if (!src) return f
   543→        const copy: Layer = {
   544→          ...src,
   545→          id: uid('ly'),
   546→          name: `${srcName} copy`,
   547→          data: new Uint8ClampedArray(src.data),
   548→        }
   549→        return {
   550→          ...f,
   551→          layers: [...f.layers.slice(0, idx + 1), copy, ...f.layers.slice(idx + 1)],
   552→        }
   553→      })
   554→    }, 'Duplicate layer')
   555→    const cur = locate(get().project, s.activeClipId, s.activeFrameIndex, '')
   556→    const copyIdOnActiveFrame = cur?.frame.layers[idx + 1]?.id ?? ''
   557→    set({ activeLayerId: copyIdOnActiveFrame })
   558→  },
   559→
   560→  mergeDown: (layerId) => {
   561→    const s = get()
   562→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   563→    if (!loc) return
   564→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   565→    if (idx <= 0) return // nothing below
   566→    const { width: w, height: h } = s.project
   567→    const bottomId = loc.frame.layers[idx - 1].id
   568→    // CLIP-WIDE: composite the same slot pair on every frame
   569→    get().update((p) => {
   570→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   571→        if (idx >= f.layers.length) return f
   572→        const fTop = f.layers[idx]
   573→        const fBottom = f.layers[idx - 1]
   574→        if (!fTop || !fBottom) return f
   575→        const merged = new Uint8ClampedArray(fBottom.data)
   576→        for (let i = 0; i < w * h; i++) {
   577→          const ta = (fTop.data[i * 4 + 3] / 255) * fTop.opacity
   578→          if (ta >= 1) {
   579→            merged[i * 4] = fTop.data[i * 4]
   580→            merged[i * 4 + 1] = fTop.data[i * 4 + 1]
   581→            merged[i * 4 + 2] = fTop.data[i * 4 + 2]
   582→            merged[i * 4 + 3] = fTop.data[i * 4 + 3]
   583→          } else if (ta > 0) {
   584→            const ba = merged[i * 4 + 3] / 255
   585→            const outA = ta + ba * (1 - ta)
   586→            if (outA === 0) continue
   587→            for (let ch = 0; ch < 3; ch++) {
   588→              merged[i * 4 + ch] = Math.round(
   589→                (fTop.data[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   590→              )
   591→            }
   592→            merged[i * 4 + 3] = Math.round(outA * 255)
   593→          }
   594→        }
   595→        return {
   596→          ...f,
   597→          layers: f.layers.filter((_, i) => i !== idx).map((l, i) => (i === idx - 1 ? { ...l, data: merged } : l)),
   598→        }
   599→      })
   600→    }, 'Merge down')
   601→    set({ activeLayerId: bottomId })
   602→  },
   603→
   604→  moveLayer: (layerId, dir) => {
   605→    const s = get()
   606→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   607→    if (!loc) return
   608→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   609→    const to = idx + dir
   610→    if (idx < 0 || to < 0 || to >= loc.frame.layers.length) return
   611→    // CLIP-WIDE: reorder the same slot on every frame
   612→    get().update((p) => {
   613→      p.clips = mapClipFrames(p, loc.clip.id, (f) => {
   614→        if (to >= f.layers.length) return f
   615→        const layers = [...f.layers]
   616→        const [l] = layers.splice(idx, 1)
   617→        if (!l) return f
   618→        layers.splice(to, 0, l)
   619→        return { ...f, layers }
   620→      })
   621→    }, 'Reorder layer')
   622→    // the moved layer keeps its id on the active frame — selection unchanged
   623→  },
   624→
   625→  setLayerProp: (layerId, patch) => {
   626→    const s = get()
   627→    const loc = locate(s.project, s.activeClipId, s.activeFrameIndex, s.activeLayerId)
   628→    if (!loc) return
   629→    const idx = loc.frame.layers.findIndex((l) => l.id === layerId)
   630→    if (idx < 0) return
   631→    // CLIP-WIDE: name/visibility/opacity apply to the slot on every frame so
   632→    // the panel shows a consistent stack everywhere (eye toggle included)
   633→    get().update((p) => {
   634→      p.clips = mapClipFrames(p, loc.clip.id, (f) => ({
   635→        ...f,
   636→        layers: f.layers.map((l, i) => (i === idx ? { ...l, ...patch } : l)),
   637→      }))
   638→    }, 'Layer property')
   639→  },
   640→
   641→  addClip: (name) => {
   642→    const s = get()
   643→    const clip: Clip = {
   644→      id: uid('cl'),
   645→      name,
   646→      fps: Math.min(60, Math.max(1, useAppSettings.getState().defaultClipFps)),
   647→      loop: true,
   648→      frames: [createFrame(s.project.width, s.project.height, ['layer 1'])],
   649→    }
   650→    get().update((p) => {
   651→      p.clips = [...p.clips, clip]
   652→    }, 'Add clip')
   653→    set({ activeClipId: clip.id, activeFrameIndex: 0, activeLayerId: clip.frames[0].layers[0].id })
   654→  },
   655→
   656→  deleteClip: (clipId) => {
   657→    const s = get()
   658→    if (s.project.clips.length <= 1) return
   659→    const idx = s.project.clips.findIndex((c) => c.id === clipId)
   660→    if (idx < 0) return
   661→    get().update((p) => {
   662→      p.clips = p.clips.filter((c) => c.id !== clipId)
   663→    }, 'Delete clip')
   664→    const next = get().project.clips[Math.min(idx, get().project.clips.length - 1)]
   665→    set({ activeClipId: next.id, activeFrameIndex: 0, activeLayerId: next.frames[0]?.layers.at(-1)?.id ?? '' })
   666→  },
   667→
   668→  setClipProp: (clipId, patch) => {
   669→    const s = get()
   670→    get().update((p) => {
   671→      p.clips = p.clips.map((c) => (c.id === clipId ? { ...c, ...patch } : c))
   672→    }, 'Clip property')
   673→  },
   674→
   675→  addPaletteColor: (hex) => {
   676→    const s = get()
   677→    if (s.project.palette.includes(hex)) return
   678→    get().update((p) => {
   679→      p.palette = [...p.palette, hex]
   680→    }, 'Add palette color')
   681→  },
   682→
   683→  removePaletteColor: (hex) => {
   684→    get().update((p) => {
   685→      p.palette = p.palette.filter((c) => c !== hex)
   686→    }, 'Remove palette color')
   687→  },
   688→
   689→  // --- selection & ui ---
   690→
   691→  selectClip: (id) => {
   692→    const s = get()
   693→    const clip = s.project.clips.find((c) => c.id === id)
   694→    if (!clip) return
   695→    set({
   696→      activeClipId: id,
   697→      activeFrameIndex: Math.min(s.activeFrameIndex, clip.frames.length - 1),
   698→      activeLayerId: clip.frames[Math.min(s.activeFrameIndex, clip.frames.length - 1)]?.layers.at(-1)?.id ?? '',
   699→      playing: false,
   700→    })
   701→  },
   702→
   703→  selectFrame: (index) => {
   704→    const s = get()
   705→    const clip = s.project.clips.find((c) => c.id === s.activeClipId) ?? s.project.clips[0]
   706→    if (!clip) return
   707→    const i = Math.min(Math.max(0, index), clip.frames.length - 1)
   708→    // keep the user's layer context across frames: layers are clip-wide and
   709→    // share names, so match the active layer by name (fall back to the top)
   710→    const prev = clip.frames[Math.min(s.activeFrameIndex, clip.frames.length - 1)]
   711→    const prevName = prev?.layers.find((l) => l.id === s.activeLayerId)?.name
   712→    const nextFrame = clip.frames[i]
   713→    const matched = prevName ? nextFrame?.layers.find((l) => l.name === prevName) : undefined
   714→    set({ activeFrameIndex: i, activeLayerId: matched?.id ?? nextFrame?.layers.at(-1)?.id ?? s.activeLayerId })
   715→  },
   716→
   717→  selectLayer: (id) => set({ activeLayerId: id }),
   718→  setTool: (t) => set({ tool: t }),
   719→  setPrimary: (hex) => set({ primary: hex }),
   720→  setSecondary: (hex) => set({ secondary: hex }),
   721→  swapColors: () => set((s) => ({ primary: s.secondary, secondary: s.primary })),

   734→  setPlaying: (v) => set({ playing: v }),
   735→  setRightTab: (t) => set({ rightTab: t }),

   742→  setProjectId: (id) =>

   750→export function getActiveClip(s: Pick<EditorState, 'project' | 'activeClipId'>): Clip {
   751→  return s.project.clips.find((c) => c.id === s.activeClipId) ?? s.project.clips[0]
   752→}
   753→
   754→export function getActiveFrame(s: Pick<EditorState, 'project' | 'activeClipId' | 'activeFrameIndex'>): Frame {
   755→  const clip = getActiveClip(s)
   756→  return clip.frames[Math.min(s.activeFrameIndex, clip.frames.length - 1)] ?? clip.frames[0]
   757→}
