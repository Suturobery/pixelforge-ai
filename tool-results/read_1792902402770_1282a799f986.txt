     7→// through `pixels.applyOp` (tile-diff history), structural edits through the

    65→export async function computeStateHash(): Promise<string> {
    66→  const p = getBus().getProject()
    67→  const parts: unknown[] = [p.id, p.name, p.width, p.height, p.palette, p.settings]
    68→  for (const clip of p.clips) {
    69→    const celCrcs: Record<string, number> = {}
    70→    for (const frame of clip.frames) {
    71→      for (const [layerId, cel] of Object.entries(frame.cels)) {
    72→        celCrcs[`${frame.id}:${layerId}`] = crc32(cel.buffer as unknown as Uint8Array)
    73→      }
    74→    }
    75→    parts.push({
    76→      id: clip.id,
    77→      name: clip.name,
    78→      fps: clip.fps,
    79→      loop: clip.loop,
    80→      loopMode: clip.loopMode,
    81→      layers: clip.layers,
    82→      parts: clip.parts.length,
    83→      frames: clip.frames.map((f) => ({ id: f.id, durationMs: f.durationMs })),
    84→      cels: celCrcs,
    85→      tags: clip.tags,
    86→    })
    87→  }
    88→  parts.push(p.trash.map((t) => t.id))
    89→  const bytes = new TextEncoder().encode(JSON.stringify(parts))
    90→  return (await sha256Hex(bytes)).slice(0, 32)
    91→}

    97→function fail(code: string, message: string, hint?: string): ToolExecResult {
    98→  return { ok: false, error: { code, message, ...(hint ? { hint } : {}) } }

   101→function isRecord(v: unknown): v is Record<string, unknown> {
   102→  return typeof v === 'object' && v !== null && !Array.isArray(v)
   103→}

   105→function asInt(v: unknown, what: string, min: number, max: number): number {
   106→  if (typeof v !== 'number' || !Number.isFinite(v) || !Number.isInteger(v) || v < min || v > max) {
   107→    throw new CommandError('VALIDATION_FAILED', `${what} must be an integer in [${min}, ${max}]`)
   108→  }
   109→  return v
   110→}
   111→
   112→function optInt(v: unknown, what: string, min: number, max: number): number | undefined {
   113→  return v === undefined ? undefined : asInt(v, what, min, max)
   114→}
   115→
   116→function asHex(v: unknown, what: string): string {
   117→  if (typeof v !== 'string' || !HEX_RE.test(v)) {
   118→    throw new CommandError('VALIDATION_FAILED', `${what} must be a hex color like #rrggbb`)
   119→  }
   120→  return v.toLowerCase()
   121→}
   122→
   123→function asString(v: unknown, what: string, max: number, min = 1): string {
   124→  if (typeof v !== 'string' || v.trim().length < min || v.trim().length > max) {
   125→    throw new CommandError('VALIDATION_FAILED', `${what} must be a string of ${min}–${max} chars`)
   126→  }
   127→  return v.trim()
   128→}

   130→function asRgba(hex: string): RGBA {

   134→function resolveClipRef(project: Project, args: Record<string, unknown>): Clip {
   135→  const clipId = typeof args.clipId === 'string' ? args.clipId : undefined
   136→  const clipName = typeof args.clipName === 'string' ? args.clipName : undefined
   137→  if (clipId) {
   138→    const clip = project.clips.find((c) => c.id === clipId)
   139→    if (!clip) throw new CommandError('NOT_FOUND', `clip "${clipId}" not found`)
   140→    return clip
   141→  }
   142→  if (clipName) {
   143→    const clip = project.clips.find((c) => c.name === clipName)
   144→    if (!clip) throw new CommandError('NOT_FOUND', `clip named "${clipName}" not found`)
   145→    return clip
   146→  }
   147→  const state = useEditor.getState()
   148→  const active = project.clips.find((c) => c.id === state.activeClipId) ?? project.clips[0]
   149→  if (!active) throw new CommandError('INVALID_STATE', 'project has no clips')
   150→  return active
   151→}

   153→function resolveFrameRef(clip: Clip, args: Record<string, unknown>): { frame: Frame; index: number } {
   154→  const frameId = typeof args.frameId === 'string' ? args.frameId : undefined
   155→  if (frameId) {
   156→    const index = clip.frames.findIndex((f) => f.id === frameId)
   157→    if (index < 0) throw new CommandError('NOT_FOUND', `frame "${frameId}" not found in clip "${clip.name}"`)
   158→    return { frame: clip.frames[index]!, index }
   159→  }
   160→  const raw = args.frameIndex
   161→  if (raw !== undefined) {
   162→    const index = asInt(raw, 'frameIndex', 0, Math.max(0, clip.frames.length - 1))
   163→    return { frame: clip.frames[index]!, index }
   164→  }
   165→  const state = useEditor.getState()
   166→  const index = Math.min(Math.max(0, state.activeFrameIndex), clip.frames.length - 1)
   167→  return { frame: clip.frames[index]!, index }
   168→}

   170→function resolveLayerRef(clip: Clip, args: Record<string, unknown>): Layer {
   171→  const layerId = typeof args.layerId === 'string' ? args.layerId : undefined
   172→  if (layerId) {
   173→    const layer = clip.layers.find((l) => l.id === layerId)
   174→    if (!layer) throw new CommandError('NOT_FOUND', `layer "${layerId}" not found in clip "${clip.name}"`)
   175→    return layer
   176→  }
   177→  const layerName = typeof args.layerName === 'string' ? args.layerName : undefined
   178→  if (layerName) {
   179→    const layer = clip.layers.find((l) => l.name === layerName)
   180→    if (!layer) throw new CommandError('NOT_FOUND', `layer named "${layerName}" not found in clip "${clip.name}"`)
   181→    return layer
   182→  }
   183→  const state = useEditor.getState()
   184→  const active = clip.layers.find((l) => l.id === state.activeLayerId) ?? clip.layers[clip.layers.length - 1]
   185→  if (!active) throw new CommandError('INVALID_STATE', `clip "${clip.name}" has no layers`)
   186→  return active
   187→}

   200→function envelope(data: unknown, opts: { undoable?: boolean; stateHash?: string; summary?: string } = {}): ToolExecResult {

   204→function base64FromBytes(bytes: Uint8Array): string {
   205→  let bin = ''
   206→  const CHUNK = 0x8000
   207→  for (let i = 0; i < bytes.length; i += CHUNK) {
   208→    bin += String.fromCharCode(...bytes.subarray(i, i + CHUNK))
   209→  }
   210→  return btoa(bin)
   211→}

   213→async function pngDataUrl(width: number, height: number, rgba: Uint8ClampedArray): Promise<string> {
   214→  const png = await encodePng(width, height, rgba)
   215→  return `data:image/png;base64,${base64FromBytes(png)}`
   216→}

   218→/** Nearest-neighbor downscale (deterministic, interpolation-free). */
   219→function nnScale(rgba: Uint8ClampedArray, w: number, h: number, scale: 0.25 | 0.5 | 1): { rgba: Uint8ClampedArray; w: number; h: number } {
   220→  if (scale === 1) return { rgba, w, h }
   221→  const ow = Math.max(1, Math.floor(w * scale))
   222→  const oh = Math.max(1, Math.floor(h * scale))
   223→  const out = new Uint8ClampedArray(ow * oh * 4)
   224→  for (let y = 0; y < oh; y++) {
   225→    const sy = Math.min(h - 1, Math.floor(y / scale))
   226→    for (let x = 0; x < ow; x++) {
   227→      const sx = Math.min(w - 1, Math.floor(x / scale))
   228→      const si = (sy * w + sx) * 4
   229→      const di = (y * ow + x) * 4
   230→      out[di] = rgba[si]!
   231→      out[di + 1] = rgba[si + 1]!
   232→      out[di + 2] = rgba[si + 2]!
   233→      out[di + 3] = rgba[si + 3]!
   234→    }
   235→  }
   236→  return { rgba: out, w: ow, h: oh }
   237→}

   239→function clipSummary(clip: Clip): Record<string, unknown> {
   240→  return {
   241→    id: clip.id,
   242→    name: clip.name,
   243→    fps: clip.fps,
   244→    loop: clip.loop,
   245→    loopMode: clip.loopMode,
   246→    tags: clip.tags,
   247→    layerCount: clip.layers.length,
   248→    frameCount: clip.frames.length,
   249→  }
   250→}

   252→function clipDetail(clip: Clip): Record<string, unknown> {
   253→  return {
   254→    ...clipSummary(clip),
   255→    layers: clip.layers.map((l) => ({ id: l.id, name: l.name, visible: l.visible, opacity: l.opacity, linked: l.linked })),
   256→    parts: clip.parts.map((p) => ({ id: p.id, name: p.name, layerId: p.layerId, rect: p.rect ?? null, z: p.z })),
   257→    frames: clip.frames.map((f, i) => ({
   258→      id: f.id,
   259→      index: i,
   260→      durationMs: f.durationMs,
   261→      cels: Object.fromEntries(
   262→        Object.entries(f.cels).map(([layerId, cel]) => [
   263→          layerId,
   264→          { linked: cel.linked, hasPixels: !cel.buffer.every((v) => v === 0) },
   265→        ])
   266→      ),
   267→    })),
   268→  }
   269→}

   271→async function framePng(project: Project, clip: Clip, frame: Frame, scale: 0.25 | 0.5 | 1): Promise<string> {
   272→  const composed = composeFrameRgba(project, clip, frame)
   273→  const scaled = nnScale(composed, project.width, project.height, scale)
   274→  return pngDataUrl(scaled.w, scaled.h, scaled.rgba)
   275→}

   277→async function withStateHash(data: unknown, opts: { undoable?: boolean; summary?: string } = {}): Promise<ToolExecResult> {
   278→  return envelope(data, { undoable: opts.undoable, summary: opts.summary, stateHash: await computeStateHash() })
   279→}

   355→  async preview_animation(args) {
   356→    const project = getBus().getProject()
   357→    const clip = resolveClipRef(project, args)
   358→    const maxFrames = optInt(args.maxFrames, 'maxFrames', 1, 64) ?? MAX_PREVIEW_FRAMES
   359→    if (clip.frames.length === 0) return fail('not_found', `clip "${clip.name}" has no frames`)
   360→    const cycle = buildPlaybackCycle(clip)
   361→    const count = Math.min(clip.frames.length, maxFrames)
   362→    const frames: string[] = []
   363→    let bytes = 0
   364→    for (let i = 0; i < count; i++) {
   365→      const dataUrl = await framePng(project, clip, clip.frames[i]!, 1)
   366→      bytes += dataUrl.length
   367→      if (bytes > MAX_PREVIEW_BYTES) {
   368→        return fail('response_too_large', `Preview exceeded ${MAX_PREVIEW_BYTES} bytes after ${i} frames — use get_frame_png per frame or a smaller canvas.`)
   369→      }
   370→      frames.push(dataUrl)
   371→    }
   372→    return withStateHash({
   373→      clipId: clip.id,
   374→      name: clip.name,
   375→      width: project.width,
   376→      height: project.height,
   377→      fps: clip.fps,
   378→      loop: clip.loop,
   379→      loopMode: clip.loopMode,
   380→      cycleMs: cycle.totalMs,
   381→      schedule: cycle.entries,
   382→      renderedFrames: frames.length,
   383→      totalFrames: clip.frames.length,
   384→      frames,
   385→    })
   386→  },

   400→  async create_project(args) {
   401→    const bus = getBus()
   402→    if (bus.isDirty()) {
   403→      return fail(
   404→        'dirty_project',
   405→        'The current project has unsaved changes — creating a new project would discard them.',
   406→        'Ask the user to save (Ctrl+S / Save button), then retry create_project. PixelForge never discards work silently.'
   407→      )
   408→    }
   409→    const name = asString(args.name, 'name', 120)
   410→    const width = asInt(args.width, 'width', 16, 256)
   411→    const height = asInt(args.height, 'height', 16, 256)
   412→    const fps = optInt(args.fps, 'fps', 1, 60) ?? 8
   413→    let palette: string[] | undefined
   414→    if (args.palette !== undefined) {
   415→      if (!Array.isArray(args.palette) || args.palette.length === 0 || args.palette.length > 256) {
   416→        throw new CommandError('VALIDATION_FAILED', 'palette must be a 1–256 hex color array')
   417→      }
   418→      palette = args.palette.map((c) => asHex(c, 'palette color'))
   419→    }
   420→    const project = coreCreateProject({
   421→      name,
   422→      width,
   423→      height,
   424→      fps,
   425→      ...(palette ? { palette } : {}),
   426→    })
   427→    useEditor.getState().loadProject(project, { projectId: null })
   428→    return withStateHash(
   429→      { created: true, name, width, height, fps, clipCount: project.clips.length, frameCount: project.clips[0]?.frames.length ?? 0 },
   430→      { summary: `Created project "${name}" (${width}×${height})` }
   431→    )
   432→  },

   434→  async open_project(args) {
   435→    const bus = getBus()
   436→    if (bus.isDirty()) {
   437→      return fail(
   438→        'dirty_project',
   439→        'The current project has unsaved changes — opening another project would discard them.',
   440→        'Ask the user to save (Ctrl+S / Save button), then retry open_project. PixelForge never discards work silently.'
   441→      )
   442→    }
   443→    const projectId = asString(args.projectId, 'projectId', 128)
   444→    try {
   445→      const res = await fetch(`/api/projects/${encodeURIComponent(projectId)}`, { headers: AGENT_FETCH_HEADERS })
   446→      const json = (await res.json()) as { ok: boolean; project?: unknown; error?: string }
   447→      if (!json.ok || !json.project) return fail('not_found', json.error ?? `project "${projectId}" not found`)

   449→      const { project } = await importLegacyProjectJson(json.project, projectId)
   450→      useEditor.getState().loadProject(project, { projectId })
   451→      return withStateHash(
   452→        { opened: true, projectId, name: project.name, width: project.width, height: project.height, clipCount: project.clips.length },
   453→        { summary: `Opened project "${project.name}"` }
   454→      )
   455→    } catch (err) {
   456→      return fail('persistence_error', err instanceof Error ? err.message : 'Failed to open project')
   457→    }

   460→  async add_frame(args) {
   461→    const project = getBus().getProject()
   462→    const clip = resolveClipRef(project, args)
   463→    const afterIndex = optInt(args.afterIndex, 'afterIndex', -1, Math.max(0, clip.frames.length - 1))
   464→    const copy = args.copyContent === true
   465→    execCommand('frame.add', { clipId: clip.id, ...(afterIndex !== undefined ? { afterIndex } : {}), ...(copy ? { copy: true } : {}) })
   466→    const after = getBus().getProject().clips.find((c) => c.id === clip.id)!
   467→    const newIndex = (afterIndex !== undefined ? afterIndex + 1 : after.frames.length - 1)
   468→    return withStateHash(
   469→      { clipId: clip.id, frameId: after.frames[newIndex]?.id ?? null, frameIndex: newIndex, frameCount: after.frames.length },
   470→      { undoable: true, summary: `${copy ? 'Duplicated' : 'Added'} frame ${newIndex} in "${clip.name}"` }
   471→    )
   472→  },

   474→  async duplicate_frame(args) {
   475→    const project = getBus().getProject()
   476→    const clip = resolveClipRef(project, args)
   477→    const { index } = resolveFrameRef(clip, args)
   478→    execCommand('frame.add', { clipId: clip.id, afterIndex: index, copy: true })
   479→    return withStateHash(
   480→      { clipId: clip.id, frameIndex: index + 1, frameCount: clip.frames.length + 1 },
   481→      { undoable: true, summary: `Duplicated frame ${index} in "${clip.name}"` }
   482→    )

   485→  async reorder_frames(args) {
   486→    const project = getBus().getProject()
   487→    const clip = resolveClipRef(project, args)
   488→    const from = asInt(args.from, 'from', 0, Math.max(0, clip.frames.length - 1))

   564→  async replace_color(args) {
   565→    const project = getBus().getProject()
   566→    const clip = resolveClipRef(project, args)
   567→    const { frame } = resolveFrameRef(clip, args)
   568→    const layer = resolveLayerRef(clip, args)
   569→    const fromHex = asHex(args.from, 'from')
   570→    const toHex = asHex(args.to, 'to')
   571→    const cel = frame.cels[layer.id]
   572→    if (!cel) return fail('not_found', `layer "${layer.name}" has no cel on frame ${frame.id}`)
   573→    const from = asRgba(fromHex)
   574→    const to = asRgba(toHex)
   575→    const src = cel.buffer
   576→    let replaced = 0
   577→    const out = new Uint8ClampedArray(src)
   578→    const exact = fromHex.length > 4 // #rrggbb(+aa) matches alpha too
   579→    for (let i = 0; i < src.length; i += 4) {
   580→      const match = exact
   581→        ? src[i] === from[0] && src[i + 1] === from[1] && src[i + 2] === from[2] && src[i + 3] === from[3]
   582→        : src[i] === from[0] && src[i + 1] === from[1] && src[i + 2] === from[2] && (from[3] === 255 ? src[i + 3] === 255 : true)
   583→      if (match) {
   584→        out[i] = to[0]
   585→        out[i + 1] = to[1]
   586→        out[i + 2] = to[2]
   587→        out[i + 3] = to[3]
   588→        replaced++
   589→      }
   590→    }
   591→    if (replaced > 0) {
   592→      execCommand('pixels.applyOp', { clipId: clip.id, frameId: frame.id, layerId: layer.id, ops: [{ op: 'setBuffer' as const, buffer: out }] })
   593→    }
   594→    return withStateHash(
   595→      { replaced, clipId: clip.id, frameId: frame.id, layerId: layer.id },
   596→      { undoable: replaced > 0, summary: `Replaced ${replaced} px of ${fromHex} with ${toHex} on "${clip.name}" / "${layer.name}"` }
   597→    )

   852→  async accept_candidate(args) {
   853→    const { aiCandidateStore, registerAcceptedTarget } = await import('./ai/candidates')
   854→    const { fitBufferToCanvas } = await import('@pixelforge/core')
   855→    const id = asString(args.candidateId, 'candidateId', 128)
   856→    const mode = args.mode === 'frame' ? 'frame' : args.mode === 'layer' ? 'layer' : null
   857→    if (!mode) throw new CommandError('VALIDATION_FAILED', "mode must be 'layer' or 'frame'")
   858→    const candidate = aiCandidateStore.getState().get(id)
   859→    if (!candidate) {
   860→      return fail('not_found', `Candidate "${id}" is not in the review cache.`, 'Call get_candidates for the current ids.')
   861→    }
   862→    if (!candidate.normalized) {
   863→      return fail('not_ready', 'Candidate has no normalized buffer yet — wait for normalization or call apply_normalize.')
   864→    }

   868→    try {
   869→      buffer = fitBufferToCanvas(candidate.normalized.buffer, candidate.normalized.w, candidate.normalized.h, project.width, project.height)
   870→    } catch {
   871→      return fail(
   872→        'dimension_mismatch',
   873→        `Normalized candidate is ${candidate.normalized.w}×${candidate.normalized.h} but the canvas is ${project.width}×${project.height} — re-normalize with apply_normalize first.`,
   874→      )
   875→    }
   876→    const clip = resolveClipRef(project, args)
   877→    const { frame } = resolveFrameRef(clip, args)
   878→    const name = `AI: ${candidate.prompt.slice(0, 24)}`
   879→
   880→    if (mode === 'layer') {
   881→      execTransaction(`Import AI candidate as layer (${id})`, () => {
   882→        getBus().execute('layer.add', { clipId: clip.id, name }, 'mcp')
   883→        const created = getBus().getProject().clips.find((c) => c.id === clip.id)!.layers.slice(-1)[0]!
   884→        getBus().execute(
   885→          'pixels.applyOp',
   886→          { clipId: clip.id, frameId: frame.id, layerId: created.id, ops: [{ op: 'setBuffer', buffer }] },
   887→          'mcp',
   888→        )
   889→      })
   890→      const after = getBus().getProject().clips.find((c) => c.id === clip.id)!
   891→      const created = after.layers[after.layers.length - 1]!
   892→      // undo/delete of this layer unlocks the candidate again (reconcile)
   893→      registerAcceptedTarget(id, { kind: 'layer', clipId: clip.id, targetId: created.id })
   894→      aiCandidateStore.getState().patch(id, { status: 'accepted' })
   895→      return withStateHash(
   896→        { candidateId: id, mode, clipId: clip.id, frameId: frame.id, layerId: created.id, layerName: created.name },
   897→        { undoable: true, summary: `Accepted AI candidate "${id}" as layer "${created.name}"` },
   898→      )
   899→    }
   900→
   901→    // frame mode: new empty frame right after the target + buffer on the frame's top layer
   902→    const frameIndex = clip.frames.findIndex((f) => f.id === frame.id)
   903→    let newFrameId: string | null = null
   904→    execTransaction(`Import AI candidate as frame (${id})`, () => {
   905→      getBus().execute('frame.add', { clipId: clip.id, afterIndex: frameIndex, copy: false }, 'mcp')
   906→      const after = getBus().getProject().clips.find((c) => c.id === clip.id)!
   907→      const newFrame = after.frames[frameIndex + 1]
   908→      if (!newFrame) throw new CommandError('INVALID_STATE', 'frame add produced no frame')
   909→      newFrameId = newFrame.id
   910→      const layer = resolveLayerRef(after, { ...args, layerId: undefined, layerName: undefined })
   911→      getBus().execute(
   912→        'pixels.applyOp',
   913→        { clipId: clip.id, frameId: newFrame.id, layerId: layer.id, ops: [{ op: 'setBuffer', buffer }] },
   914→        'mcp',
   915→      )
   916→    })
   917→    if (newFrameId) {
   918→      // undo/delete of this frame unlocks the candidate again (reconcile)
   919→      registerAcceptedTarget(id, { kind: 'frame', clipId: clip.id, targetId: newFrameId })
   920→    }
   921→    aiCandidateStore.getState().patch(id, { status: 'accepted' })
   922→    return withStateHash(
   923→      { candidateId: id, mode, clipId: clip.id, frameId: newFrameId, frameIndex: frameIndex + 1 },
   924→      { undoable: true, summary: `Accepted AI candidate "${id}" as frame ${frameIndex + 2} in "${clip.name}"` },
   925→    )
   926→  },
   927→
   928→  async reject_candidate(args) {
   929→    const { aiCandidateStore } = await import('./ai/candidates')
   930→    const id = asString(args.candidateId, 'candidateId', 128)
   931→    const candidate = aiCandidateStore.getState().get(id)
   932→    if (!candidate) {
   933→      return fail('not_found', `Candidate "${id}" is not in the review cache.`)
   934→    }
   935→    aiCandidateStore.getState().patch(id, { status: 'rejected' })
   936→    aiCandidateStore.getState().remove(id)
   937→    return withStateHash(
   938→      { candidateId: id, rejected: true },
   939→      { undoable: false, summary: `Rejected AI candidate "${id}" (review cache only — document untouched)` },
   940→    )
   941→  },

   943→  async apply_normalize(args) {
   944→    const { aiCandidateStore } = await import('./ai/candidates')
   945→    const { normalizeV2, parseHexColor } = await import('@pixelforge/core')
   946→    const id = asString(args.candidateId, 'candidateId', 128)
   947→    const candidate = aiCandidateStore.getState().get(id)
   948→    if (!candidate) {
   949→      return fail('not_found', `Candidate "${id}" is not in the review cache.`)
   950→    }
   951→    if (!candidate.originalSrc) {
   952→      return fail('not_ready', 'Candidate has no source image to normalize.')
   953→    }
   954→    const project = getBus().getProject()
   955→    // options (all optional; defaults mirror the AI panel)
   956→    const width = optInt(args.width, 'width', 8, 512) ?? project.width
   957→    const height = optInt(args.height, 'height', 8, 512) ?? project.height
   958→    const removeBg = args.removeBg !== false
   959→    const bgTolerance = optInt(args.bgTolerance, 'bgTolerance', 0, 255) ?? 48
   960→    const modeRaw = args.paletteMode
   961→    const paletteMode = modeRaw === 'project' || modeRaw === 'auto' || modeRaw === 'off' ? modeRaw : 'auto'
   962→    const maxColors = optInt(args.maxColors, 'maxColors', 2, 64) ?? 16
   963→    const outlineCleanup = args.outlineCleanup !== false
   964→    let palette: string[] = project.palette
   965→    if (Array.isArray(args.palette)) {
   966→      palette = (args.palette as unknown[]).filter((x): x is string => typeof x === 'string' && parseHexColor(x) !== null)
   967→    }
   968→    // decode the ORIGINAL image (PNG data URL) — the executor runs in the
   969→    // renderer, same process as the store; the pipeline is the deterministic
   970→    // core function (identical output to the panel's worker path)
   971→    const raw = candidate.originalSrc
   972→    const b64 = raw.includes(',') ? raw.slice(raw.indexOf(',') + 1) : raw
   973→    let binary: Uint8Array
   974→    try {
   975→      const bin = atob(b64)
   976→      binary = new Uint8Array(bin.length)
   977→      for (let i = 0; i < bin.length; i++) binary[i] = bin.charCodeAt(i)
   978→    } catch {
   979→      return fail('validation_failed', 'candidate source is not valid base64')
   980→    }
   981→    let decoded
   982→    try {
   983→      decoded = await decodePng(binary)
   984→    } catch {
   985→      return fail('validation_failed', 'candidate source is not a decodable PNG (use a PNG candidate or normalize from the AI panel)')
   986→    }
   987→    const res = normalizeV2(decoded.rgba, decoded.width, decoded.height, {
   988→      targetW: width,
   989→      targetH: height,
   990→      removeBg,
   991→      bgTolerance,
   992→      paletteMode,
   993→      palette,
   994→      maxColors,
   995→      outlineCleanup,
   996→    })
   997→    const updated = aiCandidateStore.getState().patch(id, {
   998→      normalized: { buffer: res.buffer, w: res.width, h: res.height, colors: res.stats.colorsOut },
   999→      status: 'review',
  1000→      error: null,
  1001→    })
  1002→    return withStateHash(
  1003→      {
  1004→        candidateId: id,
  1005→        width: res.width,
  1006→        height: res.height,
  1007→        stats: res.stats,
  1008→        previewPng: await pngDataUrl(res.width, res.height, res.buffer),
  1009→        ok: !!updated,
  1010→      },
  1011→      { undoable: false, summary: `Re-normalized candidate "${id}" → ${res.width}×${res.height} (${res.stats.paletteUsed} palette, ${res.stats.colorsOut} colors)` },
  1012→    )
  1013→  },
  1014→
  1015→  async undo() {
  1016→    const result = getBus().undo()
  1017→    return withStateHash(
  1018→      { undid: result.ok, label: result.label ?? null, skipped: result.skipped ?? 0, canUndo: getBus().canUndo(), canRedo: getBus().canRedo() },
  1019→      { summary: result.ok ? `Undid: ${result.label ?? 'change'}` : 'Nothing to undo' }

  1032→/** Shared executor for flip/rotate/shift (single-cel transforms). */
  1033→async function runCelTransform(
  1034→  args: Record<string, unknown>,
  1035→  makeOp: (rect: { x: number; y: number; w: number; h: number } | undefined) => Record<string, unknown>,
  1036→  verb: string
  1037→): Promise<ToolExecResult> {
  1038→  const project = getBus().getProject()
  1039→  const clip = resolveClipRef(project, args)
  1040→  const { frame } = resolveFrameRef(clip, args)
  1041→  const layer = resolveLayerRef(clip, args)
  1042→  let rect: { x: number; y: number; w: number; h: number } | undefined
  1043→  if (args.rect !== undefined) {
  1044→    if (!isRecord(args.rect)) throw new CommandError('VALIDATION_FAILED', 'rect must be {x, y, w, h}')
  1045→    rect = {
  1046→      x: asInt(args.rect.x, 'rect.x', 0, 1e6),
  1047→      y: asInt(args.rect.y, 'rect.y', 0, 1e6),
  1048→      w: asInt(args.rect.w, 'rect.w', 1, 4096),
  1049→      h: asInt(args.rect.h, 'rect.h', 1, 4096),
  1050→    }
  1051→  }
  1052→  execCommand('pixels.applyOp', { clipId: clip.id, frameId: frame.id, layerId: layer.id, ops: [makeOp(rect)] })
  1053→  return withStateHash(
  1054→    { clipId: clip.id, frameId: frame.id, layerId: layer.id },
  1055→    { undoable: true, summary: `${verb} on "${clip.name}" / "${layer.name}"` }
  1056→  )
  1057→}

  1063→export const EXECUTOR_TOOL_NAMES = Object.keys(handlers).sort()
  1064→
  1065→export async function executeMcpTool(tool: string, args: unknown, ctx: ToolExecContext): Promise<ToolExecResult> {
  1066→  const handler = handlers[tool]
  1067→  if (!handler) {
  1068→    return fail('unknown_tool', `Tool "${tool}" is not implemented by this editor (it may be deferred to a later phase).`)
  1069→  }
  1070→  const bus = getBus()
  1071→  if (bus.activeStroke) {
  1072→    return fail('editor_busy', 'A manual stroke is in progress — retry in a moment.', 'The tool will run as soon as the pointer interaction ends.')
  1073→  }
  1074→  const safeArgs = isRecord(args) ? args : {}
  1075→  // optimistic staleness guard on every L1 mutation
  1076→  const expected = typeof safeArgs.expectedStateHash === 'string' ? safeArgs.expectedStateHash : null
  1077→  if (expected) {
  1078→    const current = await computeStateHash()
  1079→    if (current !== expected) {
  1080→      return fail('stale_document', 'The document changed since you read it (expectedStateHash mismatch).', 'Re-read get_project_state and retry with the fresh stateHash.')
  1081→    }
  1082→  }
  1083→  try {
  1084→    return await handler(safeArgs, ctx)
  1085→  } catch (err) {
  1086→    if (err instanceof CommandError) {
  1087→      return fail(err.code.toLowerCase(), err.message)
  1088→    }
  1089→    return fail('internal_error', err instanceof Error ? err.message : 'Tool execution failed')
  1090→  }
  1091→}
