    53→/** Small PNG thumbnail of a normalized buffer (history strip). */
    54→function makeThumb(buffer: Uint8ClampedArray, w: number, h: number): string {
    55→  const canvas = document.createElement('canvas')
    56→  canvas.width = w
    57→  canvas.height = h
    58→  const ctx = canvas.getContext('2d')
    59→  if (!ctx) return ''
    60→  ctx.putImageData(new ImageData(new Uint8ClampedArray(buffer), w, h), 0, 0)
    61→  return canvas.toDataURL('image/png')
    62→}

    64→function fmtCost(usd: number | null): string {
    65→  if (usd === null) return 'cost n/a'
    66→  if (usd === 0) return 'cost $0 (cache)'
    67→  return `≈ $${usd.toFixed(3)} (estimate)`
    68→}
