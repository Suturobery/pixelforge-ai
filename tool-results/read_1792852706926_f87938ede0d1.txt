   183→export const frameAdd: CommandDefinition<{
   184→  clipId: string;
   185→  afterIndex?: number;
   186→  copy?: boolean;
   187→}> = {
   188→  type: 'frame.add',
   189→  defaultLabel: 'Add frame',
   190→  validate(payload, ctx) {
   191→    const p = assertObject(payload, 'frame.add');
   192→    const { clipId } = parseRef(p);
   193→    const clip = findClip(ctx.project, clipId);
   194→    if (!clip) failNotFound(`frame.add: clip "${clipId}" not found`);
   195→    const afterIndex = getOptionalInt(p, 'afterIndex', 'frame.add', -1, clip.frames.length - 1);
   196→    if (getOptionalInt(p, 'afterIndex', 'frame.add') !== undefined && afterIndex === undefined) {
   197→      fail('frame.add: bad afterIndex');
   198→    }
   199→  },
   200→  apply(payload, ctx) {
   201→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   202→    const insertAt =
   203→      payload.afterIndex !== undefined
   204→        ? Math.min(Math.max(0, payload.afterIndex + 1), clip.frames.length)
   205→        : clip.frames.length;
   206→    const srcIndex =
   207→      payload.afterIndex !== undefined
   208→        ? Math.min(Math.max(0, payload.afterIndex), clip.frames.length - 1)
   209→        : clip.frames.length - 1;
   210→    const frame: Frame = payload.copy && clip.frames[srcIndex]
   211→      ? duplicateFrame(clip.frames[srcIndex], clip.layers) // duplicate preserves cels exactly (§2.7)
   212→      : createFrame(ctx.project.width, ctx.project.height, clip.layers);
   213→    const frames = [...clip.frames.slice(0, insertAt), frame, ...clip.frames.slice(insertAt)];
   214→    return {
   215→      project: mapClip(ctx.project, clip.id, (c) => ({ ...c, frames })),
   216→      label: payload.copy ? 'Duplicate frame' : 'Add frame',
   217→    };
   218→  },

   336→  type: 'layer.delete',
   337→  defaultLabel: 'Delete layer',
   338→  validate(payload, ctx) {
   339→    const p = assertObject(payload, 'layer.delete');
   340→    const { clipId, layerId } = parseLayerRef(p);
   341→    const clip = findClip(ctx.project, clipId);
   342→    if (!clip) failNotFound(`layer.delete: clip "${clipId}" not found`);
   343→    if (!findLayer(clip, layerId)) failNotFound(`layer.delete: layer "${layerId}" not found`);

   349→    const celsByFrameId: Record<string, Cel> = {};
   350→    for (const f of clip.frames) {
   351→      const cel = f.cels[layer.id];
   352→      if (cel) celsByFrameId[f.id] = cel; // linked frames keep the shared reference
   353→    }
   354→    const project = mapClip(ctx.project, clip.id, (c) => ({
   355→      ...c,
   356→      layers: c.layers.filter((l) => l.id !== layer.id),

   358→        if (!f.cels[layer.id]) return f;
   359→        const cels = { ...f.cels };
   360→        delete cels[layer.id];
   361→        return { ...f, cels };
   362→      }),
   363→    }));
   364→    return {
   365→      project: pushTrash(project, {
   366→        id: newTrashId(),
   367→        label: `Layer "${layer.name}" of "${clip.name}"`,
   368→        deletedAt: Date.now(),
   369→        item: { kind: 'layer', clipId: clip.id, layerIndex: index, layer, celsByFrameId },

   371→      label: 'Delete layer',

   414→/**
   415→ * Alpha compositing of the top cel onto the one below, per frame (ported
   416→ * from the prototype store's mergeDown). The top layer is removed; the
   417→ * operation is undoable via the structural snapshot, so it is not
   418→ * trash-backed (documented deviation from delete semantics).
   419→ */
   420→export const layerMergeDown: CommandDefinition<{ clipId: string; layerId: string }> = {
   421→  type: 'layer.mergeDown',
   422→  defaultLabel: 'Merge down',
   423→  validate(payload, ctx) {
   424→    const p = assertObject(payload, 'layer.mergeDown');
   425→    const { clipId, layerId } = parseLayerRef(p);
   426→    const clip = findClip(ctx.project, clipId);
   427→    if (!clip) failNotFound(`layer.mergeDown: clip "${clipId}" not found`);
   428→    const found = findLayer(clip, layerId);
   429→    if (!found) failNotFound(`layer.mergeDown: layer "${layerId}" not found`);
   430→    if (found.index === 0) fail('layer.mergeDown: nothing below the bottom layer');
   431→    if (found.layer.linked) {
   432→      fail('layer.mergeDown: merging a linked layer is not supported (unlink first — C2)');
   433→    }
   434→  },
   435→  apply(payload, ctx) {
   436→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   437→    const { index } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   438→    const { width, height } = ctx.project;
   439→    const project = mapClip(ctx.project, clip.id, (c) => ({
   440→      ...c,
   441→      layers: c.layers.filter((_, i) => i !== index),
   442→      frames: c.frames.map((f) => {
   443→        const top = f.cels[payload.layerId];
   444→        const bottomLayer = c.layers[index - 1];
   445→        if (!bottomLayer) return f;
   446→        const bottom = f.cels[bottomLayer.id];
   447→        if (!top && !bottom) return f;
   448→        // Missing cel = transparent (C2) — merge treats it as empty.
   449→        const merged = bottom
   450→          ? new Uint8ClampedArray(bottom.buffer)
   451→          : new Uint8ClampedArray(width * height * 4);
   452→        if (top) {
   453→          for (let i = 0; i < width * height; i++) {
   454→            const ta = (top.buffer[i * 4 + 3] / 255) * (top.linked ? 1 : (c.layers[index]?.opacity ?? 1));
   455→            if (ta >= 1) {
   456→              merged[i * 4] = top.buffer[i * 4];
   457→              merged[i * 4 + 1] = top.buffer[i * 4 + 1];
   458→              merged[i * 4 + 2] = top.buffer[i * 4 + 2];
   459→              merged[i * 4 + 3] = top.buffer[i * 4 + 3];
   460→            } else if (ta > 0) {
   461→              const ba = merged[i * 4 + 3] / 255;
   462→              const outA = ta + ba * (1 - ta);
   463→              if (outA === 0) continue;
   464→              for (let ch = 0; ch < 3; ch++) {
   465→                merged[i * 4 + ch] = Math.round(
   466→                  (top.buffer[i * 4 + ch] * ta + merged[i * 4 + ch] * ba * (1 - ta)) / outA
   467→                );
   468→              }
   469→              merged[i * 4 + 3] = Math.round(outA * 255);
   470→            }
   471→          }
   472→        }
   473→        return { ...f, cels: { ...f.cels, [bottomLayer.id]: { buffer: merged, linked: false } } };
   474→      }),
   475→    }));
   476→    // Remove the top layer's cels everywhere.
   477→    const cleaned = mapClip(project, clip.id, (c) => ({
   478→      ...c,
   479→      frames: c.frames.map((f) => {
   480→        if (!f.cels[payload.layerId]) return f;
   481→        const cels = { ...f.cels };
   482→        delete cels[payload.layerId];
   483→        return { ...f, cels };
   484→      }),
   485→    }));
   486→    return { project: cleaned, label: 'Merge down' };
   487→  },
   488→};

   491→  type: 'layer.move',
   492→  defaultLabel: 'Reorder layer',
   493→  validate(payload, ctx) {

   513→    return { project: mapClip(ctx.project, clip.id, (c) => ({ ...c, layers })), label: 'Reorder layer' };

   529→    const clip = findClip(ctx.project, clipId);
   530→    if (!clip) failNotFound(`layer.setProps: clip "${clipId}" not found`);
   531→    if (!findLayer(clip, layerId)) failNotFound(`layer.setProps: layer "${layerId}" not found`);
   532→    const patch = assertObject(p.patch, 'layer.setProps patch');
   533→    const name = getOptionalString(patch, 'name', 'layer.setProps');
   534→    if (name !== undefined && name.length > 120) fail('layer.setProps: name too long');
   535→    const opacity = patch.opacity;
   536→    if (opacity !== undefined && (typeof opacity !== 'number' || opacity < 0 || opacity > 1)) {
   537→      fail('layer.setProps: opacity must be 0..1');
   538→    }
   539→  },
   540→  apply(payload, ctx) {
   541→    const { clipId, layerId, patch } = payload;
   542→    return {
   543→      project: mapClip(ctx.project, clipId, (c) => ({
   544→        ...c,
   545→        layers: c.layers.map((l) =>
   546→          l.id === layerId
   547→            ? {
   548→                ...l,
   549→                ...(patch.name !== undefined ? { name: patch.name } : {}),
   550→                ...(patch.visible !== undefined ? { visible: patch.visible } : {}),
   551→                ...(patch.opacity !== undefined ? { opacity: patch.opacity } : {}),
   552→              }
   553→            : l
   554→        ),
   555→      })),
   556→      label: 'Layer properties',
   557→    };
   558→  },

   561→/**
   562→ * Linked-cel engine switch (C2/C19 — P0 engine support; UI toggle is P2).
   563→ * ON  → pick the anchor frame's cel (first frame holding one) and share it
   564→ *        across ALL frames (content of other frames collapses to the anchor).
   565→ * OFF → break sharing: every frame gets an independent deep copy.
   566→ */
   567→export const layerSetLinked: CommandDefinition<{ clipId: string; layerId: string; linked: boolean }> = {
   568→  type: 'layer.setLinked',
   569→  defaultLabel: 'Linked cel',
   570→  validate(payload, ctx) {
   571→    const p = assertObject(payload, 'layer.setLinked');
   572→    const { clipId, layerId } = parseLayerRef(p);
   573→    const clip = findClip(ctx.project, clipId);
   574→    if (!clip) failNotFound(`layer.setLinked: clip "${clipId}" not found`);
   575→    if (!findLayer(clip, layerId)) failNotFound(`layer.setLinked: layer "${layerId}" not found`);
   576→    if (typeof p.linked !== 'boolean') fail('layer.setLinked: "linked" must be a boolean');
   577→  },
   578→  apply(payload, ctx) {
   579→    const clip = findClip(ctx.project, payload.clipId) as Clip;
   580→    const { layer } = findLayer(clip, payload.layerId) as { layer: Layer; index: number };
   581→    if (payload.linked === layer.linked) {
   582→      return { project: ctx.project, label: 'Linked cel' };
   583→    }
   584→    if (payload.linked) {
   585→      // Anchor = first frame that has a cel (else first frame → create).
   586→      const anchorFrame = clip.frames.find((f) => f.cels[layer.id]) ?? clip.frames[0];
   587→      if (!anchorFrame) fail('layer.setLinked: clip has no frames');
   588→      const existing = anchorFrame.cels[layer.id];
   589→      const shared: Cel = existing
   590→        ? { buffer: existing.buffer, linked: true }
   591→        : { buffer: new Uint8ClampedArray(ctx.project.width * ctx.project.height * 4), linked: true };
   592→      const withFlag = mapClip(ctx.project, clip.id, (c) => ({
   593→        ...c,
   594→        layers: c.layers.map((l) => (l.id === layer.id ? { ...l, linked: true } : l)),
   595→      }));
   596→      return {
   597→        project: setCelEverywhere(withFlag, clip.id, layer.id, shared),
   598→        label: 'Link cel (shared)',
   599→      };
   600→    }
   601→    // Unlink: give every frame its own copy of the current shared buffer.
   602→    const project = mapClip(ctx.project, clip.id, (c) => ({
   603→      ...c,
   604→      layers: c.layers.map((l) => (l.id === layer.id ? { ...l, linked: false } : l)),
   605→    }));
   606→    const withCopies = mapClip(project, clip.id, (c) => ({
   607→      ...c,
   608→      frames: c.frames.map((f) => {
   609→        const cel = f.cels[layer.id];
   610→        if (!cel) return f;
   611→        return { ...f, cels: { ...f.cels, [layer.id]: { buffer: cloneBuffer(cel.buffer), linked: false } } };
   612→      }),
   613→    }));
   614→    return { project: withCopies, label: 'Unlink cel' };
   615→  },
   616→};

   664→export const trashRestore: CommandDefinition<{ entryId: string }> = {
   665→  type: 'trash.restore',
   666→  defaultLabel: 'Restore from trash',
   667→  validate(payload, ctx) {
   668→    const p = assertObject(payload, 'trash.restore');
   669→    const entryId = getString(p, 'entryId', 'trash.restore');
   670→    if (!ctx.project.trash.some((t) => t.id === entryId)) {
   671→      failNotFound(`trash.restore: entry "${entryId}" not found`);
   672→    }
   673→  },
   674→  apply(payload, ctx) {
   675→    const taken = takeTrash(ctx.project, payload.entryId);
   676→    if (!taken) failNotFound('trash.restore: entry vanished');
   677→    const { entry, project: without } = taken;
   678→    const item = entry.item;
   679→    if (item.kind === 'frame') {
   680→      const clip = findClip(without, item.clipId);
   681→      if (!clip) failNotFound('trash.restore: owning clip no longer exists');
   682→      const index = Math.min(Math.max(0, item.frameIndex), clip.frames.length);
   683→      return {
   684→        project: mapClip(without, clip.id, (c) => ({
   685→          ...c,
   686→          frames: [...c.frames.slice(0, index), item.frame, ...c.frames.slice(index)],
   687→        })),
   688→        label: 'Restore frame',
   689→      };
   690→    }
   691→    if (item.kind === 'layer') {
   692→      const clip = findClip(without, item.clipId);
   693→      if (!clip) failNotFound('trash.restore: owning clip no longer exists');
   694→      const index = Math.min(Math.max(0, item.layerIndex), clip.layers.length);
   695→      const layers = [...clip.layers.slice(0, index), item.layer, ...clip.layers.slice(index)];
   696→      const restored = mapClip(without, clip.id, (c) => ({ ...c, layers }));
   697→      // Restore cels to the frames they came from (missing frame ids skipped).
   698→      const withCels = mapClip(restored, clip.id, (c) => ({
   699→        ...c,
   700→        frames: c.frames.map((f) => {
   701→          const cel = item.celsByFrameId[f.id];
   702→          if (!cel) return f;
   703→          return { ...f, cels: { ...f.cels, [item.layer.id]: cel } };
   704→        }),
   705→      }));
   706→      return { project: withCels, label: 'Restore layer' };
   707→    }
   708→    const index = Math.min(Math.max(0, item.clipIndex), without.clips.length);
   709→    const clips = [...without.clips.slice(0, index), item.clip, ...without.clips.slice(index)];
   710→    return { project: { ...without, clips }, label: 'Restore clip' };
   711→  },
   712→};
